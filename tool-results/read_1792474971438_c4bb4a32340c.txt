   572→   572→  hsv: { h: number; s: number; v: number }; // h:0-360, s/v:0-100

   574→   574→}

   579→   579→  wheelThickness?: number; // default 24
   580→   580→  wheelSpacing?: number;   // default 8
   581→   581→  hue?: number;            // default 210 (0-360)
   582→   582→  color?: string;          // default "#4287f5" (overrides hue)
   583→   583→  rotateShape?: boolean;   // default false
   584→   584→}
   585→   585→
   586→   586→class ColorWheelPicker {
   587→   587→  constructor(container: HTMLElement, options?: ColorPickerOptions);
   588→   588→  setColor(color: string): void;
   589→   589→  getColor(): ColorOutput;
   590→   590→  setShape(shape: Shape): void;
   591→   591→  setSize(size: number): void;
   592→   592→  setWheelThickness(thickness: number): void;
   593→   593→  setWheelSpacing(spacing: number): void;
   594→   594→  setHue(hue: number): void;
   595→   595→  setRotateShape(rotate: boolean): void;
   596→   596→  on(event: "input" | "change", handler: (c: ColorOutput) => void): void;

   600→   600→
