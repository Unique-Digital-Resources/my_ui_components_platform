   472→function clampNum(v: number, min: number, max: number): number {
   473→  if (Number.isNaN(v)) return min;
   474→  return v < min ? min : v > max ? max : v;
   475→}
   476→
   477→function formatNum(v: number): string {
   478→  const rounded = Math.round(v * 100) / 100;
   479→  if (Number.isInteger(rounded)) return String(rounded);
   480→  return rounded.toFixed(2);
   481→}

   545→/**
   546→ * A number input that defers committing its value to the parent until the
   547→ * user blurs the field or presses Enter. While focused, the user can type
   548→ * freely (e.g. "33") without each keystroke being snapped / clamped back by
   549→ * the underlying component (which would make typing multi-digit numbers
   550→ * impossible when step snapping is on).
   551→ *
   552→ * The displayed value tracks the `value` prop while unfocused, and tracks
   553→ * the local draft while focused.
   554→ */
   555→function DeferredNumberInput({
   556→  label,
   557→  value,
   558→  onCommit,
   559→  min,
   560→  max,
   561→}: {
   562→  label: string;
   563→  value: number;
   564→  onCommit: (v: number) => void;
   565→  min?: number;
   566→  max?: number;
   567→}) {
   568→  const [draft, setDraft] = React.useState<string>(String(value));
   569→  const [focused, setFocused] = React.useState(false);
   570→
   571→  // Sync the draft from the external value whenever the input is not focused
   572→  // (so external changes — e.g. canvas drags, reset — are reflected).
   573→  React.useEffect(() => {
   574→    if (!focused) setDraft(String(value));
   575→  }, [value, focused]);
   576→
   577→  const commit = React.useCallback(() => {
   578→    const v = Number(draft);
   579→    if (Number.isFinite(v)) {
   580→      onCommit(v);
   581→    } else {
   582→      // Reset to the last known good value on invalid input.
   583→      setDraft(String(value));
   584→    }
   585→  }, [draft, value, onCommit]);
   586→
   587→  return (
   588→    <div className="relative">
   589→      <input
   590→        type="number"
   591→        value={focused ? draft : String(value)}
   592→        min={min}
   593→        max={max}
   594→        onChange={(e) => setDraft(e.target.value)}
