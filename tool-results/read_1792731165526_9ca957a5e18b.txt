   162→function countsByColumn(items: KanbanItem[], columns: KanbanColumn[]): { id: string; title: string; count: number; color?: string }[] {
   163→  return columns.map((c) => ({
   164→    id: c.id,
   165→    title: c.title,
   166→    color: c.color,
   167→    count: items.filter((it) => it.columnId === c.id).length,
   168→  }));
