     1→     1→/**
     2→     2→ * AnimationTrackEditor — a timeline-based animation track editor (vanilla TS).
     3→     3→ *
     4→     4→ * Inspired by Godot's animation timeline + DCC tools. Supports two track types:
     5→     5→ *   - **Keyframe tracks**: discrete keyframes with values and transition paths.
     6→     6→ *   - **Clip tracks**: resizable boxes (for audio, image, video clips).
     7→     7→ *
     8→     8→ * Features:
     9→     9→ *   - Ruler with tick marks + time labels (seconds)
    10→    10→ *   - Playhead with scrubbing (click/drag ruler)
    11→    11→ *   - Playback: play, pause, reverse, loop, speed control
    12→    12→ *   - Zoom (pixels per second) + horizontal pan
    13→    13→ *   - Keyframe drag (with snapping), selection, add/remove
    14→    14→ *   - Clip drag + resize (both edges), selection, add/remove
    15→    15→ *   - Track management: add/remove, visibility, lock, reorder
    16→    16→ *   - Grouped tracks (by owner)
    17→    17→ *   - Keyboard: Space (play/pause), Left/Right (step), Home/End
    18→    18→ *   - Events for all changes
    19→    19→ */
    20→    20→
    21→    21→// ---- Public types ----
    22→    22→
    23→    23→export type TrackType = "keyframe" | "clip";
    24→    24→
    25→    25→export interface Keyframe {
    26→    26→  id: string;
    27→    27→  time: number; // seconds
    28→    28→  value: number;
    29→    29→  /** Transition curve to the NEXT keyframe. */
    30→    30→  transition?: "linear" | "ease" | "ease-in" | "ease-out" | "step" | "hold";
    31→    31→  selected?: boolean;
    32→    32→}
    33→    33→
    34→    34→export interface Clip {
    35→    35→  id: string;
    36→    36→  start: number; // seconds
    37→    37→  end: number;   // seconds
    38→    38→  label?: string;
    39→    39→  color?: string;
    40→    40→  selected?: boolean;
    41→    41→}
    42→    42→
    43→    43→export interface Track {
    44→    44→  id: string;
    45→    45→  name: string;
    46→    46→  type: TrackType;
    47→    47→  owner?: string;
    48→    48→  keyframes?: Keyframe[];
    49→    49→  clips?: Clip[];
    50→    50→  visible?: boolean;
    51→    51→  locked?: boolean;
    52→    52→  color?: string;
    53→    53→}
    54→    54→
    55→    55→export interface AnimationTrackEditorOptions {
    56→    56→  tracks?: Track[];
    57→    57→  duration?: number;     // total timeline duration in seconds
    58→    58→  pixelsPerSecond?: number; // zoom (px/s)
    59→    59→  playhead?: number;     // current playhead position (seconds)
    60→    60→  playing?: boolean;
    61→    61→  loop?: boolean;
    62→    62→  speed?: number;        // playback speed multiplier
    63→    63→  rowHeight?: number;
    64→    64→  snapInterval?: number; // seconds; 0 = no snap
    65→    65→  showRuler?: boolean;
    66→    66→  showToolbar?: boolean;
    67→    67→  readOnly?: boolean;
    68→    68→}

    72→    72→  trackId?: string;

    77→    77→  value?: number;
    78→    78→  oldValue?: number;
    79→    79→  playhead?: number;
    80→    80→  playing?: boolean;
    81→    81→  pixelsPerSecond?: number;
    82→    82→  source?: "pointer" | "keyboard" | "programmatic" | "playback";
    83→    83→  reason?: string;
    84→    84→}

   452→   452→function clamp(n: number, min: number, max: number): number {
   453→   453→  return Math.max(min, Math.min(max, n));
   454→   454→}
   455→   455→
   456→   456→function snapTime(t: number, snap: number): number {
   457→   457→  if (!snap || snap <= 0) return t;
   458→   458→  return Math.round(t / snap) * snap;
   459→   459→}

   865→   865→  private drawRuler(): void {
   866→   866→    if (!this.rulerCanvas) return;
   867→   867→    const canvas = this.rulerCanvas;
   868→   868→    const rect = canvas.getBoundingClientRect();
   869→   869→    const dpr = window.devicePixelRatio || 1;
   870→   870→    canvas.width = rect.width * dpr;
   871→   871→    canvas.height = rect.height * dpr;
   872→   872→    const ctx = canvas.getContext("2d")!;
   873→   873→    ctx.scale(dpr, dpr);
   874→   874→
   875→   875→    const w = rect.width;
   876→   876→    const h = rect.height;
   877→   877→    const pps = this.opts.pixelsPerSecond;
   878→   878→    const duration = this.opts.duration;
   879→   879→
   880→   880→    // Background
   881→   881→    ctx.fillStyle = "#28282E";
   882→   882→    ctx.fillRect(0, 0, w, h);
   883→   883→
   884→   884→    // Determine tick interval (aim for ~60px between ticks)
   885→   885→    const targetSpacing = 60;
   886→   886→    const rawInterval = targetSpacing / pps;
   887→   887→    // Round to nice numbers: 0.1, 0.25, 0.5, 1, 2, 5, 10, 30, 60
   888→   888→    const niceIntervals = [0.1, 0.25, 0.5, 1, 2, 5, 10, 30, 60];
   889→   889→    let interval = niceIntervals[0];
   890→   890→    for (const ni of niceIntervals) {
   891→   891→      if (ni >= rawInterval) { interval = ni; break; }
   892→   892→      interval = ni;
   893→   893→    }
   894→   894→
   895→   895→    // Draw ticks + labels
   896→   896→    ctx.strokeStyle = "#555560";
   897→   897→    ctx.fillStyle = "#888894";
   898→   898→    ctx.font = "9px ui-monospace, monospace";
   899→   899→    ctx.textAlign = "center";
   900→   900→    ctx.textBaseline = "middle";
   901→   901→
   902→   902→    for (let t = 0; t <= duration; t += interval) {
   903→   903→      const x = t * pps;
   904→   904→      if (x > w + 10) break;
   905→   905→      // Major tick
   906→   906→      ctx.beginPath();
   907→   907→      ctx.moveTo(x, h - 8);
   908→   908→      ctx.lineTo(x, h);
   909→   909→      ctx.lineWidth = 1;
   910→   910→      ctx.stroke();
   911→   911→      // Label
   912→   912→      ctx.fillText(this.formatTime(t), x, h / 2);
   913→   913→      // Minor ticks
   914→   914→      const minorStep = interval / 5;
   915→   915→      for (let i = 1; i < 5; i++) {
   916→   916→        const mx = (t + minorStep * i) * pps;
   917→   917→        if (mx > w) break;
   918→   918→        ctx.beginPath();
   919→   919→        ctx.moveTo(mx, h - 4);
   920→   920→        ctx.lineTo(mx, h);
   921→   921→        ctx.lineWidth = 0.5;
   922→   922→        ctx.stroke();
   923→   923→      }
   924→   924→    }
   925→   925→  }

   932→   932→    const duration = this.opts.duration;

   934→   934→    const rawInterval = targetSpacing / pps;

   936→   936→    let interval = niceIntervals[0];
   937→   937→    for (const ni of niceIntervals) {

   940→   940→    }

  1024→  1024→      if (!this.opts.playing || this._destroyed) { this.playbackRAF = null; return; }

  1028→  1028→      if (newTime >= this.opts.duration) {
  1029→  1029→        if (this.opts.loop) newTime = 0;
  1030→  1030→        else { newTime = this.opts.duration; this.pause(); }
  1031→  1031→      }
  1032→  1032→      this.opts.playhead = newTime;
  1033→  1033→      this.updatePlayhead();
  1034→  1034→      this.emit("playback_tick", { playhead: newTime, source: "playback" });
  1035→  1035→      this.playbackRAF = requestAnimationFrame(tick);
  1036→  1036→    };

  1038→  1038→  }

  1382→  1382→  private onKeyDown(e: KeyboardEvent): void {
  1383→  1383→    if (this.opts.readOnly) return;
  1384→  1384→    let handled = true;
  1385→  1385→    switch (e.key) {
  1386→  1386→      case " ":
  1387→  1387→        e.preventDefault();
  1388→  1388→        this.togglePlay();
  1389→  1389→        break;
  1390→  1390→      case "ArrowLeft":
  1391→  1391→        this.setPlayhead(this.opts.playhead - (e.shiftKey ? 1 : 0.1), "keyboard");
  1392→  1392→        break;
  1393→  1393→      case "ArrowRight":
  1394→  1394→        this.setPlayhead(this.opts.playhead + (e.shiftKey ? 1 : 0.1), "keyboard");
  1395→  1395→        break;
  1396→  1396→      case "Home":
  1397→  1397→        this.setPlayhead(0, "keyboard");
  1398→  1398→        break;
  1399→  1399→      case "End":
  1400→  1400→        this.setPlayhead(this.opts.duration, "keyboard");
  1401→  1401→        break;
  1402→  1402→      case "Delete":
  1403→  1403→      case "Backspace":
  1404→  1404→        // Remove selected keyframes + clips
  1405→  1405→        for (const track of this.tracks) {
  1406→  1406→          if (track.keyframes) {
  1407→  1407→            const toRemove = track.keyframes.filter(k => k.selected).map(k => k.id);
  1408→  1408→            for (const id of toRemove) this.removeKeyframe(track.id, id);
  1409→  1409→          }
  1410→  1410→          if (track.clips) {
  1411→  1411→            const toRemove = track.clips.filter(c => c.selected).map(c => c.id);
  1412→  1412→            for (const id of toRemove) this.removeClip(track.id, id);
  1413→  1413→          }
  1414→  1414→        }
  1415→  1415→        break;
  1416→  1416→      case "Escape":
  1417→  1417→        this.clearSelection();
  1418→  1418→        break;
  1419→  1419→      default:
  1420→  1420→        handled = false;
  1421→  1421→    }
  1422→  1422→    if (handled) e.preventDefault();
  1423→  1423→  }
