    69→// ---- Date helpers (pure, no Date object to avoid TZ issues) ----
    70→
    71→function isLeap(y: number): boolean {
    72→  return (y % 4 === 0 && y % 100 !== 0) || y % 400 === 0;
    73→}
    74→
    75→function daysInMonth(y: number, m: number): number {
    76→  return [31, isLeap(y) ? 29 : 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31][m - 1];
    77→}
    78→
    79→function dayOfWeek(y: number, m: number, d: number): number {
    80→  // 0=Sunday … 6=Saturday (Zeller-ish via Date, but we normalize to local)
    81→  const dt = new Date(y, m - 1, d);
    82→  return dt.getDay();
    83→}
    84→
    85→function dateKey(d: CalendarDate): string {
    86→  return `${d.year}-${String(d.month).padStart(2, "0")}-${String(d.day).padStart(2, "0")}`;
    87→}
    88→
    89→function sameDate(a: CalendarDate | null, b: CalendarDate | null): boolean {
    90→  if (!a || !b) return false;
    91→  return a.year === b.year && a.month === b.month && a.day === b.day;
    92→}
    93→
    94→function compareDate(a: CalendarDate, b: CalendarDate): number {
    95→  if (a.year !== b.year) return a.year - b.year;
    96→  if (a.month !== b.month) return a.month - b.month;
    97→  return a.day - b.day;
    98→}
    99→
   100→function isBetween(d: CalendarDate, start: CalendarDate, end: CalendarDate): boolean {
   101→  return compareDate(d, start) >= 0 && compareDate(d, end) <= 0;
   102→}
   103→
   104→function addMonths(d: CalendarDate, n: number): CalendarDate {
   105→  const totalMonths = (d.year * 12 + (d.month - 1)) + n;
   106→  const y = Math.floor(totalMonths / 12);
   107→  const m = (totalMonths % 12) + 1;
   108→  const maxDay = daysInMonth(y, m);
   109→  return { year: y, month: m, day: Math.min(d.day, maxDay) };
   110→}
   111→
   112→function addDays(d: CalendarDate, n: number): CalendarDate {
   113→  let { year, month, day } = d;
   114→  day += n;
   115→  while (day > daysInMonth(year, month)) {
   116→    day -= daysInMonth(year, month);
   117→    month++;
   118→    if (month > 12) { month = 1; year++; }
   119→  }
   120→  while (day < 1) {
   121→    month--;
   122→    if (month < 1) { month = 12; year--; }
   123→    day += daysInMonth(year, month);
   124→  }
   125→  return { year, month, day };
   126→}
   127→
   128→function today(): CalendarDate {
   129→  const d = new Date();
   130→  return { year: d.getFullYear(), month: d.getMonth() + 1, day: d.getDate() };
   131→}
   132→
   133→// ISO week number
   134→function weekNumber(d: CalendarDate): number {
   135→  const dt = new Date(d.year, d.month - 1, d.day);
   136→  dt.setHours(0, 0, 0, 0);
   137→  dt.setDate(dt.getDate() + 3 - ((dt.getDay() + 6) % 7));
   138→  const week1 = new Date(dt.getFullYear(), 0, 4);
   139→  return 1 + Math.round(((dt.getTime() - week1.getTime()) / 86400000 - 3 + ((week1.getDay() + 6) % 7)) / 7);
   140→}

   155→interface GridCell {
   156→  date: CalendarDate;
   157→  isOutside: boolean;
   158→  isToday: boolean;
   159→  isSelected: boolean;
   160→  isRangeStart: boolean;
   161→  isRangeEnd: boolean;
   162→  isInRange: boolean;
   163→  isDisabled: boolean;
   164→  isFocused: boolean;
   165→  isPreviewed: boolean;
   166→  weekNumber?: number;
   167→}
   168→
   169→function buildGrid(
   170→  viewDate: CalendarDate,
   171→  weekStart: WeekStart,
   172→  showOutside: boolean,
   173→  selectionMode: SelectionMode,
   174→  value: CalendarDate | null,
   175→  selectedDates: CalendarDate[],
   176→  rangeStart: CalendarDate | null,
   177→  rangeEnd: CalendarDate | null,
   178→  minDate: CalendarDate | null,
   179→  maxDate: CalendarDate | null,
   180→  disabledDates: Set<string>,
   181→  focusedDate: CalendarDate | null,
   182→  previewEnd: CalendarDate | null,
   183→  showWeekNumbers: boolean,
   184→): GridCell[][] {
   185→  const rows: GridCell[][] = [];
   186→  const firstOfMonth = { year: viewDate.year, month: viewDate.month, day: 1 };
   187→  const firstDayOfWeek = dayOfWeek(firstOfMonth.year, firstOfMonth.month, firstOfMonth.day);
   188→  const offset = (firstDayOfWeek - weekStart + 7) % 7;
   189→  const gridStart = addDays(firstOfMonth, -offset);
   190→  const todayDate = today();
   191→
   192→  for (let week = 0; week < 6; week++) {
   193→    const row: GridCell[] = [];
   194→    for (let col = 0; col < 7; col++) {
   195→      const date = addDays(gridStart, week * 7 + col);
   196→      const isOutside = date.month !== viewDate.month || date.year !== viewDate.year;
   197→      if (isOutside && !showOutside) {
   198→        row.push({
   199→          date, isOutside: true, isToday: false, isSelected: false,
   200→          isRangeStart: false, isRangeEnd: false, isInRange: false,
   201→          isDisabled: true, isFocused: false, isPreviewed: false,
   202→        });
   203→        continue;
   204→      }
   205→      const dk = dateKey(date);
   206→      const isSelected =
   207→        (selectionMode === "single" && sameDate(date, value)) ||
   208→        (selectionMode === "multiple" && selectedDates.some((s) => sameDate(s, date)));
   209→      const isRangeStart = sameDate(date, rangeStart);
   210→      const isRangeEnd = sameDate(date, rangeEnd);
   211→      const isInRange =
   212→        selectionMode === "range" && rangeStart && rangeEnd &&
   213→        isBetween(date, rangeStart, rangeEnd) && !isRangeStart && !isRangeEnd;
   214→      const isPreviewed =
   215→        selectionMode === "range" && rangeStart && !rangeEnd && previewEnd &&
   216→        ((compareDate(previewEnd, rangeStart) >= 0 && isBetween(date, rangeStart, previewEnd) && !isRangeStart) ||
   217→         (compareDate(previewEnd, rangeStart) < 0 && isBetween(date, previewEnd, rangeStart) && !sameDate(date, rangeStart)));
   218→      const isDisabled =
   219→        (minDate && compareDate(date, minDate) < 0) ||
   220→        (maxDate && compareDate(date, maxDate) > 0) ||
   221→        disabledDates.has(dk);
   222→      row.push({
   223→        date, isOutside, isToday: sameDate(date, todayDate),
   224→        isSelected, isRangeStart, isRangeEnd, isInRange,
   225→        isDisabled, isFocused: sameDate(date, focusedDate),
   226→        isPreviewed, weekNumber: showWeekNumbers && col === 0 ? weekNumber(date) : undefined,
   227→      });
   228→    }
   229→    rows.push(row);
   230→  }
   231→  return rows;
   232→}

   273→  private root: HTMLDivElement;
   274→  private opts: Required<CalendarOptions>;
   275→  private listeners = new Map<string, Set<Handler>>();
   276→  private focusedDate: CalendarDate | null = null;
   277→  private previewEnd: CalendarDate | null = null;
   278→  private _destroyed = false;
   279→  /** CustomButton instances for each day cell, tracked for per-instance prop updates. */
   280→  private cellButtons: { button: CustomButton; dateKey: string }[] = [];
   281→
   282→  constructor(container: HTMLElement, options: CalendarOptions = {}) {
   283→    this.container = container;
   284→    this.opts = {
   285→      selectionMode: options.selectionMode ?? "single",
   286→      value: options.value ?? null,
   287→      selectedDates: options.selectedDates ?? [],
   288→      rangeStart: options.rangeStart ?? null,
   289→      rangeEnd: options.rangeEnd ?? null,
   290→      viewDate: options.viewDate ?? today(),
   291→      minDate: options.minDate ?? null,
   292→      maxDate: options.maxDate ?? null,
   293→      disabledDates: options.disabledDates ?? [],

   298→      showWeekNumbers: options.showWeekNumbers ?? false,
   299→      showNavigation: options.showNavigation ?? true,
   300→      showFooter: options.showFooter ?? false,
   301→      readOnly: options.readOnly ?? false,
   302→      disabled: options.disabled ?? false,
   303→      allowClear: options.allowClear ?? true,
   304→      allowToday: options.allowToday ?? true,

   311→    this.root.className = "cal";

   318→    this.attachListeners();

   320→
   321→  // ---- Public API (commands per spec §9) ----
   322→
   323→  setProperty<K extends keyof CalendarOptions>(name: K, value: CalendarOptions[K]): void {
   324→    (this.opts as any)[name] = value;
   325→    this.render();
   326→  }
   327→
   328→  navigate(direction: "prev" | "next"): void {
   329→    this.opts.viewDate = addMonths(this.opts.viewDate, direction === "next" ? 1 : -1);
   330→    this.render();
   331→    this.emit("view_changed", { viewDate: this.opts.viewDate, reason: "navigate" });
   332→  }
   333→
   334→  navigateTo(date: CalendarDate): void {
   335→    this.opts.viewDate = { ...date };
   336→    this.render();
   337→    this.emit("view_changed", { viewDate: this.opts.viewDate, reason: "navigate_to" });
   338→  }
   339→
   340→  selectDate(date: CalendarDate, source: "pointer" | "keyboard" | "programmatic" = "programmatic"): void {
   341→    if (this.opts.readOnly || this.opts.disabled) return;
   342→    if (this.isDisabled(date)) {
   343→      this.emit("validation_failed", { date, reason: "disabled" });
   344→      return;
   345→    }
   346→    const prev = this.getSnapshot();
   347→    switch (this.opts.selectionMode) {
   348→      case "none": break;
   349→      case "single":
   350→        this.opts.value = sameDate(date, this.opts.value) && this.opts.allowClear ? null : date;
   351→        this.focusedDate = date;
   352→        break;
   353→      case "multiple": {
   354→        const idx = this.opts.selectedDates.findIndex((d) => sameDate(d, date));
   355→        if (idx >= 0) this.opts.selectedDates.splice(idx, 1);
   356→        else this.opts.selectedDates.push(date);
   357→        this.focusedDate = date;
   358→        break;
   359→      }
   360→      case "range":
   361→        if (!this.opts.rangeStart || (this.opts.rangeStart && this.opts.rangeEnd)) {
   362→          this.opts.rangeStart = date;
   363→          this.opts.rangeEnd = null;
   364→          this.previewEnd = null;
   365→          this.emit("range_changed", { startDate: date, endDate: null, complete: false });
   366→        } else {
   367→          if (compareDate(date, this.opts.rangeStart) < 0) {
   368→            this.opts.rangeEnd = this.opts.rangeStart;
   369→            this.opts.rangeStart = date;
   370→          } else {
   371→            this.opts.rangeEnd = date;
   372→          }
   373→          this.previewEnd = null;
   374→          this.emit("range_changed", { startDate: this.opts.rangeStart, endDate: this.opts.rangeEnd, complete: true });
   375→        }
   376→        this.focusedDate = date;
   377→        break;
   378→    }
   379→    this.render();
   380→    this.emit("date_selected", { date, previousValue: prev, source });
   381→    this.emit("selection_changed", { selection: this.getSnapshot(), previousSelection: prev, selectionMode: this.opts.selectionMode });
   382→  }
   383→
   384→  setSelection(selection: CalendarDate | CalendarDate[] | { start: CalendarDate; end: CalendarDate } | null): void {
   385→    if (this.opts.readOnly || this.opts.disabled) return;
   386→    if (selection === null) { this.clearSelection(); return; }
   387→    if (this.opts.selectionMode === "single" && !Array.isArray(selection) && !(selection as any).start) {
   388→      this.opts.value = selection as CalendarDate;
   389→    } else if (this.opts.selectionMode === "multiple" && Array.isArray(selection)) {
   390→      this.opts.selectedDates = [...selection];
   391→    } else if (this.opts.selectionMode === "range" && (selection as any).start) {
   392→      this.opts.rangeStart = (selection as any).start;
   393→      this.opts.rangeEnd = (selection as any).end;
   394→    }
   395→    this.render();
   396→    this.emit("selection_changed", { selection: this.getSnapshot(), selectionMode: this.opts.selectionMode });
   397→  }
   398→
   399→  clearSelection(): void {
   400→    if (!this.opts.allowClear) return;
   401→    const prev = this.getSnapshot();
   402→    this.opts.value = null;
   403→    this.opts.selectedDates = [];
   404→    this.opts.rangeStart = null;
   405→    this.opts.rangeEnd = null;
   406→    this.previewEnd = null;
   407→    this.render();
   408→    this.emit("selection_changed", { selection: null, previousSelection: prev, selectionMode: this.opts.selectionMode });
   409→  }
   410→
   411→  focusDate(date: CalendarDate): void {
   412→    this.focusedDate = date;
   413→    if (date.month !== this.opts.viewDate.month || date.year !== this.opts.viewDate.year) {
   414→      this.opts.viewDate = { year: date.year, month: date.month, day: 1 };
   415→    }
   416→    this.render();
   417→    this.emit("focused_date_changed", { date, previousDate: null });
   418→  }
   419→
   420→  getValue(): CalendarDate | null { return this.opts.value; }
   421→  getSelectedDates(): CalendarDate[] { return [...this.opts.selectedDates]; }
   422→  getRange(): { start: CalendarDate | null; end: CalendarDate | null } {
   423→    return { start: this.opts.rangeStart, end: this.opts.rangeEnd };
   424→  }
   425→  getViewDate(): CalendarDate { return { ...this.opts.viewDate }; }
   426→
   427→  on(event: string, handler: Handler): void {
   428→    let set = this.listeners.get(event);
   429→    if (!set) { set = new Set(); this.listeners.set(event, set); }
   430→    set.add(handler);
   431→  }
   432→  off(event: string, handler: Handler): void { this.listeners.get(event)?.delete(handler); }
   433→
   434→  destroy(): void {
   435→    if (this._destroyed) return;
   436→    this._destroyed = true;
   437→    this.listeners.clear();
   438→    if (this.root.parentNode) this.root.parentNode.removeChild(this.root);
   439→  }

   441→  // ---- Internal ----
   442→
   443→  private isDisabled(date: CalendarDate): boolean {
   444→    const dk = dateKey(date);
   445→    return (
   446→      (this.opts.minDate && compareDate(date, this.opts.minDate) < 0) ||
   447→      (this.opts.maxDate && compareDate(date, this.opts.maxDate) > 0) ||
   448→      this.opts.disabledDates.some((d) => dateKey(d) === dk)
   449→    );
   450→  }
   451→
   452→  private getSnapshot(): any {
   453→    switch (this.opts.selectionMode) {
   454→      case "single": return this.opts.value;
   455→      case "multiple": return [...this.opts.selectedDates];
   456→      case "range": return { start: this.opts.rangeStart, end: this.opts.rangeEnd };
   457→      default: return null;
   458→    }
   459→  }
   460→
   461→  private emit(event: string, payload: CalendarEventPayload): void {
   462→    const set = this.listeners.get(event);
   463→    if (!set) return;
   464→    for (const h of Array.from(set)) { try { h(payload); } catch (e) { console.error(e); } }
   465→  }
   466→
   467→  private attachListeners(): void {
   468→    this.root.addEventListener("keydown", (e) => this.onKeyDown(e));
   469→  }
   470→
   471→  private onKeyDown(e: KeyboardEvent): void {
   472→    if (this.opts.disabled) return;
   473→    if (!this.focusedDate) this.focusedDate = today();
   474→    let handled = true;
   475→    switch (e.key) {
   476→      case "ArrowLeft": this.focusedDate = addDays(this.focusedDate, -1); break;
   477→      case "ArrowRight": this.focusedDate = addDays(this.focusedDate, 1); break;
   478→      case "ArrowUp": this.focusedDate = addDays(this.focusedDate, -7); break;
   479→      case "ArrowDown": this.focusedDate = addDays(this.focusedDate, 7); break;
   480→      case "Home": {
   481→        const dow = dayOfWeek(this.focusedDate.year, this.focusedDate.month, this.focusedDate.day);
   482→        const offset = (dow - this.opts.weekStart + 7) % 7;
   483→        this.focusedDate = addDays(this.focusedDate, -offset); break;
   484→      }
   485→      case "End": {
   486→        const dow = dayOfWeek(this.focusedDate.year, this.focusedDate.month, this.focusedDate.day);
   487→        const offset = (this.opts.weekStart + 6 - dow + 7) % 7;
   488→        this.focusedDate = addDays(this.focusedDate, offset); break;
   489→      }
   490→      case "PageUp": this.focusedDate = addMonths(this.focusedDate, e.shiftKey ? -12 : -1); break;
   491→      case "PageDown": this.focusedDate = addMonths(this.focusedDate, e.shiftKey ? 12 : 1); break;
   492→      case "Enter":
   493→      case " ":
   494→        this.selectDate(this.focusedDate, "keyboard");
   495→        handled = true;
   496→        break;
   497→      case "Escape":
   498→        if (this.opts.selectionMode === "range" && this.opts.rangeStart && !this.opts.rangeEnd) {
   499→          this.opts.rangeStart = null;
   500→          this.previewEnd = null;
   501→          this.render();
   502→        }
   503→        break;
   504→      default: handled = false;
   505→    }
   506→    if (handled) {
   507→      e.preventDefault();
   508→      if (this.focusedDate.month !== this.opts.viewDate.month || this.focusedDate.year !== this.opts.viewDate.year) {
   509→        this.opts.viewDate = { year: this.focusedDate.year, month: this.focusedDate.month, day: 1 };
   510→        this.emit("view_changed", { viewDate: this.opts.viewDate, reason: "keyboard" });
   511→      }
   512→      this.render();
   513→      this.emit("focused_date_changed", { date: this.focusedDate });
   514→    }
   515→  }
   516→
   517→  private render(): void {
   518→    if (this._destroyed) return;
   519→    const disabledSet = new Set(this.opts.disabledDates.map(dateKey));
   520→    const grid = buildGrid(
   521→      this.opts.viewDate, this.opts.weekStart, this.opts.showOutsideDates,
   522→      this.opts.selectionMode, this.opts.value, this.opts.selectedDates,
   523→      this.opts.rangeStart, this.opts.rangeEnd, this.opts.minDate, this.opts.maxDate,
   524→      disabledSet, this.focusedDate, this.previewEnd, this.opts.showWeekNumbers,
   525→    );
   526→    const wkNames = WEEKDAY_NAMES[this.opts.weekdayFormat];
   527→    const wkHeaders: string[] = [];
   528→    for (let i = 0; i < 7; i++) wkHeaders.push(wkNames[(this.opts.weekStart + i) % 7]);
   529→    const monthName = MONTH_NAMES[this.opts.viewDate.month - 1];
   530→    const sz = this.opts.cellSize;
   531→
   532→    let html = "";
   533→
   534→    // Header
   535→    if (this.opts.showNavigation) {
   536→      html += `<div class="cal-header">`;
   537→      html += `<button class="cal-nav-btn" data-act="prev" aria-label="Previous month">‹</button>`;
   538→      html += `<span class="cal-title">${monthName} ${this.opts.viewDate.year}</span>`;
   539→      html += `<button class="cal-nav-btn" data-act="next" aria-label="Next month">›</button>`;
   540→      html += `</div>`;
   541→    }
   542→
   543→    // Weekday header
   544→    html += `<div class="cal-weekdays">`;
   545→    if (this.opts.showWeekNumbers) html += `<div class="cal-weekday"></div>`;
   546→    for (const w of wkHeaders) html += `<div class="cal-weekday">${w}</div>`;
   547→    html += `</div>`;
   548→
   549→    // Grid
   550→    html += `<div class="cal-grid">`;
   551→    for (const row of grid) {
   552→      if (this.opts.showWeekNumbers && row[0].weekNumber !== undefined) {
   553→        html += `<div class="cal-weekday" style="display:flex;align-items:center;justify-content:center;">${row[0].weekNumber}</div>`;
   554→      }
   555→      for (const cell of row) {
   556→        const classes = ["cal-cell"];
   557→        if (cell.isOutside) classes.push("cal-outside");
   558→        if (cell.isToday && this.opts.showToday) classes.push("cal-today");
   559→        if (cell.isSelected) classes.push("cal-selected");
   560→        if (cell.isRangeStart) classes.push("cal-range-start");
   561→        if (cell.isRangeEnd) classes.push("cal-range-end");
   562→        if (cell.isInRange) classes.push("cal-in-range");
   563→        if (cell.isPreviewed) classes.push("cal-preview");
   564→        if (cell.isDisabled) classes.push("cal-disabled");
   565→        if (cell.isFocused) classes.push("cal-focused");
   566→        const dk = dateKey(cell.date);
   567→        const label = `${cell.date.month}/${cell.date.day}/${cell.date.year}`;
   568→        html += `<button class="${classes.join(" ")}" data-date="${dk}" data-o="${cell.isOutside?1:0}" aria-label="${label}" style="width:${sz}px;height:${sz}px;"${cell.isDisabled ? " disabled" : ""}>${cell.date.day}</button>`;
   569→      }
   570→    }
   571→    html += `</div>`;
   572→
   573→    // Footer
   574→    if (this.opts.showFooter) {
   575→      html += `<div class="cal-footer">`;
   576→      if (this.opts.allowToday) html += `<button class="cal-foot-btn" data-act="today">Today</button>`;
   577→      if (this.opts.allowClear) html += `<button class="cal-foot-btn" data-act="clear">Clear</button>`;
   578→      html += `</div>`;
   579→    }
   580→
   581→    this.root.innerHTML = html;
   582→
   583→    // Day cells are always rendered as CustomButton instances
   584→    this.applyCustomButtonCells();
   585→
   586→    // Wire up events
   587→    this.root.querySelectorAll("[data-act]").forEach((el) => {
   588→      const act = (el as HTMLElement).dataset.act;
   589→      el.addEventListener("click", (e) => {
   590→        e.preventDefault();
   591→        if (act === "prev") this.navigate("prev");
   592→        else if (act === "next") this.navigate("next");
   593→        else if (act === "today") {
   594→          const t = today();
   595→          this.navigateTo(t);
   596→          this.focusedDate = t;
   597→          this.render();
   598→          this.emit("today_requested", { today: t, previousViewDate: this.opts.viewDate });
   599→        }
   600→        else if (act === "clear") {
   601→          this.clearSelection();
   602→          this.emit("clear_requested", {});
   603→        }
   604→      });
   605→    });
   606→
   607→    this.root.querySelectorAll("[data-date]").forEach((el) => {
   608→      const dk = (el as HTMLElement).dataset.date!;
   609→      const [y, m, d] = dk.split("-").map(Number);
   610→      const date: CalendarDate = { year: y, month: m, day: d };
   611→      const isOutside = (el as HTMLElement).dataset.o === "1";
   612→
   613→      el.addEventListener("click", (e) => {
   614→        e.preventDefault();
   615→        if (this.opts.readOnly || this.opts.disabled) return;
   616→        if (isOutside) {
   617→          this.navigateTo({ year: date.year, month: date.month, day: 1 });
   618→        }
   619→        this.selectDate(date, "pointer");
   620→      });
   621→
   622→      if (this.opts.hoverPreview && this.opts.selectionMode === "range" && this.opts.rangeStart && !this.opts.rangeEnd) {
   623→        el.addEventListener("mouseenter", () => {
   624→          this.previewEnd = date;
   625→          this.render();
   626→        });
   627→      }
   628→    });
   629→  }

   631→  /** Replace each .cal-cell with a CustomButton instance (variant="cell"). */
   632→  private applyCustomButtonCells(): void {
   633→    this.cellButtons = [];
   634→    const cells = this.root.querySelectorAll(".cal-cell");
   635→    cells.forEach((cellEl) => {
   636→      const el = cellEl as HTMLElement;
   637→      const text = el.textContent || "";
   638→      const classes = el.className.split(" ").filter(c => c !== "cal-cell");
   639→      const isDisabled = el.hasAttribute("disabled");
   640→      const isSelected = classes.includes("cal-selected") || classes.includes("cal-range-start") || classes.includes("cal-range-end");
   641→      const sz = this.opts.cellSize;
   642→
   643→      // Create a CustomButton in cell mode
   644→      const wrapper = document.createElement("div");
   645→      wrapper.style.display = "inline-flex";
   646→      el.parentNode?.insertBefore(wrapper, el);
   647→
   648→      const btn = new CustomButton(wrapper, {
   649→        text,
   650→        variant: "cell",
   651→        size: "sm",
   652→        width: sz,
   653→        height: sz,
   654→        disabled: isDisabled,
   655→        active: isSelected,
   656→      });
   657→
   658→      // Copy data attributes for event wiring
   659→      const dk = el.getAttribute("data-date");
   660→      const o = el.getAttribute("data-o");
   661→      if (dk) btn.getElement().setAttribute("data-date", dk);
   662→      if (o) btn.getElement().setAttribute("data-o", o);
   663→      btn.getElement().setAttribute("aria-label", el.getAttribute("aria-label") || "");
   664→
   665→      // Copy state classes
   666→      for (const cls of classes) {
   667→        if (cls.startsWith("cal-")) {
   668→          btn.getElement().classList.add(cls);
   669→        }
   670→      }
   671→
   672→      // Track the button + its date key for per-instance prop updates
   673→      if (dk) this.cellButtons.push({ button: btn, dateKey: dk });
   674→
   675→      // Remove the original cell
   676→      el.remove();
   677→    });
   678→  }
   679→
   680→  /**
   681→   * Update a specific day cell's CustomButton properties at runtime.
   682→   * dateKey format: "YYYY-MM-DD". Pass null to apply to ALL cells.
   683→   */
   684→  setCellProp(prop: string, value: unknown, dateKey: string | null = null): void {
   685→    for (const entry of this.cellButtons) {
   686→      if (dateKey !== null && entry.dateKey !== dateKey) continue;
   687→      const btn = entry.button;
   688→      switch (prop) {
   689→        case "variant": btn.setVariant(value as any); break;
   690→        case "text": btn.setText(value as string); break;
   691→        case "icon": btn.setIcon(value as string); break;
   692→        case "iconType": btn.setIcon(btn["options"].icon || "", value as any); break;
   693→        case "iconPosition": btn.setIconPosition(value as any); break;
   694→        case "size":
   695→          btn.setSize(value as any);
   696→          btn.setWidth(0);
   697→          btn.setHeight(0);
   698→          break;
   699→        case "fullWidth": btn.setFullWidth(!!value); break;
   700→        case "draggable": btn.setDraggable(!!value); break;
   701→      }
   702→    }
   703→  }
