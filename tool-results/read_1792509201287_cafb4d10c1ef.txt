   434→function PropertyField({ label, hint, value, children }: { label: string; hint?: string; value?: string; children: React.ReactNode }) {

   447→function RangeInput({ min, max, step, value, onChange }: { min: number; max: number; step: number; value: number; onChange: (v: number) => void }) {
   448→  return (
   449→    <div className="flex items-center gap-2">
   450→      <input
   451→        type="range" min={min} max={max} step={step} value={value}

   454→        style={{ background: `linear-gradient(to right, var(--brand-lemon) 0%, var(--brand-lemon) ${((value - min) / (max - min)) * 100}%, var(--muted) ${((value - min) / (max - min)) * 100}%, var(--muted) 100%)` }}
   455→      />
   456→      <input
   457→        type="number" min={min} max={max} value={value}
   458→        onChange={(e) => onChange(Number(e.target.value))}
   459→        className="h-7 w-14 rounded-md border border-border bg-background px-1.5 text-center font-mono text-[11px] text-foreground focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring"
   460→      />
   461→    </div>
   462→  );
   463→}

   537→interface ColorEntry {
   538→  id: number;
   539→  hue: number;        // 0–360 absolute
   540→  saturation: number; // 0–1
   541→  isBase: boolean;
   542→  offset: number;     // hue offset from base (degrees)

   544→
   545→interface ColorCircleOutput {
   546→  baseHue: number;
   547→  baseIndex: number;
   548→  harmony: Harmony;
   549→  colors: (ColorCircleColor & { isBase: boolean; saturation: number; offset: number })[];

   552→class ColorCircle {
   553→  constructor(container: HTMLElement, options?: ColorCircleOptions);
   554→  // Palette
   555→  getColors(): ColorEntry[];
   556→  getBaseIndex(): number;
   557→  setBaseIndex(i: number): void;       // change which color drives the palette
   558→  addColor(hue?: number, saturation?: number): number;  // returns new index
   559→  removeColor(i: number): void;
   560→  // Configuration

   562→  setSize(n: number): void;

   567→  // Output

   571→  off(event: string, handler: Function): void;
   572→  destroy(): void;
   573→}
   574→
   575→// Behavior:
   576→//   - Moving the BASE color shifts all other colors (they maintain offsets).
