   488→function generateApiCode(): string {
   489→  return `type SmartTooltipPosition = "top" | "bottom" | "left" | "right";
   490→
   491→interface SmartTooltipOptions {
   492→  /** The container element that bounds the tooltip. */
   493→  bounds: HTMLElement;
   494→  /** Command ID to display. */
   495→  commandId: string;
   496→  /** Position relative to the target element. Default "top". */
   497→  position?: SmartTooltipPosition;
   498→}
   499→
   500→class SmartTooltip {
   501→  constructor(container: HTMLElement, options: SmartTooltipOptions);
   502→
   503→  /** Attach to a target element. mouseenter shows; mouseleave hides. */
   504→  attachTo(target: HTMLElement): void;
   505→
   506→  /** Detach from the current target (and hide). */
   507→  detach(): void;
   508→
   509→  /** Manually show the tooltip (still requires a target). */
   510→  show(): void;
   511→
   512→  /** Manually hide the tooltip. */
   513→  hide(): void;
   514→
   515→  /** Change the displayed command at runtime. */
   516→  setCommandId(id: string): void;
   517→
   518→  /** Change the preferred position. The tooltip auto-flips if the preferred
   519→      side overflows the bounds. */
   520→  setPosition(pos: SmartTooltipPosition): void;
   521→
   522→  on(event: "show" | "hide", handler: () => void): void;
   523→  off(event: string, handler: Function): void;
   524→  destroy(): void;
   525→}
   526→
   527→// Smart positioning
   528→//   On show, the tooltip:
   529→//     1. Measures its natural width/height.
