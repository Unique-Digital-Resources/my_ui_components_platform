   160→    const next = cur.slice();
   161→    const [moved] = next.splice(idx, 1);

   372→                              active
   373→                                ? "bg-brand-purple-strong text-foreground"
   374→                                : "text-muted-foreground hover:text-foreground hover:bg-accent",
   375→                            )}
   376→                          >
   377→                            {s}
   378→                          </button>
   379→                        );
   380→                      })}
   381→                    </div>
   382→                  </PropertyField>
   383→                </div>
   384→              </section>
   385→
   386→              <div className="h-px bg-border" />
   387→
   388→              {/* Nested component properties — Buttons section (Figma/Godot style) */}

   390→                <div className="mb-3 flex items-center justify-between">
   391→                  <h3 className="text-[12px] font-semibold uppercase tracking-wider text-muted-foreground">
   392→                    Buttons ({items.length})
   393→                  </h3>
   394→                  <button
   395→                    type="button"
   396→                    onClick={addItem}
   397→                    className="inline-flex h-7 items-center gap-1.5 rounded-md border border-brand-lemon/40 bg-brand-lemon/10 px-2.5 text-[12px] font-medium text-brand-lemon transition-colors hover:bg-brand-lemon/20"
   398→                  >
   399→                    <Plus className="h-3 w-3" />
   400→                    Add button
   401→                  </button>
   402→                </div>
   403→
   404→                <div className="flex flex-col gap-2">
   405→                  {items.length === 0 && (
   406→                    <div className="rounded-md border border-dashed border-border p-6 text-center">
   407→                      <p className="text-[12px] text-muted-foreground">
   408→                        No buttons. Click &quot;Add button&quot; to create one.

   427→                      onRemove={() => removeItem(it.id)}
   428→                    />
   429→                  ))}
   430→                </div>
   431→              </section>
   432→            </div>
   433→          </aside>
   434→        </div>
   435→
   436→        {/* Code panel */}
   437→        <section className="panel mt-6 overflow-hidden">
   438→          <div className="flex items-center justify-between border-b border-border px-2">
   439→            <div className="flex">
   440→              {([["javascript","JavaScript"],["html","HTML"],["css","CSS"],["api","API"]] as const).map(([key, label]) => (
   441→                <button
   442→                  key={key}
   443→                  type="button"
   444→                  onClick={() => setActiveTab(key)}

   447→                    activeTab === key ? "text-foreground" : "text-muted-foreground hover:text-foreground",
   448→                  )}
   449→                >
   450→                  {label}
   451→                  {activeTab === key && <span className="absolute inset-x-2 -bottom-px h-px bg-brand-lemon" />}
   452→                </button>
   453→              ))}
   454→            </div>
   455→            <span className="pr-2 text-[11px] font-mono text-muted-foreground/70">
   456→              Synchronized with live configuration
   457→            </span>

   466→
   467→        {/* Documentation */}

   496→function Switch({
   497→  checked,
   498→  onChange,
   499→}: {
   500→  checked: boolean;

   528→function ButtonItemEditor({
   529→  item,
   530→  index,
   531→  total,
   532→  multiselect,
   533→  iconOptions,
   534→  onLabelChange,
   535→  onIconChange,
   536→  onSelectedToggle,
   537→  onMoveUp,
   538→  onMoveDown,
   539→  onRemove,
   540→}: {
   541→  item: ToggleButtonItem;
   542→  index: number;
   543→  total: number;
   544→  multiselect: boolean;
   545→  iconOptions: string[];
   546→  onLabelChange: (label: string) => void;
   547→  onIconChange: (icon: string) => void;
   548→  onSelectedToggle: () => void;
   549→  onMoveUp: () => void;
   550→  onMoveDown: () => void;
   551→  onRemove: () => void;
   552→}) {
   553→  const [expanded, setExpanded] = React.useState(true);
   554→  const isSelected = !!item.selected;
   555→
   556→  return (
   557→    <div className="rounded-md border border-border bg-muted/20">
   558→      {/* Row 1: summary + collapse */}
   559→      <div className="flex items-center gap-2 px-2 py-1.5">
   560→        <button
   561→          type="button"
   562→          onClick={() => setExpanded((e) => !e)}
   563→          className="inline-flex h-6 w-6 items-center justify-center rounded text-muted-foreground transition-colors hover:bg-accent hover:text-foreground"

   771→function generateApiCode(): string {
   772→  return `type ToggleGroupOrientation = "horizontal" | "vertical";
   773→type ToggleGroupVariant = "primary" | "secondary" | "outline";
   774→type ToggleGroupSize = "sm" | "md" | "lg";
   775→
   776→interface ToggleButtonItem {
   777→  id: string;
   778→  label: string;
   779→  icon?: string;          // built-in name | data URI | raw SVG
   780→  selected?: boolean;     // default false
   781→}
   782→
   783→interface ToggleGroupOptions {
   784→  items?: ToggleButtonItem[];
   785→  multiselect?: boolean;          // default false
   786→  closable?: boolean;             // default false
   787→  reorderable?: boolean;          // default false
   788→  orientation?: ToggleGroupOrientation; // default "horizontal"
   789→  variant?: ToggleGroupVariant;   // default "outline"
   790→  size?: ToggleGroupSize;         // default "md"
   791→}
   792→
   793→class ToggleGroup {
   794→  constructor(container: HTMLElement, options?: ToggleGroupOptions);
   795→  getItems(): ToggleButtonItem[];
   796→  setItems(items: ToggleButtonItem[]): void;
   797→  addItem(item: ToggleButtonItem): void;
   798→  removeItem(id: string): void;
   799→  getSelected(): string[];
   800→  setSelected(ids: string[]): void;
   801→  setMultiselect(on: boolean): void;
   802→  setClosable(on: boolean): void;
   803→  setReorderable(on: boolean): void;
   804→  setOrientation(orientation: ToggleGroupOrientation): void;
   805→  setVariant(variant: ToggleGroupVariant): void;
   806→  setSize(size: ToggleGroupSize): void;
   807→  on(event: "change" | "reorder" | "close" | "select",
   808→     handler: (data: any) => void): void;
   809→  off(event: string, handler: Function): void;
   810→  destroy(): void;
   811→}
   812→
   813→// Events
   814→//   "change"  — fires on selection, reorder, add, remove, or setItems
   815→//   "select"  — fires when the user clicks a toggle (single or multi)
   816→//   "reorder" — fires after a drag-to-reorder completes
   817→//   "close"   — fires when a button's X is clicked (item is removed)
   818→
   819→// Interaction
   820→//   - Click a button to toggle its selection
   821→//   - Single-select: clicking selects and deselects others
   822→//   - Multiselect: each button toggles independently
   823→//   - Closable: click the X on a button to remove it
   824→//   - Reorderable: drag a button to swap positions; Arrow keys also work
   825→//   - Keyboard: Enter/Space toggles; Delete/Backspace closes (if closable)`;
   826→}
