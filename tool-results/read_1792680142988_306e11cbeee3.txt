   829→function generateApiCode(): string {
   830→  return `type ToggleGroupOrientation = "horizontal" | "vertical";
   831→type ToggleGroupVariant = "primary" | "secondary" | "outline";
   832→type ToggleGroupSize = "sm" | "md" | "lg";
   833→
   834→interface ToggleButtonItem {
   835→  id: string;
   836→  label: string;
   837→  icon?: string;          // built-in name | data URI | raw SVG
   838→  selected?: boolean;     // default false
   839→}
   840→
   841→interface ToggleGroupOptions {
   842→  items?: ToggleButtonItem[];
   843→  multiselect?: boolean;          // default false
   844→  closable?: boolean;             // default false
   845→  reorderable?: boolean;          // default false
   846→  orientation?: ToggleGroupOrientation; // default "horizontal"
   847→  variant?: ToggleGroupVariant;   // default "outline"
   848→  size?: ToggleGroupSize;         // default "md"
   849→}
   850→
   851→class ToggleGroup {
   852→  constructor(container: HTMLElement, options?: ToggleGroupOptions);
   853→  getItems(): ToggleButtonItem[];
   854→  setItems(items: ToggleButtonItem[]): void;
   855→  addItem(item: ToggleButtonItem): void;
   856→  removeItem(id: string): void;
   857→  getSelected(): string[];
   858→  setSelected(ids: string[]): void;
   859→  setMultiselect(on: boolean): void;
   860→  setClosable(on: boolean): void;
   861→  setReorderable(on: boolean): void;
   862→  setOrientation(orientation: ToggleGroupOrientation): void;
   863→  setVariant(variant: ToggleGroupVariant): void;
   864→  setSize(size: ToggleGroupSize): void;
   865→  on(event: "change" | "reorder" | "close" | "select",
   866→     handler: (data: any) => void): void;
   867→  off(event: string, handler: Function): void;
   868→  destroy(): void;
   869→}
   870→
   871→// Events
   872→//   "change"  — fires on selection, reorder, add, remove, or setItems
   873→//   "select"  — fires when the user clicks a toggle (single or multi)
   874→//   "reorder" — fires after a drag-to-reorder completes
   875→//   "close"   — fires when a button's X is clicked (item is removed)
   876→
   877→// Interaction
   878→//   - Click a button to toggle its selection
   879→//   - Single-select: clicking selects and deselects others
   880→//   - Multiselect: each button toggles independently
   881→//   - Closable: click the X on a button to remove it
   882→//   - Reorderable: drag a button to swap positions; Arrow keys also work
   883→//   - Keyboard: Enter/Space toggles; Delete/Backspace closes (if closable)`;
   884→}
