    47→
    48→function parseDateInput(s: string): Date | undefined {
    49→  if (!s) return undefined;
    50→  // HTML date inputs produce "YYYY-MM-DD" — parse as local midnight.
    51→  const [y, m, d] = s.split("-").map(Number);
    52→  if (!y || !m || !d) return undefined;
    53→  return new Date(y, m - 1, d);
    54→}
    55→
    56→function toDateInput(d: Date | null): string {
    57→  if (!d) return "";
    58→  return formatDateISO(d);
    59→}

   514→  checked,

   528→        checked
   529→          ? "border-brand-lemon bg-brand-lemon"
   530→          : "border-border bg-muted",
   531→      )}
   532→    >
   533→      <span

   563→  lines.push(`});`);

   569→    lines.push(`  console.log("selected:", payload.selectedDate);`);
   570→  } else if (state.mode === "range") {

   579→  lines.push(`  // Fires when the visible month changes`);
   580→  lines.push(`  console.log("view:", payload.viewDate);`);
   581→  lines.push(`});`);
   582→  lines.push(``);
   583→  lines.push(`cal.on("change", (payload) => {`);
   584→  lines.push(`  // Fires on any selection or view change (user or programmatic)`);
   585→  lines.push(`});`);
   586→  lines.push(``);
   587→  lines.push(`// Programmatic control`);
   588→  if (state.mode === "single") {

   590→  } else if (state.mode === "range") {
   591→    lines.push(`// cal.setSelectedRange(new Date(2024, 5, 1), new Date(2024, 5, 30));`);

   599→  lines.push(`// cal.today(); // jump to current month, today selected`);
   600→  lines.push(`// cal.destroy();`);
   601→  return lines.join("\n");

   646→function generateApiCode(): string {
   647→  return `type CalendarMode = "single" | "range" | "multi";
   648→type WeekdayFormat = "narrow" | "short" | "long";
   649→
   650→interface CalendarRange {
   651→  start: Date | null;
   652→  end: Date | null;
   653→}
   654→
   655→interface CalendarOptions {
   656→  // Date selection
   657→  selectedDate?: Date | null;
   658→  minDate?: Date;
   659→  maxDate?: Date;
   660→  // Multi-selection
   661→  mode?: CalendarMode;                 // default "single"
   662→  selectedRange?: CalendarRange;
   663→  selectedDates?: Date[];              // for multi mode
   664→  // Display
   665→  firstDayOfWeek?: number;             // 0=Sunday, 1=Monday. Default 0
   666→  weekdayFormat?: WeekdayFormat;       // default "short"
   667→  showWeekNumbers?: boolean;           // default false
   668→  showTodayButton?: boolean;           // default true
   669→  showAdjacentMonths?: boolean;        // default true
   670→  // Behavior
   671→  disabledDates?: Date[];
   672→  disabledDaysOfWeek?: number[];       // 0–6
   673→  // Visual
   674→  todayHighlight?: boolean;            // default true
   675→  weekendHighlight?: boolean;          // default false
   676→  // Size
   677→  cellSize?: number;                   // default 36
   678→}
   679→
   680→interface CalendarEventPayload {
   681→  date: Date | null;
   682→  selectedDate: Date | null;
   683→  selectedRange: CalendarRange;
   684→  selectedDates: Date[];
   685→  viewDate: Date;
   686→  mode: CalendarMode;
   687→  source: "select" | "navigate" | "programmatic";
   688→}
   689→
   690→class Calendar {
   691→  constructor(container: HTMLElement, options?: CalendarOptions);
   692→  getSelectedDate(): Date | null;
   693→  getSelectedRange(): CalendarRange;
   694→  getSelectedDates(): Date[];
   695→  setSelectedDate(date: Date | null): void;
   696→  setSelectedRange(start: Date | null, end: Date | null): void;
   697→  setSelectedDates(dates: Date[]): void;
   698→  setMode(mode: CalendarMode): void;
   699→  setMinDate(date: Date): void;
   700→  setMaxDate(date: Date): void;
   701→  setFirstDayOfWeek(day: number): void;
   702→  setShowWeekNumbers(on: boolean): void;
   703→  setCellSize(n: number): void;
   704→  today(): void;
   705→  on(event: "select" | "change" | "navigate",
   706→     handler: (payload: CalendarEventPayload) => void): void;
   707→  off(event: string, handler: Function): void;
   708→  destroy(): void;
   709→}
   710→
   711→// Events
   712→//   "select"   — fires when the user clicks an enabled day
   713→//   "change"   — fires on any selection or view change (user or programmatic)
   714→//   "navigate" — fires when the visible month changes
   715→
   716→// Selection modes
   717→//   "single" — clicking a date selects it (deselects previous)
   718→//   "range"  — first click sets start, second click sets end.
   719→//              If end < start, swap them. Hovering after start shows preview.
   720→//   "multi"  — clicking toggles a date in the selection set`;
   721→}
