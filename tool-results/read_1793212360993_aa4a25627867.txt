     1→/**
     2→ * AnimationTrackEditor — a timeline-based animation track editor (vanilla TS).
     3→ *
     4→ * Inspired by Godot's animation timeline + DCC tools. Supports two track types:
     5→ *   - **Keyframe tracks**: discrete keyframes with values and transition paths.
     6→ *   - **Clip tracks**: resizable boxes (for audio, image, video clips).
     7→ *
     8→ * Features:
     9→ *   - Ruler with tick marks + time labels (seconds)
    10→ *   - Playhead with scrubbing (click/drag ruler)
    11→ *   - Playback: play, pause, reverse, loop, speed control
    12→ *   - Zoom (pixels per second) + horizontal pan
    13→ *   - Keyframe drag (with snapping), selection, add/remove
    14→ *   - Clip drag + resize (both edges), selection, add/remove
    15→ *   - Track management: add/remove, visibility, lock, reorder
    16→ *   - Grouped tracks (by owner)
    17→ *   - Keyboard: Space (play/pause), Left/Right (step), Home/End
    18→ *   - Events for all changes
    19→ */
    20→
    21→// ---- Public types ----
    22→
    23→export type TrackType = "keyframe" | "clip";
    24→
    25→export interface Keyframe {
    26→  id: string;
    27→  time: number; // seconds
    28→  value: number;
    29→  /** Transition curve to the NEXT keyframe. */
    30→  transition?: "linear" | "ease" | "ease-in" | "ease-out" | "step" | "hold";
    31→  selected?: boolean;
    32→}
    33→
    34→export interface Clip {
    35→  id: string;
    36→  start: number; // seconds
    37→  end: number;   // seconds
    38→  label?: string;
    39→  color?: string;
    40→  selected?: boolean;
    41→}
    42→
    43→export interface Track {
    44→  id: string;
    45→  name: string;
    46→  type: TrackType;
    47→  owner?: string;
    48→  keyframes?: Keyframe[];
    49→  clips?: Clip[];
    50→  visible?: boolean;
    51→  locked?: boolean;
    52→  color?: string;
    53→}
    54→
    55→export interface AnimationTrackEditorOptions {
    56→  tracks?: Track[];
    57→  duration?: number;     // total timeline duration in seconds
    58→  pixelsPerSecond?: number; // zoom (px/s)
    59→  playhead?: number;     // current playhead position (seconds)
    60→  playing?: boolean;
    61→  loop?: boolean;
    62→  speed?: number;        // playback speed multiplier
    63→  rowHeight?: number;
    64→  snapInterval?: number; // seconds; 0 = no snap
    65→  showRuler?: boolean;
    66→  showToolbar?: boolean;
    67→  readOnly?: boolean;
    68→}
    69→

    71→  type?: string;
    72→  trackId?: string;
    73→  keyframeId?: string;
    74→  clipId?: string;
    75→  time?: number;
    76→  oldTime?: number;
    77→  value?: number;
    78→  oldValue?: number;
    79→  playhead?: number;
    80→  playing?: boolean;
    81→  pixelsPerSecond?: number;
    82→  source?: "pointer" | "keyboard" | "programmatic" | "playback";
    83→  reason?: string;
    84→}

   446→// ---- Utilities ----
   447→
   448→function uid(prefix: string): string {
   449→  return `${prefix}-${Math.random().toString(36).slice(2, 9)}`;
   450→}
   451→
   452→function clamp(n: number, min: number, max: number): number {
   453→  return Math.max(min, Math.min(max, n));
   454→}
   455→
   456→function snapTime(t: number, snap: number): number {
   457→  if (!snap || snap <= 0) return t;
   458→  return Math.round(t / snap) * snap;
   459→}

   494→      tracks: options.tracks ?? [],

   503→      showRuler: options.showRuler ?? true,
   504→      showToolbar: options.showToolbar ?? true,
   505→      readOnly: options.readOnly ?? false,
   506→    };
   507→    this.tracks = (this.opts.tracks as Track[]).map(t => ({ ...t }));
   508→    this.render();
   509→
   510→    // Sync scrolls
   511→    this.rightScroll.addEventListener("scroll", () => {
   512→      this.leftScroll.scrollTop = this.rightScroll.scrollTop;

   515→    // Resize observer
   516→    this.resizeObserver = new ResizeObserver(() => this.drawRuler());
   517→    this.resizeObserver.observe(this.root);
   518→
   519→    // Start playback if needed
   520→    if (this.opts.playing) this.startPlayback();
   521→  }

   865→  private drawRuler(): void {
   866→    if (!this.rulerCanvas) return;
   867→    const canvas = this.rulerCanvas;
   868→    const rect = canvas.getBoundingClientRect();
   869→    const dpr = window.devicePixelRatio || 1;
   870→    canvas.width = rect.width * dpr;
   871→    canvas.height = rect.height * dpr;
   872→    const ctx = canvas.getContext("2d")!;
   873→    ctx.scale(dpr, dpr);
   874→
   875→    const w = rect.width;
   876→    const h = rect.height;
   877→    const pps = this.opts.pixelsPerSecond;
   878→    const duration = this.opts.duration;
   879→
   880→    // Background
   881→    ctx.fillStyle = "#28282E";
   882→    ctx.fillRect(0, 0, w, h);
   883→
   884→    // Determine tick interval (aim for ~60px between ticks)
   885→    const targetSpacing = 60;
   886→    const rawInterval = targetSpacing / pps;
   887→    // Round to nice numbers: 0.1, 0.25, 0.5, 1, 2, 5, 10, 30, 60
   888→    const niceIntervals = [0.1, 0.25, 0.5, 1, 2, 5, 10, 30, 60];
   889→    let interval = niceIntervals[0];
   890→    for (const ni of niceIntervals) {
   891→      if (ni >= rawInterval) { interval = ni; break; }
   892→      interval = ni;
   893→    }
   894→
   895→    // Draw ticks + labels
   896→    ctx.strokeStyle = "#555560";
   897→    ctx.fillStyle = "#888894";
   898→    ctx.font = "9px ui-monospace, monospace";
   899→    ctx.textAlign = "center";
   900→    ctx.textBaseline = "middle";
   901→
   902→    for (let t = 0; t <= duration; t += interval) {
   903→      const x = t * pps;
   904→      if (x > w + 10) break;
   905→      // Major tick
   906→      ctx.beginPath();
   907→      ctx.moveTo(x, h - 8);
   908→      ctx.lineTo(x, h);
   909→      ctx.lineWidth = 1;
   910→      ctx.stroke();
   911→      // Label
   912→      ctx.fillText(this.formatTime(t), x, h / 2);
   913→      // Minor ticks
   914→      const minorStep = interval / 5;
   915→      for (let i = 1; i < 5; i++) {
   916→        const mx = (t + minorStep * i) * pps;
   917→        if (mx > w) break;
   918→        ctx.beginPath();
   919→        ctx.moveTo(mx, h - 4);
   920→        ctx.lineTo(mx, h);
   921→        ctx.lineWidth = 0.5;
   922→        ctx.stroke();
   923→      }
   924→    }
   925→  }

  1021→  private startPlayback(): void {
  1022→    if (this.playbackRAF !== null) return;
  1023→    const tick = (now: number) => {
  1024→      if (!this.opts.playing || this._destroyed) { this.playbackRAF = null; return; }
  1025→      const dt = (now - this.lastFrameTime) / 1000;
  1026→      this.lastFrameTime = now;
  1027→      let newTime = this.opts.playhead + dt * this.opts.speed;
  1028→      if (newTime >= this.opts.duration) {
  1029→        if (this.opts.loop) newTime = 0;
  1030→        else { newTime = this.opts.duration; this.pause(); }
  1031→      }
  1032→      this.opts.playhead = newTime;
  1033→      this.updatePlayhead();
  1034→      this.emit("playback_tick", { playhead: newTime, source: "playback" });
  1035→      this.playbackRAF = requestAnimationFrame(tick);
  1036→    };
  1037→    this.playbackRAF = requestAnimationFrame(tick);
  1038→  }

  1041→    if (this.playbackRAF !== null) {
  1042→      cancelAnimationFrame(this.playbackRAF);
  1043→      this.playbackRAF = null;
  1044→    }
  1045→  }

  1048→    this.opts.loop = !this.opts.loop;

  1051→  }

  1055→  nextKey(): void {
  1056→    let next = this.opts.duration;
  1057→    for (const track of this.tracks) {
  1058→      if (track.type === "keyframe" && track.keyframes) {
  1059→        for (const kf of track.keyframes) {
  1060→          if (kf.time > this.opts.playhead + 0.001 && kf.time < next) next = kf.time;
  1061→        }
  1062→      }
  1063→    }
  1064→    if (next < this.opts.duration || next === this.opts.duration) {
  1065→      // If there's no key after playhead, jump to end
  1066→      this.setPlayhead(next, "keyboard");
  1067→    }
  1068→  }

  1070→  prevKey(): void {
  1071→    let prev = -1;
  1072→    for (const track of this.tracks) {
  1073→      if (track.type === "keyframe" && track.keyframes) {
  1074→        for (const kf of track.keyframes) {
  1075→          if (kf.time < this.opts.playhead - 0.001 && kf.time > prev) prev = kf.time;
  1076→        }
  1077→      }
  1078→    }
  1079→    if (prev >= 0) this.setPlayhead(prev, "keyboard");
  1080→    else this.setPlayhead(0, "keyboard");
  1081→  }

  1085→  setZoom(pps: number, source: "pointer" | "keyboard" | "programmatic" = "programmatic"): void {
  1086→    if (this._destroyed) return;

  1100→      if (track.type === "clip" && track.clips) {
  1101→        for (const clip of track.clips) {
  1102→          const clipEl = rowEl.querySelector(`[data-clip-id="${clip.id}"]`) as HTMLElement;

  1105→            clipEl.style.width = `${(clip.end - clip.start) * this.opts.pixelsPerSecond}px`;
  1106→          }
  1107→        }
  1108→      }
  1109→    }
  1110→    this.updatePlayhead();
  1111→    this.emit("zoom_change", { pixelsPerSecond: this.opts.pixelsPerSecond, source });

  1122→    const rect = this.rulerCanvas.getBoundingClientRect();
  1123→    const setFromX = (clientX: number) => {
  1124→      const x = clientX - rect.left;

  1126→      this.setPlayhead(t, "pointer");

  1128→    setFromX(e.clientX);
  1129→    const onMove = (ev: PointerEvent) => setFromX(ev.clientX);
  1130→    const onUp = () => {
  1131→      window.removeEventListener("pointermove", onMove);
  1132→      window.removeEventListener("pointerup", onUp);
  1133→      this.dragMode = "none";
  1134→    };
  1135→    window.addEventListener("pointermove", onMove);
  1136→    window.addEventListener("pointerup", onUp);

  1141→  private onKeyframePointerDown(e: PointerEvent, trackId: string, kfId: string): void {
  1142→    if (this.opts.readOnly) return;
  1143→    const track = this.tracks.find(t => t.id === trackId);
  1144→    if (!track || track.locked) return;
  1145→    e.preventDefault();
  1146→    e.stopPropagation();
  1147→    this.dragMode = "keyframe";
  1148→    this.dragTrackId = trackId;
  1149→    this.dragItemId = kfId;
  1150→    this.dragStartX = e.clientX;
  1151→    const kf = track.keyframes?.find(k => k.id === kfId);
  1152→    this.dragOrigTime = kf?.time ?? 0;
  1153→    (e.target as HTMLElement).classList.add("ate-dragging");
  1154→
  1155→    const onMove = (ev: PointerEvent) => {
  1156→      const dx = (ev.clientX - this.dragStartX) / this.opts.pixelsPerSecond;
  1157→      let newTime = snapTime(this.dragOrigTime + dx, this.opts.snapInterval);
  1158→      newTime = clamp(newTime, 0, this.opts.duration);
  1159→      if (kf) { kf.time = newTime; (e.target as HTMLElement).style.left = `${newTime * this.opts.pixelsPerSecond}px`; }
  1160→      this.emit("keyframe_drag", { trackId, keyframeId: kfId, time: newTime, source: "pointer" });
  1161→    };
  1162→    const onUp = () => {
  1163→      window.removeEventListener("pointermove", onMove);
  1164→      window.removeEventListener("pointerup", onUp);
  1165→      (e.target as HTMLElement).classList.remove("ate-dragging");
  1166→      if (kf) this.emit("keyframe_change", { trackId, keyframeId: kfId, time: kf.time, oldTime: this.dragOrigTime, source: "pointer" });
  1167→      this.dragMode = "none";
  1168→      this.dragTrackId = null;
  1169→      this.dragItemId = null;
  1170→    };
  1171→    window.addEventListener("pointermove", onMove);
  1172→    window.addEventListener("pointerup", onUp);
  1173→  }

  1177→  private onClipPointerDown(e: PointerEvent, trackId: string, clipId: string, mode: "move" | "resize-left" | "resize-right"): void {
  1178→    if (this.opts.readOnly) return;
  1179→    const track = this.tracks.find(t => t.id === trackId);
  1180→    if (!track || track.locked) return;
  1181→    e.preventDefault();
  1182→    e.stopPropagation();
  1183→    this.dragMode = mode === "move" ? "clip-move" : mode === "resize-left" ? "clip-resize-left" : "clip-resize-right";
  1184→    this.dragTrackId = trackId;
  1185→    this.dragItemId = clipId;
  1186→    this.dragStartX = e.clientX;
  1187→    const clip = track.clips?.find(c => c.id === clipId);
  1188→    this.dragOrigStart = clip?.start ?? 0;
  1189→    this.dragOrigEnd = clip?.end ?? 0;
  1190→    (e.target as HTMLElement).closest(".ate-clip")?.classList.add("ate-dragging");
  1191→
  1192→    const onMove = (ev: PointerEvent) => {
  1193→      const dx = (ev.clientX - this.dragStartX) / this.opts.pixelsPerSecond;
  1194→      if (!clip) return;
  1195→      const clipEl = this.trackRowEls.get(trackId)?.querySelector(`[data-clip-id="${clipId}"]`) as HTMLElement;
  1196→      if (this.dragMode === "clip-move") {
  1197→        let newStart = snapTime(this.dragOrigStart + dx, this.opts.snapInterval);
  1198→        const dur = this.dragOrigEnd - this.dragOrigStart;
  1199→        newStart = clamp(newStart, 0, this.opts.duration - dur);
  1200→        clip.start = newStart;
  1201→        clip.end = newStart + dur;
  1202→      } else if (this.dragMode === "clip-resize-left") {
  1203→        let newStart = snapTime(this.dragOrigStart + dx, this.opts.snapInterval);
  1204→        newStart = clamp(newStart, 0, this.dragOrigEnd - 0.05);
  1205→        clip.start = newStart;
  1206→      } else if (this.dragMode === "clip-resize-right") {
  1207→        let newEnd = snapTime(this.dragOrigEnd + dx, this.opts.snapInterval);
  1208→        newEnd = clamp(newEnd, this.dragOrigStart + 0.05, this.opts.duration);
  1209→        clip.end = newEnd;
  1210→      }
  1211→      if (clipEl) {
  1212→        clipEl.style.left = `${clip.start * this.opts.pixelsPerSecond}px`;
  1213→        clipEl.style.width = `${(clip.end - clip.start) * this.opts.pixelsPerSecond}px`;
  1214→      }
  1215→      this.emit("clip_drag", { trackId, clipId, time: clip.start, source: "pointer" });
  1216→    };
  1217→    const onUp = () => {
  1218→      window.removeEventListener("pointermove", onMove);
  1219→      window.removeEventListener("pointerup", onUp);
  1220→      (e.target as HTMLElement).closest(".ate-clip")?.classList.remove("ate-dragging");
  1221→      if (clip) this.emit("clip_change", { trackId, clipId, time: clip.start, oldTime: this.dragOrigStart, source: "pointer" });
  1222→      this.dragMode = "none";
  1223→      this.dragTrackId = null;
  1224→      this.dragItemId = null;
  1225→    };
  1226→    window.addEventListener("pointermove", onMove);
  1227→    window.addEventListener("pointerup", onUp);
  1228→  }

  1380→  // ---- Keyboard ----
  1381→
  1382→  private onKeyDown(e: KeyboardEvent): void {
  1383→    if (this.opts.readOnly) return;
  1384→    let handled = true;
  1385→    switch (e.key) {
  1386→      case " ":
  1387→        e.preventDefault();
  1388→        this.togglePlay();
  1389→        break;
  1390→      case "ArrowLeft":
  1391→        this.setPlayhead(this.opts.playhead - (e.shiftKey ? 1 : 0.1), "keyboard");
  1392→        break;
  1393→      case "ArrowRight":
  1394→        this.setPlayhead(this.opts.playhead + (e.shiftKey ? 1 : 0.1), "keyboard");
  1395→        break;
  1396→      case "Home":
  1397→        this.setPlayhead(0, "keyboard");
  1398→        break;
  1399→      case "End":
  1400→        this.setPlayhead(this.opts.duration, "keyboard");
  1401→        break;
  1402→      case "Delete":
  1403→      case "Backspace":
  1404→        // Remove selected keyframes + clips
  1405→        for (const track of this.tracks) {
  1406→          if (track.keyframes) {
  1407→            const toRemove = track.keyframes.filter(k => k.selected).map(k => k.id);
  1408→            for (const id of toRemove) this.removeKeyframe(track.id, id);
  1409→          }
  1410→          if (track.clips) {
  1411→            const toRemove = track.clips.filter(c => c.selected).map(c => c.id);
  1412→            for (const id of toRemove) this.removeClip(track.id, id);
  1413→          }
  1414→        }
  1415→        break;
  1416→      case "Escape":
  1417→        this.clearSelection();
  1418→        break;
  1419→      default:
  1420→        handled = false;
  1421→    }
  1422→    if (handled) e.preventDefault();
  1423→  }
