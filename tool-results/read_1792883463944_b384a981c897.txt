    69→    69→// ---- Date helpers (pure, no Date object to avoid TZ issues) ----
    70→    70→
    71→    71→function isLeap(y: number): boolean {
    72→    72→  return (y % 4 === 0 && y % 100 !== 0) || y % 400 === 0;
    73→    73→}
    74→    74→
    75→    75→function daysInMonth(y: number, m: number): number {
    76→    76→  return [31, isLeap(y) ? 29 : 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31][m - 1];
    77→    77→}
    78→    78→
    79→    79→function dayOfWeek(y: number, m: number, d: number): number {
    80→    80→  // 0=Sunday … 6=Saturday (Zeller-ish via Date, but we normalize to local)
    81→    81→  const dt = new Date(y, m - 1, d);
    82→    82→  return dt.getDay();
    83→    83→}
    84→    84→
    85→    85→function dateKey(d: CalendarDate): string {
    86→    86→  return `${d.year}-${String(d.month).padStart(2, "0")}-${String(d.day).padStart(2, "0")}`;
    87→    87→}
    88→    88→
    89→    89→function sameDate(a: CalendarDate | null, b: CalendarDate | null): boolean {
    90→    90→  if (!a || !b) return false;
    91→    91→  return a.year === b.year && a.month === b.month && a.day === b.day;
    92→    92→}
    93→    93→
    94→    94→function compareDate(a: CalendarDate, b: CalendarDate): number {
    95→    95→  if (a.year !== b.year) return a.year - b.year;
    96→    96→  if (a.month !== b.month) return a.month - b.month;
    97→    97→  return a.day - b.day;
    98→    98→}
    99→    99→
   100→   100→function isBetween(d: CalendarDate, start: CalendarDate, end: CalendarDate): boolean {
   101→   101→  return compareDate(d, start) >= 0 && compareDate(d, end) <= 0;
   102→   102→}
   103→   103→
   104→   104→function addMonths(d: CalendarDate, n: number): CalendarDate {
   105→   105→  const totalMonths = (d.year * 12 + (d.month - 1)) + n;
   106→   106→  const y = Math.floor(totalMonths / 12);
   107→   107→  const m = (totalMonths % 12) + 1;
   108→   108→  const maxDay = daysInMonth(y, m);
   109→   109→  return { year: y, month: m, day: Math.min(d.day, maxDay) };
   110→   110→}
   111→   111→
   112→   112→function addDays(d: CalendarDate, n: number): CalendarDate {
   113→   113→  let { year, month, day } = d;
   114→   114→  day += n;
   115→   115→  while (day > daysInMonth(year, month)) {
   116→   116→    day -= daysInMonth(year, month);
   117→   117→    month++;
   118→   118→    if (month > 12) { month = 1; year++; }
   119→   119→  }
   120→   120→  while (day < 1) {
   121→   121→    month--;
   122→   122→    if (month < 1) { month = 12; year--; }
   123→   123→    day += daysInMonth(year, month);
   124→   124→  }
   125→   125→  return { year, month, day };
   126→   126→}
   127→   127→
   128→   128→function today(): CalendarDate {
   129→   129→  const d = new Date();
   130→   130→  return { year: d.getFullYear(), month: d.getMonth() + 1, day: d.getDate() };
   131→   131→}
   132→   132→
   133→   133→// ISO week number
   134→   134→function weekNumber(d: CalendarDate): number {
   135→   135→  const dt = new Date(d.year, d.month - 1, d.day);
   136→   136→  dt.setHours(0, 0, 0, 0);
   137→   137→  dt.setDate(dt.getDate() + 3 - ((dt.getDay() + 6) % 7));
   138→   138→  const week1 = new Date(dt.getFullYear(), 0, 4);
   139→   139→  return 1 + Math.round(((dt.getTime() - week1.getTime()) / 86400000 - 3 + ((week1.getDay() + 6) % 7)) / 7);
   140→   140→}

   155→   155→interface GridCell {
   156→   156→  date: CalendarDate;
   157→   157→  isOutside: boolean;
   158→   158→  isToday: boolean;
   159→   159→  isSelected: boolean;
   160→   160→  isRangeStart: boolean;
   161→   161→  isRangeEnd: boolean;
   162→   162→  isInRange: boolean;
   163→   163→  isDisabled: boolean;
   164→   164→  isFocused: boolean;
   165→   165→  isPreviewed: boolean;
   166→   166→  weekNumber?: number;

   168→   168→
   169→   169→function buildGrid(
   170→   170→  viewDate: CalendarDate,
   171→   171→  weekStart: WeekStart,
   172→   172→  showOutside: boolean,
   173→   173→  selectionMode: SelectionMode,
   174→   174→  value: CalendarDate | null,
   175→   175→  selectedDates: CalendarDate[],
   176→   176→  rangeStart: CalendarDate | null,
   177→   177→  rangeEnd: CalendarDate | null,
   178→   178→  minDate: CalendarDate | null,
   179→   179→  maxDate: CalendarDate | null,
   180→   180→  disabledDates: Set<string>,
   181→   181→  focusedDate: CalendarDate | null,
   182→   182→  previewEnd: CalendarDate | null,
   183→   183→  showWeekNumbers: boolean,
   184→   184→): GridCell[][] {
   185→   185→  const rows: GridCell[][] = [];
   186→   186→  const firstOfMonth = { year: viewDate.year, month: viewDate.month, day: 1 };
   187→   187→  const firstDayOfWeek = dayOfWeek(firstOfMonth.year, firstOfMonth.month, firstOfMonth.day);
   188→   188→  const offset = (firstDayOfWeek - weekStart + 7) % 7;
   189→   189→  const gridStart = addDays(firstOfMonth, -offset);
   190→   190→  const todayDate = today();
   191→   191→
   192→   192→  for (let week = 0; week < 6; week++) {
   193→   193→    const row: GridCell[] = [];
   194→   194→    for (let col = 0; col < 7; col++) {
   195→   195→      const date = addDays(gridStart, week * 7 + col);
   196→   196→      const isOutside = date.month !== viewDate.month || date.year !== viewDate.year;
   197→   197→      if (isOutside && !showOutside) {
   198→   198→        row.push({
   199→   199→          date, isOutside: true, isToday: false, isSelected: false,
   200→   200→          isRangeStart: false, isRangeEnd: false, isInRange: false,
   201→   201→          isDisabled: true, isFocused: false, isPreviewed: false,
   202→   202→        });
   203→   203→        continue;
   204→   204→      }
   205→   205→      const dk = dateKey(date);
   206→   206→      const isSelected =
   207→   207→        (selectionMode === "single" && sameDate(date, value)) ||
   208→   208→        (selectionMode === "multiple" && selectedDates.some((s) => sameDate(s, date)));
   209→   209→      const isRangeStart = sameDate(date, rangeStart);
   210→   210→      const isRangeEnd = sameDate(date, rangeEnd);
   211→   211→      const isInRange =
   212→   212→        selectionMode === "range" && rangeStart && rangeEnd &&
   213→   213→        isBetween(date, rangeStart, rangeEnd) && !isRangeStart && !isRangeEnd;
   214→   214→      const isPreviewed =
   215→   215→        selectionMode === "range" && rangeStart && !rangeEnd && previewEnd &&
   216→   216→        ((compareDate(previewEnd, rangeStart) >= 0 && isBetween(date, rangeStart, previewEnd) && !isRangeStart) ||
   217→   217→         (compareDate(previewEnd, rangeStart) < 0 && isBetween(date, previewEnd, rangeStart) && !sameDate(date, rangeStart)));
   218→   218→      const isDisabled =
   219→   219→        (minDate && compareDate(date, minDate) < 0) ||
   220→   220→        (maxDate && compareDate(date, maxDate) > 0) ||
   221→   221→        disabledDates.has(dk);
   222→   222→      row.push({
   223→   223→        date, isOutside, isToday: sameDate(date, todayDate),
   224→   224→        isSelected, isRangeStart, isRangeEnd, isInRange,
   225→   225→        isDisabled, isFocused: sameDate(date, focusedDate),
   226→   226→        isPreviewed, weekNumber: showWeekNumbers && col === 0 ? weekNumber(date) : undefined,
   227→   227→      });
   228→   228→    }
   229→   229→    rows.push(row);
   230→   230→  }
   231→   231→  return rows;
   232→   232→}

   328→   328→  navigate(direction: "prev" | "next"): void {
   329→   329→    this.opts.viewDate = addMonths(this.opts.viewDate, direction === "next" ? 1 : -1);
   330→   330→    this.render();

   335→   335→    this.opts.viewDate = { ...date };
   336→   336→    this.render();
   337→   337→    this.emit("view_changed", { viewDate: this.opts.viewDate, reason: "navigate_to" });

   340→   340→  selectDate(date: CalendarDate, source: "pointer" | "keyboard" | "programmatic" = "programmatic"): void {
   341→   341→    if (this.opts.readOnly || this.opts.disabled) return;
   342→   342→    if (this.isDisabled(date)) {
   343→   343→      this.emit("validation_failed", { date, reason: "disabled" });
   344→   344→      return;
   345→   345→    }
   346→   346→    const prev = this.getSnapshot();
   347→   347→    switch (this.opts.selectionMode) {
   348→   348→      case "none": break;
   349→   349→      case "single":
   350→   350→        this.opts.value = sameDate(date, this.opts.value) && this.opts.allowClear ? null : date;
   351→   351→        this.focusedDate = date;
   352→   352→        break;
   353→   353→      case "multiple": {
   354→   354→        const idx = this.opts.selectedDates.findIndex((d) => sameDate(d, date));
   355→   355→        if (idx >= 0) this.opts.selectedDates.splice(idx, 1);
   356→   356→        else this.opts.selectedDates.push(date);
   357→   357→        this.focusedDate = date;
   358→   358→        break;
   359→   359→      }
   360→   360→      case "range":
   361→   361→        if (!this.opts.rangeStart || (this.opts.rangeStart && this.opts.rangeEnd)) {
   362→   362→          this.opts.rangeStart = date;
   363→   363→          this.opts.rangeEnd = null;
   364→   364→          this.previewEnd = null;
   365→   365→          this.emit("range_changed", { startDate: date, endDate: null, complete: false });
   366→   366→        } else {
   367→   367→          if (compareDate(date, this.opts.rangeStart) < 0) {
   368→   368→            this.opts.rangeEnd = this.opts.rangeStart;
   369→   369→            this.opts.rangeStart = date;
   370→   370→          } else {
   371→   371→            this.opts.rangeEnd = date;
   372→   372→          }
   373→   373→          this.previewEnd = null;
   374→   374→          this.emit("range_changed", { startDate: this.opts.rangeStart, endDate: this.opts.rangeEnd, complete: true });
   375→   375→        }
   376→   376→        this.focusedDate = date;
   377→   377→        break;
   378→   378→    }
   379→   379→    this.render();
   380→   380→    this.emit("date_selected", { date, previousValue: prev, source });
   381→   381→    this.emit("selection_changed", { selection: this.getSnapshot(), previousSelection: prev, selectionMode: this.opts.selectionMode });
   382→   382→  }
   383→   383→
   384→   384→  setSelection(selection: CalendarDate | CalendarDate[] | { start: CalendarDate; end: CalendarDate } | null): void {
   385→   385→    if (this.opts.readOnly || this.opts.disabled) return;
   386→   386→    if (selection === null) { this.clearSelection(); return; }
   387→   387→    if (this.opts.selectionMode === "single" && !Array.isArray(selection) && !(selection as any).start) {
   388→   388→      this.opts.value = selection as CalendarDate;
   389→   389→    } else if (this.opts.selectionMode === "multiple" && Array.isArray(selection)) {
   390→   390→      this.opts.selectedDates = [...selection];
   391→   391→    } else if (this.opts.selectionMode === "range" && (selection as any).start) {
   392→   392→      this.opts.rangeStart = (selection as any).start;
   393→   393→      this.opts.rangeEnd = (selection as any).end;
   394→   394→    }
   395→   395→    this.render();
   396→   396→    this.emit("selection_changed", { selection: this.getSnapshot(), selectionMode: this.opts.selectionMode });
   397→   397→  }
   398→   398→
   399→   399→  clearSelection(): void {
   400→   400→    if (!this.opts.allowClear) return;
   401→   401→    const prev = this.getSnapshot();
   402→   402→    this.opts.value = null;
   403→   403→    this.opts.selectedDates = [];
   404→   404→    this.opts.rangeStart = null;
   405→   405→    this.opts.rangeEnd = null;
   406→   406→    this.previewEnd = null;
   407→   407→    this.render();
   408→   408→    this.emit("selection_changed", { selection: null, previousSelection: prev, selectionMode: this.opts.selectionMode });
   409→   409→  }
   410→   410→
   411→   411→  focusDate(date: CalendarDate): void {
   412→   412→    this.focusedDate = date;
   413→   413→    if (date.month !== this.opts.viewDate.month || date.year !== this.opts.viewDate.year) {
   414→   414→      this.opts.viewDate = { year: date.year, month: date.month, day: 1 };
   415→   415→    }

   471→   471→  private onKeyDown(e: KeyboardEvent): void {
   472→   472→    if (this.opts.disabled) return;
   473→   473→    if (!this.focusedDate) this.focusedDate = today();
   474→   474→    let handled = true;
   475→   475→    switch (e.key) {
   476→   476→      case "ArrowLeft": this.focusedDate = addDays(this.focusedDate, -1); break;
   477→   477→      case "ArrowRight": this.focusedDate = addDays(this.focusedDate, 1); break;
   478→   478→      case "ArrowUp": this.focusedDate = addDays(this.focusedDate, -7); break;
   479→   479→      case "ArrowDown": this.focusedDate = addDays(this.focusedDate, 7); break;
   480→   480→      case "Home": {
   481→   481→        const dow = dayOfWeek(this.focusedDate.year, this.focusedDate.month, this.focusedDate.day);
   482→   482→        const offset = (dow - this.opts.weekStart + 7) % 7;
   483→   483→        this.focusedDate = addDays(this.focusedDate, -offset); break;
   484→   484→      }
   485→   485→      case "End": {
   486→   486→        const dow = dayOfWeek(this.focusedDate.year, this.focusedDate.month, this.focusedDate.day);
   487→   487→        const offset = (this.opts.weekStart + 6 - dow + 7) % 7;
   488→   488→        this.focusedDate = addDays(this.focusedDate, offset); break;
   489→   489→      }
   490→   490→      case "PageUp": this.focusedDate = addMonths(this.focusedDate, e.shiftKey ? -12 : -1); break;
   491→   491→      case "PageDown": this.focusedDate = addMonths(this.focusedDate, e.shiftKey ? 12 : 1); break;
   492→   492→      case "Enter":
   493→   493→      case " ":
   494→   494→        this.selectDate(this.focusedDate, "keyboard");
   495→   495→        handled = true;
   496→   496→        break;
   497→   497→      case "Escape":
   498→   498→        if (this.opts.selectionMode === "range" && this.opts.rangeStart && !this.opts.rangeEnd) {
   499→   499→          this.opts.rangeStart = null;
   500→   500→          this.previewEnd = null;
   501→   501→          this.render();
   502→   502→        }
   503→   503→        break;
   504→   504→      default: handled = false;
   505→   505→    }
   506→   506→    if (handled) {
   507→   507→      e.preventDefault();
   508→   508→      if (this.focusedDate.month !== this.opts.viewDate.month || this.focusedDate.year !== this.opts.viewDate.year) {
   509→   509→        this.opts.viewDate = { year: this.focusedDate.year, month: this.focusedDate.month, day: 1 };
   510→   510→        this.emit("view_changed", { viewDate: this.opts.viewDate, reason: "keyboard" });
   511→   511→      }
   512→   512→      this.render();
   513→   513→      this.emit("focused_date_changed", { date: this.focusedDate });
   514→   514→    }
   515→   515→  }

   517→   517→  private render(): void {
   518→   518→    if (this._destroyed) return;
   519→   519→    const disabledSet = new Set(this.opts.disabledDates.map(dateKey));
   520→   520→    const grid = buildGrid(
   521→   521→      this.opts.viewDate, this.opts.weekStart, this.opts.showOutsideDates,
   522→   522→      this.opts.selectionMode, this.opts.value, this.opts.selectedDates,
   523→   523→      this.opts.rangeStart, this.opts.rangeEnd, this.opts.minDate, this.opts.maxDate,
   524→   524→      disabledSet, this.focusedDate, this.previewEnd, this.opts.showWeekNumbers,
   525→   525→    );
   526→   526→    const wkNames = WEEKDAY_NAMES[this.opts.weekdayFormat];
   527→   527→    const wkHeaders: string[] = [];
   528→   528→    for (let i = 0; i < 7; i++) wkHeaders.push(wkNames[(this.opts.weekStart + i) % 7]);
   529→   529→    const monthName = MONTH_NAMES[this.opts.viewDate.month - 1];
   530→   530→    const sz = this.opts.cellSize;
   531→   531→
   532→   532→    let html = "";
   533→   533→
   534→   534→    // Header
   535→   535→    if (this.opts.showNavigation) {
   536→   536→      html += `<div class="cal-header">`;
   537→   537→      html += `<button class="cal-nav-btn" data-act="prev" aria-label="Previous month">‹</button>`;
   538→   538→      html += `<span class="cal-title">${monthName} ${this.opts.viewDate.year}</span>`;
   539→   539→      html += `<button class="cal-nav-btn" data-act="next" aria-label="Next month">›</button>`;
   540→   540→      html += `</div>`;
   541→   541→    }
   542→   542→
   543→   543→    // Weekday header
   544→   544→    html += `<div class="cal-weekdays">`;
   545→   545→    if (this.opts.showWeekNumbers) html += `<div class="cal-weekday"></div>`;
   546→   546→    for (const w of wkHeaders) html += `<div class="cal-weekday">${w}</div>`;
   547→   547→    html += `</div>`;
   548→   548→
   549→   549→    // Grid
   550→   550→    html += `<div class="cal-grid">`;
   551→   551→    for (const row of grid) {
   552→   552→      if (this.opts.showWeekNumbers && row[0].weekNumber !== undefined) {
   553→   553→        html += `<div class="cal-weekday" style="display:flex;align-items:center;justify-content:center;">${row[0].weekNumber}</div>`;
   554→   554→      }
   555→   555→      for (const cell of row) {
   556→   556→        const classes = ["cal-cell"];
   557→   557→        if (cell.isOutside) classes.push("cal-outside");
   558→   558→        if (cell.isToday && this.opts.showToday) classes.push("cal-today");
   559→   559→        if (cell.isSelected) classes.push("cal-selected");
   560→   560→        if (cell.isRangeStart) classes.push("cal-range-start");
   561→   561→        if (cell.isRangeEnd) classes.push("cal-range-end");
   562→   562→        if (cell.isInRange) classes.push("cal-in-range");
   563→   563→        if (cell.isPreviewed) classes.push("cal-preview");
   564→   564→        if (cell.isDisabled) classes.push("cal-disabled");
   565→   565→        if (cell.isFocused) classes.push("cal-focused");
   566→   566→        const dk = dateKey(cell.date);
   567→   567→        const label = `${cell.date.month}/${cell.date.day}/${cell.date.year}`;
   568→   568→        html += `<button class="${classes.join(" ")}" data-date="${dk}" data-o="${cell.isOutside?1:0}" aria-label="${label}" style="width:${sz}px;height:${sz}px;"${cell.isDisabled ? " disabled" : ""}>${cell.date.day}</button>`;
   569→   569→      }
   570→   570→    }
   571→   571→    html += `</div>`;
   572→   572→
   573→   573→    // Footer
   574→   574→    if (this.opts.showFooter) {
   575→   575→      html += `<div class="cal-footer">`;
   576→   576→      if (this.opts.allowToday) html += `<button class="cal-foot-btn" data-act="today">Today</button>`;
   577→   577→      if (this.opts.allowClear) html += `<button class="cal-foot-btn" data-act="clear">Clear</button>`;
   578→   578→      html += `</div>`;
   579→   579→    }
   580→   580→
   581→   581→    this.root.innerHTML = html;
   582→   582→
   583→   583→    // Day cells are always rendered as CustomButton instances
   584→   584→    this.applyCustomButtonCells();
   585→   585→
   586→   586→    // Wire up events
   587→   587→    this.root.querySelectorAll("[data-act]").forEach((el) => {
   588→   588→      const act = (el as HTMLElement).dataset.act;
   589→   589→      el.addEventListener("click", (e) => {
   590→   590→        e.preventDefault();
   591→   591→        if (act === "prev") this.navigate("prev");
   592→   592→        else if (act === "next") this.navigate("next");
   593→   593→        else if (act === "today") {
   594→   594→          const t = today();
   595→   595→          this.navigateTo(t);
   596→   596→          this.focusedDate = t;
   597→   597→          this.render();
   598→   598→          this.emit("today_requested", { today: t, previousViewDate: this.opts.viewDate });
   599→   599→        }
   600→   600→        else if (act === "clear") {
   601→   601→          this.clearSelection();
   602→   602→          this.emit("clear_requested", {});
   603→   603→        }
   604→   604→      });
   605→   605→    });
   606→   606→
   607→   607→    this.root.querySelectorAll("[data-date]").forEach((el) => {
   608→   608→      const dk = (el as HTMLElement).dataset.date!;
   609→   609→      const [y, m, d] = dk.split("-").map(Number);
   610→   610→      const date: CalendarDate = { year: y, month: m, day: d };
   611→   611→      const isOutside = (el as HTMLElement).dataset.o === "1";
   612→   612→
   613→   613→      el.addEventListener("click", (e) => {
   614→   614→        e.preventDefault();
   615→   615→        if (this.opts.readOnly || this.opts.disabled) return;
   616→   616→        if (isOutside) {
   617→   617→          this.navigateTo({ year: date.year, month: date.month, day: 1 });
   618→   618→        }
   619→   619→        this.selectDate(date, "pointer");
   620→   620→      });
   621→   621→
   622→   622→      if (this.opts.hoverPreview && this.opts.selectionMode === "range" && this.opts.rangeStart && !this.opts.rangeEnd) {
   623→   623→        el.addEventListener("mouseenter", () => {
   624→   624→          this.previewEnd = date;
   625→   625→          this.render();
   626→   626→        });
   627→   627→      }
   628→   628→    });
   629→   629→  }

   631→   631→  /** Replace each .cal-cell with a CustomButton instance (variant="cell"). */
   632→   632→  private applyCustomButtonCells(): void {
   633→   633→    this.cellButtons = [];
   634→   634→    const cells = this.root.querySelectorAll(".cal-cell");
   635→   635→    cells.forEach((cellEl) => {
   636→   636→      const el = cellEl as HTMLElement;
   637→   637→      const text = el.textContent || "";
   638→   638→      const classes = el.className.split(" ").filter(c => c !== "cal-cell");
   639→   639→      const isDisabled = el.hasAttribute("disabled");
   640→   640→      const isSelected = classes.includes("cal-selected") || classes.includes("cal-range-start") || classes.includes("cal-range-end");
   641→   641→      const sz = this.opts.cellSize;
   642→   642→
   643→   643→      // Create a CustomButton in cell mode
   644→   644→      const wrapper = document.createElement("div");
   645→   645→      wrapper.style.display = "inline-flex";
   646→   646→      el.parentNode?.insertBefore(wrapper, el);
   647→   647→
   648→   648→      const btn = new CustomButton(wrapper, {
   649→   649→        text,
   650→   650→        variant: "cell",
   651→   651→        size: "sm",
   652→   652→        width: sz,
   653→   653→        height: sz,
   654→   654→        disabled: isDisabled,
   655→   655→        active: isSelected,
   656→   656→      });
   657→   657→
   658→   658→      // Copy data attributes for event wiring
   659→   659→      const dk = el.getAttribute("data-date");
   660→   660→      const o = el.getAttribute("data-o");
   661→   661→      if (dk) btn.getElement().setAttribute("data-date", dk);
   662→   662→      if (o) btn.getElement().setAttribute("data-o", o);
   663→   663→      btn.getElement().setAttribute("aria-label", el.getAttribute("aria-label") || "");
   664→   664→
   665→   665→      // Copy state classes
   666→   666→      for (const cls of classes) {
   667→   667→        if (cls.startsWith("cal-")) {
   668→   668→          btn.getElement().classList.add(cls);
   669→   669→        }
   670→   670→      }
   671→   671→
   672→   672→      // Track the button + its date key for per-instance prop updates
   673→   673→      if (dk) this.cellButtons.push({ button: btn, dateKey: dk });
   674→   674→
   675→   675→      // Remove the original cell
   676→   676→      el.remove();
   677→   677→    });
   678→   678→  }

   680→   680→  /**
   681→   681→   * Update a specific day cell's CustomButton properties at runtime.
   682→   682→   * dateKey format: "YYYY-MM-DD". Pass null to apply to ALL cells.
   683→   683→   */
   684→   684→  setCellProp(prop: string, value: unknown, dateKey: string | null = null): void {
   685→   685→    for (const entry of this.cellButtons) {
   686→   686→      if (dateKey !== null && entry.dateKey !== dateKey) continue;
   687→   687→      const btn = entry.button;
   688→   688→      switch (prop) {
   689→   689→        case "variant": btn.setVariant(value as any); break;
   690→   690→        case "text": btn.setText(value as string); break;
   691→   691→        case "icon": btn.setIcon(value as string); break;
   692→   692→        case "iconType": btn.setIcon(btn["options"].icon || "", value as any); break;
   693→   693→        case "iconPosition": btn.setIconPosition(value as any); break;
   694→   694→        case "size":
   695→   695→          btn.setSize(value as any);
   696→   696→          btn.setWidth(0);
   697→   697→          btn.setHeight(0);
   698→   698→          break;
   699→   699→        case "fullWidth": btn.setFullWidth(!!value); break;
   700→   700→        case "draggable": btn.setDraggable(!!value); break;
   701→   701→      }
   702→   702→    }
   703→   703→  }
