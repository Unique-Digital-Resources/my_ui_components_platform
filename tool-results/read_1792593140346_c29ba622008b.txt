   663→function generateApiCode(): string {

   675→interface ColorCircleOutput {

   684→  // Palette

   686→  getBaseIndex(): number;
   687→  setBaseIndex(i: number): void;       // change which color drives the palette
   688→  addColor(hue?: number, saturation?: number): number;  // returns new index
   689→  removeColor(i: number): void;

   701→  off(event: string, handler: Function): void;
   702→  destroy(): void;
   703→}

   705→// Behavior:
   706→//   - Moving the BASE color shifts all other colors (they maintain offsets).
