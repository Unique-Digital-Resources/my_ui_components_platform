     1→     1→/**
     2→     2→ * GanttTimeline — a Gantt Chart Timeline Bars component (vanilla TypeScript).
     3→     3→ *
     4→     4→ * Spec reference: §15 Minimum viable specification.
     5→     5→ *
     6→     6→ *   - Display generic items as horizontal bars.
     7→     7→ *   - Configure start and end fields independently of item content.
     8→     8→ *   - Map time intervals to a shared horizontal coordinate system.
     9→     9→ *   - Support configurable visible range, scale, and scrolling.
    10→    10→ *   - Support optional row assignment and row headers.
    11→    11→ *   - Provide a minimal default bar appearance.
    12→    12→ *   - Allow custom templates without requiring them.
    13→    13→ *   - Keep bar geometry independent of rendered content.
    14→    14→ *   - Support selection and optional move/resize interactions.
    15→    15→ *   - Define interval validation, snapping, and time-zone behavior.
    16→    16→ *   - Expose view-range, scale, selection, move, and resize events.
    17→    17→ *   - Support responsive layout and usable targets for short bars.
    18→    18→ *   - Provide keyboard alternatives and accessible item descriptions.
    19→    19→ *   - Separate visualization behavior from scheduling logic and persistence.
    20→    20→ *
    21→    21→ * Architecture: single-file DOM-based renderer with scoped CSS (`.gantt-`
    22→    22→ * prefix). Time is epoch milliseconds (number) throughout — the host converts
    23→    23→ * to/from timezones. The time-to-pixel mapping is linear: x(t) = W * (t - T0)
    24→    24→ * / (T1 - T0). Bar content is laid out by the content-template system
    25→    25→ * (src/lib/templates/); pass "none" to use native rendering (plain bar).
    26→    26→ */

   141→   141→
   142→   142→/** Default snap intervals per scale (in ms). */
   143→   143→const SCALE_SNAP: Record<GanttScale, number> = {
   144→   144→  hour: HOUR_MS,
   145→   145→  day: DAY_MS,
   146→   146→  week: 7 * DAY_MS,
   147→   147→  month: 30 * DAY_MS,

   152→   152→/** Header label format per scale. */
   153→   153→const SCALE_FORMAT: Record<GanttScale, (t: number, locale: string) => string> = {
   154→   154→  hour: (t, l) => new Date(t).toLocaleString(l, { month: "short", day: "numeric", hour: "2-digit" }),
   155→   155→  day: (t, l) => new Date(t).toLocaleDateString(l, { month: "short", day: "numeric" }),
   156→   156→  week: (t, l) => new Date(t).toLocaleDateString(l, { month: "short", day: "numeric" }),
   157→   157→  month: (t, l) => new Date(t).toLocaleDateString(l, { month: "short", year: "numeric" }),
   158→   158→  quarter: (t, l) => {
   159→   159→    const d = new Date(t);
   160→   160→    const q = Math.floor(d.getMonth() / 3) + 1;
   161→   161→    return `Q${q} ${d.getFullYear()}`;
   162→   162→  },
   163→   163→  year: (t) => String(new Date(t).getFullYear()),
   164→   164→};

   172→   172→  quarter: 30 * DAY_MS,

   543→   543→function uid(prefix: string): string {
   544→   544→  return `${prefix}-${Math.random().toString(36).slice(2, 9)}-${Date.now().toString(36).slice(-4)}`;

   547→   547→function clamp(n: number, min: number, max: number): number {
   548→   548→  return Math.max(min, Math.min(max, n));
   549→   549→}

   551→   551→/** Snap a time (epoch ms) to the nearest multiple of snapInterval. */
   552→   552→function snapTime(t: number, snap: number): number {
   553→   553→  if (!snap || snap <= 0) return t;
   554→   554→  return Math.round(t / snap) * snap;

   638→   638→  /** Convert a time (epoch ms) to a pixel x within the grid. */
   639→   639→  private timeToX(t: number): number {
   640→   640→    const range = this.opts.viewEnd - this.opts.viewStart;
   641→   641→    if (range <= 0) return 0;
   642→   642→    return ((t - this.opts.viewStart) / range) * this.getGridWidth();
   643→   643→  }
   644→   644→
   645→   645→  /** Convert a pixel x to a time (epoch ms). */
   646→   646→  private xToTime(x: number): number {
   647→   647→    const w = this.getGridWidth();
   648→   648→    if (w <= 0) return this.opts.viewStart;
   649→   649→    return this.opts.viewStart + (x / w) * (this.opts.viewEnd - this.opts.viewStart);
   650→   650→  }

   829→   829→  private renderRow(row: GanttRow): HTMLElement {

   840→   840→    for (const item of rowItems) {

   848→   848→  private renderBar(item: GanttItem, _row: GanttRow): HTMLElement {
   849→   849→    const el = document.createElement("div");
   850→   850→    el.className = "gantt-bar";
   851→   851→    el.setAttribute("data-item-id", item.id);
   852→   852→    el.setAttribute("role", "button");
   853→   853→    el.setAttribute("tabindex", this.opts.readOnly ? "-1" : "0");
   854→   854→    el.setAttribute(
   855→   855→      "aria-label",
   856→   856→      this.buildBarAriaLabel(item),
   857→   857→    );
   858→   858→    if (this.selectedItemId === item.id) el.classList.add("gantt-selected");
   859→   859→    if (this.opts.readOnly) el.classList.add("gantt-readonly-bar");
   860→   860→    if (item.disabled) el.classList.add("gantt-disabled-bar");
   861→   861→    if (item.color) el.style.background = item.color;
   862→   862→
   863→   863→    // Progress overlay
   864→   864→    if (this.opts.progressEnabled && typeof item.progress === "number" && item.progress > 0) {
   865→   865→      const prog = document.createElement("div");
   866→   866→      prog.className = "gantt-bar-progress";
   867→   867→      prog.style.width = `${clamp(item.progress, 0, 100)}%`;
   868→   868→      el.appendChild(prog);
   869→   869→    }
   870→   870→
   871→   871→    // Templated content
   872→   872→    const tpl = this.getTemplate();
   873→   873→    const density: TemplateDensity = "standard";
   874→   874→    const hostState = {
   875→   875→      readOnly: this.opts.readOnly,
   876→   876→      movable: this.opts.movable,
   877→   877→      resizable: this.opts.resizable,
   878→   878→      scale: this.opts.scale,
   879→   879→    };
   880→   880→    if (tpl) {
   881→   881→      for (const slot of tpl.slots) {
   882→   882→        if (!slotVisible(slot, density)) continue;
   883→   883→        const renderer = rendererRegistry.get(slot.renderer);
   884→   884→        if (!renderer) continue;
   885→   885→        const slotEl = renderer({ item, slot, density, host: hostState });
   886→   886→        if (slotEl) {
   887→   887→          slotEl.style.pointerEvents = "none";
   888→   888→          el.appendChild(slotEl);
   889→   889→        }
   890→   890→      }
   891→   891→    } else {
   892→   892→      // Native rendering: plain bar with optional title
   893→   893→      if (item.title) {
   894→   894→        const title = document.createElement("div");
   895→   895→        title.className = "gantt-bar-title";
   896→   896→        title.textContent = item.title;
   897→   897→        el.appendChild(title);
   898→   898→      }
   899→   899→    }
   900→   900→
   901→   901→    // Resize handles
   902→   902→    if (this.opts.resizable && !this.opts.readOnly && !item.disabled) {
   903→   903→      const left = document.createElement("div");
   904→   904→      left.className = "gantt-resize-handle gantt-resize-handle-left";
   905→   905→      left.setAttribute("role", "slider");
   906→   906→      left.setAttribute("aria-label", `Resize start of ${item.title ?? item.id}`);
   907→   907→      left.addEventListener("pointerdown", (e) => this.onResizeStart(e, item, "start"));
   908→   908→      el.appendChild(left);
   909→   909→
   910→   910→      const right = document.createElement("div");
   911→   911→      right.className = "gantt-resize-handle gantt-resize-handle-right";
   912→   912→      right.setAttribute("role", "slider");
   913→   913→      right.setAttribute("aria-label", `Resize end of ${item.title ?? item.id}`);
   914→   914→      right.addEventListener("pointerdown", (e) => this.onResizeStart(e, item, "end"));
   915→   915→      el.appendChild(right);
   916→   916→    }
   917→   917→
   918→   918→    // Pointer interactions
   919→   919→    if (!this.opts.readOnly && !item.disabled) {
   920→   920→      el.addEventListener("pointerdown", (e) => this.onBarPointerDown(e, item));
   921→   921→      el.addEventListener("click", (e) => {
   922→   922→        const target = e.target as HTMLElement;
   923→   923→        if (target.closest(".gantt-resize-handle")) return;
   924→   924→        if (this.opts.selectable) this.selectBar(item.id, "pointer");
   925→   925→      });
   926→   926→      el.addEventListener("dblclick", () => {
   927→   927→        this.emit("bar_activated", { itemId: item.id, source: "pointer" });
   928→   928→      });
   929→   929→      el.addEventListener("keydown", (e) => this.onBarKeyDown(e, item));
   930→   930→    } else if (this.opts.selectable) {
   931→   931→      el.addEventListener("click", () => this.selectBar(item.id, "pointer"));
   932→   932→    }
   933→   933→
   934→   934→    // Position the bar
   935→   935→    this.positionBar(el, item);
   936→   936→    return el;
   937→   937→  }

  1230→  1230→  private onDragMove(e: PointerEvent): void {
  1231→  1231→    if (!this.dragItemId || !this.dragMode) return;
  1232→  1232→    const item = this.findItem(this.dragItemId);
  1233→  1233→    if (!item) return;
  1234→  1234→    const dx = e.clientX - this.dragStartX;
  1235→  1235→    if (Math.abs(dx) < 2 && !this.dragHappened) return;
  1236→  1236→    this.dragHappened = true;
  1237→  1237→
  1238→  1238→    const dt = this.xToTime(this.timeToX(this.dragOriginalStart) + dx) - this.dragOriginalStart;
  1239→  1239→    const snap = this.opts.snapInterval;
  1240→  1240→
  1241→  1241→    if (this.dragMode === "move") {
  1242→  1242→      const newStart = snapTime(this.dragOriginalStart + dt, snap);
  1243→  1243→      const newEnd = snapTime(this.dragOriginalEnd + dt, snap);
  1244→  1244→      // Update visual position without committing
  1245→  1245→      item.start = newStart;
  1246→  1246→      item.end = newEnd;
  1247→  1247→      const el = this.barEls.get(item.id);
  1248→  1248→      if (el) this.positionBar(el, item);
  1249→  1249→      this.emit("bar_move_requested", {
  1250→  1250→        itemId: item.id,
  1251→  1251→        oldStart: this.dragOriginalStart,
  1252→  1252→        oldEnd: this.dragOriginalEnd,
  1253→  1253→        start: newStart,
  1254→  1254→        end: newEnd,
  1255→  1255→        source: "pointer",
  1256→  1256→      });
  1257→  1257→    } else if (this.dragMode === "resize-start") {
  1258→  1258→      const newStart = snapTime(this.dragOriginalStart + dt, snap);
  1259→  1259→      if (newStart < this.dragOriginalEnd - (this.opts.minDuration || 1)) {
  1260→  1260→        item.start = newStart;
  1261→  1261→        const el = this.barEls.get(item.id);
  1262→  1262→        if (el) this.positionBar(el, item);
  1263→  1263→        this.emit("bar_resize_requested", {
  1264→  1264→          itemId: item.id,
  1265→  1265→          endpoint: "start",
  1266→  1266→          oldStart: this.dragOriginalStart,
  1267→  1267→          start: newStart,
  1268→  1268→          source: "pointer",
  1269→  1269→        });
  1270→  1270→      }
  1271→  1271→    } else if (this.dragMode === "resize-end") {
  1272→  1272→      const newEnd = snapTime(this.dragOriginalEnd + dt, snap);
  1273→  1273→      if (newEnd > this.dragOriginalStart + (this.opts.minDuration || 1)) {
  1274→  1274→        item.end = newEnd;
  1275→  1275→        const el = this.barEls.get(item.id);
  1276→  1276→        if (el) this.positionBar(el, item);
  1277→  1277→        this.emit("bar_resize_requested", {
  1278→  1278→          itemId: item.id,
  1279→  1279→          endpoint: "end",
  1280→  1280→          oldEnd: this.dragOriginalEnd,
  1281→  1281→          end: newEnd,
  1282→  1282→          source: "pointer",
  1283→  1283→        });
  1284→  1284→      }
  1285→  1285→    }
  1286→  1286→  }

  1288→  1288→  private onDragEnd(_e: PointerEvent, onMove: (ev: PointerEvent) => void, onUp: (ev: PointerEvent) => void): void {
  1289→  1289→    window.removeEventListener("pointermove", onMove);

  1295→  1295→

  1299→  1299→          itemId: item.id,
  1300→  1300→          oldStart: this.dragOriginalStart,
  1301→  1301→          oldEnd: this.dragOriginalEnd,

  1303→  1303→          newEnd: item.end,
  1304→  1304→          source: "pointer",
  1305→  1305→        });
  1306→  1306→      } else {
  1307→  1307→        this.emit("bar_resized", {
  1308→  1308→          itemId: item.id,

  1310→  1310→          oldEnd: this.dragOriginalEnd,
  1311→  1311→          newStart: item.start,
  1312→  1312→          newEnd: item.end,

  1316→  1316→      }
  1317→  1317→    } else if (!this.dragHappened) {

  1319→  1319→    }
  1320→  1320→
  1321→  1321→    this.dragItemId = null;

  1323→  1323→    this.dragHappened = false;

  1339→  1339→      case "ArrowLeft":
  1340→  1340→        if (this.opts.movable && !e.shiftKey) {

  1357→  1357→        }

  1359→  1359→      case "ArrowDown":

  1363→  1363→        }
  1364→  1364→        break;
  1365→  1365→      case "Home":
  1366→  1366→        if (this.opts.resizable) this.resizeBar(item.id, "start", this.opts.viewStart, "keyboard");
  1367→  1367→        break;
  1368→  1368→      case "End":
  1369→  1369→        if (this.opts.resizable) this.resizeBar(item.id, "end", this.opts.viewEnd, "keyboard");
  1370→  1370→        break;
  1371→  1371→      case "Delete":
  1372→  1372→      case "Backspace":
  1373→  1373→        this.emit("bar_delete_requested", { itemId: item.id, source: "keyboard" });
  1374→  1374→        break;
  1375→  1375→      case "Escape":

  1377→  1377→        break;
  1378→  1378→      default:
  1379→  1379→        handled = false;
  1380→  1380→    }
  1381→  1381→    if (handled) e.preventDefault();
  1382→  1382→  }
