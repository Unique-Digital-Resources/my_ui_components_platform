     1→     1→/**
     2→     2→ * ColorWheelPicker — a vanilla TypeScript color picker component.
     3→     3→ *
     4→     4→ * Renders a hue wheel surrounding a configurable inner selection shape
     5→     5→ * (triangle / square / disk / none). Canvas-based, pointer + keyboard
     6→     6→ * interaction, framework-independent. No React or other UI framework.
     7→     7→ *
     8→     8→ * ----------------------------------------------------------------------------
     9→     9→ * GEOMETRY
    10→    10→ * ----------------------------------------------------------------------------
    11→    11→ *   `size`            — diameter (px) of the inner shape's bounding circle.
    12→    12→ *   `wheelSpacing`    — gap (px) between the inner shape and the wheel's
    13→    13→ *                       inner edge.
    14→    14→ *   `wheelThickness`  — radial thickness (px) of the hue ring.
    15→    15→ *
    16→    16→ *   innerR       = size / 2
    17→    17→ *   wheelInnerR  = innerR + wheelSpacing
    18→    18→ *   wheelOuterR  = wheelInnerR + wheelThickness
    19→    19→ *   total        = wheelOuterR * 2  (full component size in CSS pixels)
    20→    20→ *   cx = cy      = total / 2         (wheel is centered on the inner shape)
    21→    21→ *
    22→    22→ * ----------------------------------------------------------------------------
    23→    23→ * HUE MAPPING
    24→    24→ * ----------------------------------------------------------------------------
    25→    25→ *   Hue 0° (red) is at the top (12 o'clock), increasing clockwise.
    26→    26→ *   Canvas y-axis is inverted (+y is down) so "clockwise" on screen equals
    27→    27→ *   increasing canvas angle.
    28→    28→ *
    29→    29→ *     angle(hue)  = -π/2 + hue * π/180            (canvas radians)
    30→    30→ *     hue(angle)  = ((angle + π/2) * 180/π) mod 360
    31→    31→ *
    32→    32→ *   Equivalently, using atan2 from the "up" vector:
    33→    33→ *     theta = atan2(dx, -dy)   (in radians, 0 = top, increasing clockwise)
    34→    34→ *
    35→    35→ * ----------------------------------------------------------------------------
    36→    36→ * INNER SHAPE COLOR MAPPINGS (hue from the wheel, fixed)
    37→    37→ * ----------------------------------------------------------------------------
    38→    38→ *   triangle — Vertices: A = top (S=0, V=1, white), B = bottom-right
    39→    39→ *              (S=1, V=1, pure hue), C = bottom-left (V=0, black).
    40→    40→ *              For any interior point P = α·A + β·B + γ·C with α+β+γ=1:
    41→    41→ *                 V = α + β
    42→    42→ *                 S = β / V   (when V > 0; S is undefined at the black vertex)
    43→    43→ *              Inverse:
    44→    44→ *                 α = V·(1 − S),  β = S·V,  γ = 1 − V
    45→    45→ *
    46→    46→ *   square   — Inscribed square (side = innerR · √2) inside the inner circle.
    47→    47→ *              x → saturation (0..1, left → right)
    48→    48→ *              y → value      (1..0, top → bottom)
    49→    49→ *
    50→    50→ *   disk     — r → saturation (0..1, center → edge)
    51→    51→ *              θ → value, mapped via V = (1 + cos θ) / 2.
    52→    52→ *              This is a smooth, seam-free mapping. The top of the disk is
    53→    53→ *              V = 1 (bright), the bottom is V = 0 (black); left/right at
    54→    54→ *              V = 0.5 are mirror images. Because two θ values can produce
    55→    55→ *              the same V, the component tracks the user's last angular
    56→    56→ *              position (`diskTheta`) so the inner marker stays where the
    57→    57→ *              user dragged it.
    58→    58→ *
    59→    59→ *   none     — Only the wheel renders; the inner area shows the current
    60→    60→ *              solid color and is not interactive.
    61→    61→ *
    62→    62→ * ----------------------------------------------------------------------------
    63→    63→ * RENDERING STRATEGY
    64→    64→ * ----------------------------------------------------------------------------
    65→    65→ *   A single <canvas> renders the whole component:
    66→    66→ *     1. Hue wheel   — drawn as 360 one-degree arc segments (cached).
    67→    67→ *     2. Inner shape — drawn pixel-by-pixel (triangle/disk) or with three
    68→    68→ *                      canvas gradients (square). Cached per (shape, size, hue).
    69→    69→ *     3. Markers     — hue marker on the wheel ring; inner marker on the
    70→    70→ *                      selected (S, V) position. Re-drawn every frame (cheap).
    71→    71→ *
    72→    72→ *   Caching keeps drag interactions smooth: SV drags only redraw markers,
    73→    73→ *   and hue drags only re-render the inner shape (not the wheel).
    74→    74→ */

   137→   137→/**
   138→   138→ * Rotate a point `(p.x, p.y)` around the center `(cx, cy)` by `angle`
   139→   139→ * radians (0 = top / 12 o'clock, increasing clockwise).
   140→   140→ */
   141→   141→function rotatePoint(
   142→   142→  p: { x: number; y: number },
   143→   143→  cx: number,
   144→   144→  cy: number,
   145→   145→  angle: number,
   146→   146→): { x: number; y: number } {
   147→   147→  if (angle === 0) return p;
   148→   148→  const cos = Math.cos(angle);
   149→   149→  const sin = Math.sin(angle);
   150→   150→  const dx = p.x - cx;
   151→   151→  const dy = p.y - cy;
   152→   152→  // Standard 2D rotation. In canvas coordinates (y-down), a positive angle
   153→   153→  // rotates clockwise when using this matrix.
   154→   154→  return {
   155→   155→    x: cx + dx * cos - dy * sin,
   156→   156→    y: cy + dx * sin + dy * cos,
   157→   157→  };
   158→   158→}

   231→   231→  constructor(container: HTMLElement, options: ColorPickerOptions = {}) {

   234→   234→    }

   236→   236→
   237→   237→    // Resolve options (per-option defaults; `color` overrides `hue`).
   238→   238→    this.options = {
   239→   239→      shape: (options.shape as Shape) ?? DEFAULTS.shape,
   240→   240→      size: Math.max(40, options.size ?? DEFAULTS.size),
   241→   241→      wheelThickness: Math.max(4, options.wheelThickness ?? DEFAULTS.wheelThickness),

   396→   396→  on(event: string, handler: ColorHandler): void {
   397→   397→    let set = this.listeners.get(event);
   398→   398→    if (!set) {
   399→   399→      set = new Set<ColorHandler>();
   400→   400→      this.listeners.set(event, set);
   401→   401→    }
   402→   402→    set.add(handler);
   403→   403→  }

   408→   408→    if (set) {
   409→   409→      set.delete(handler);
   410→   410→    }
   411→   411→  }

   413→   413→  /** Tear down: remove listeners, observers, and DOM. Idempotent. */
   414→   414→  destroy(): void {

   416→   416→    this._destroyed = true;
   417→   417→
   418→   418→    this.canvas.removeEventListener("pointerdown", this._onPointerDown);

   420→   420→    this.canvas.removeEventListener("pointerup", this._onPointerUp);
   421→   421→    this.canvas.removeEventListener("pointercancel", this._onPointerUp);

   426→   426→        this.canvas.releasePointerCapture(this.pointerState.pointerId);
   427→   427→      } catch {
   428→   428→        /* ignore */
   429→   429→      }
   430→   430→      this.pointerState = null;
   431→   431→    }
   432→   432→
   433→   433→    if (this.resizeObserver) {
   434→   434→      this.resizeObserver.disconnect();
   435→   435→      this.resizeObserver = null;
   436→   436→    }
   437→   437→
   438→   438→    this.listeners.clear();
   439→   439→
   440→   440→    if (this.canvas.parentNode) {
   441→   441→      this.canvas.parentNode.removeChild(this.canvas);
   442→   442→    }
   443→   443→    this.container.classList.remove("color-wheel-picker");
   444→   444→
   445→   445→    this.ctx.clearRect(0, 0, this.canvas.width, this.canvas.height);

   448→   448→    this.innerCache = null;

   456→   456→  private getGeometry(): Geometry {
   457→   457→    const { size, wheelThickness, wheelSpacing } = this.options;
   458→   458→    const innerR = size / 2;
   459→   459→    const wheelInnerR = innerR + wheelSpacing;
   460→   460→    const wheelOuterR = wheelInnerR + wheelThickness;
   461→   461→    const total = wheelOuterR * 2;
   462→   462→    return {
   463→   463→      innerR,
   464→   464→      wheelInnerR,
   465→   465→      wheelOuterR,
   466→   466→      total,
   467→   467→      cx: total / 2,
   468→   468→      cy: total / 2,
   469→   469→    };
   470→   470→  }

   476→   476→  private updateCanvasSize(): void {
   477→   477→    const geom = this.getGeometry();
   478→   478→    const dpr =
   479→   479→      (typeof window !== "undefined" && window.devicePixelRatio) || 1;
   480→   480→    this.canvas.style.width = `${geom.total}px`;
   481→   481→    this.canvas.style.height = `${geom.total}px`;
   482→   482→    this.canvas.width = Math.round(geom.total * dpr);
   483→   483→    this.canvas.height = Math.round(geom.total * dpr);
   484→   484→    this.ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
   485→   485→  }
   486→   486→
   487→   487→  private handleResize(): void {
   488→   488→    if (this._destroyed) return;
   489→   489→    // The canvas's CSS size may have changed (host CSS) or the device pixel
   490→   490→    // ratio may have changed (window moved between monitors). Refresh the
   491→   491→    // backing store and re-render.
   492→   492→    const dpr =
   493→   493→      (typeof window !== "undefined" && window.devicePixelRatio) || 1;
   494→   494→    const geom = this.getGeometry();
   495→   495→    const expectedW = Math.round(geom.total * dpr);
   496→   496→    if (this.canvas.width !== expectedW) {
   497→   497→      this.updateCanvasSize();
   498→   498→      this.invalidateWheelCache();
   499→   499→      this.invalidateInnerCache();
   500→   500→    }
   501→   501→    this.scheduleRender();
   502→   502→  }
   503→   503→
   504→   504→  private scheduleRender(): void {
   505→   505→    if (this._destroyed) return;
   506→   506→    if (this._renderScheduled) return;
   507→   507→    this._renderScheduled = true;
   508→   508→    requestAnimationFrame(() => {
   509→   509→      this._renderScheduled = false;
   510→   510→      if (!this._destroyed) this.render();
   511→   511→    });
   512→   512→  }
   513→   513→
   514→   514→  private invalidateWheelCache(): void {
   515→   515→    this.wheelCacheKey = "";
   516→   516→  }
   517→   517→
   518→   518→  private invalidateInnerCache(): void {
   519→   519→    this.innerCacheKey = "";
   520→   520→  }
   521→   521→
   522→   522→  private render(): void {
   523→   523→    const geom = this.getGeometry();
   524→   524→    const { ctx } = this;
   525→   525→    ctx.clearRect(0, 0, geom.total, geom.total);
   526→   526→    this.renderWheel(geom);
   527→   527→    this.renderInner(geom);
   528→   528→    this.renderMarkers(geom);
   529→   529→  }

   531→   531→  // ---- Hue wheel ----------------------------------------------------------
   532→   532→
   533→   533→  private renderWheel(geom: Geometry): void {
   534→   534→    const dpr =
   535→   535→      (typeof window !== "undefined" && window.devicePixelRatio) || 1;
   536→   536→    const key = `wheel-${geom.total}-${dpr}`;
   537→   537→    if (key !== this.wheelCacheKey || !this.wheelCache || !this.wheelCacheCtx) {
   538→   538→      if (!this.wheelCache) {
   539→   539→        this.wheelCache = document.createElement("canvas");
   540→   540→        this.wheelCacheCtx = this.wheelCache.getContext("2d");
   541→   541→      }
   542→   542→      const cache = this.wheelCache;
   543→   543→      const cacheCtx = this.wheelCacheCtx!;
   544→   544→      const backing = Math.round(geom.total * dpr);
   545→   545→      if (cache.width !== backing) {
   546→   546→        cache.width = backing;
   547→   547→        cache.height = backing;
   548→   548→      }
   549→   549→      cacheCtx.setTransform(dpr, 0, 0, dpr, 0, 0);
   550→   550→      cacheCtx.clearRect(0, 0, geom.total, geom.total);
   551→   551→      this.drawWheelTo(cacheCtx, geom);
   552→   552→      this.wheelCacheKey = key;
   553→   553→    }
   554→   554→    this.ctx.drawImage(this.wheelCache, 0, 0, geom.total, geom.total);
   555→   555→  }
   556→   556→
   557→   557→  /**
   558→   558→   * Draw the hue wheel directly into a target context. The wheel is composed
   559→   559→   * of 360 one-degree segments, each filled with the HSL color of its hue.
   560→   560→   */
   561→   561→  private drawWheelTo(ctx: CanvasRenderingContext2D, geom: Geometry): void {
   562→   562→    const { cx, cy, wheelInnerR, wheelOuterR } = geom;
   563→   563→    const segments = 360;
   564→   564→    const inner = Math.max(0.5, wheelInnerR);
   565→   565→    const outer = Math.max(inner + 0.5, wheelOuterR);
   566→   566→
   567→   567→    for (let i = 0; i < segments; i++) {
   568→   568→      // Start at the top (-π/2) and go clockwise (increasing angle).
   569→   569→      const a0 = -Math.PI / 2 + (i / segments) * Math.PI * 2;
   570→   570→      const a1 = -Math.PI / 2 + ((i + 1) / segments) * Math.PI * 2;
   571→   571→      ctx.beginPath();
   572→   572→      ctx.arc(cx, cy, outer, a0, a1);
   573→   573→      ctx.arc(cx, cy, inner, a1, a0, true);
   574→   574→      ctx.closePath();
   575→   575→      ctx.fillStyle = `hsl(${i}, 100%, 50%)`;
   576→   576→      ctx.fill();
   577→   577→    }
   578→   578→  }
   579→   579→
   580→   580→  // ---- Inner shape --------------------------------------------------------
   581→   581→
   582→   582→  private renderInner(geom: Geometry): void {
   583→   583→    const { shape } = this.options;
   584→   584→    if (shape === "none") {
   585→   585→      this.renderNone(geom);
   586→   586→      return;
   587→   587→    }
   588→   588→
   589→   589→    const dpr =
   590→   590→      (typeof window !== "undefined" && window.devicePixelRatio) || 1;
   591→   591→    const rotation = this.getShapeRotation();
   592→   592→    const key = `inner-${shape}-${geom.total}-${dpr}-${this.hsv.h}-${rotation}`;
   593→   593→    if (key !== this.innerCacheKey || !this.innerCache || !this.innerCacheCtx) {
   594→   594→      if (!this.innerCache) {
   595→   595→        this.innerCache = document.createElement("canvas");
   596→   596→        this.innerCacheCtx = this.innerCache.getContext("2d");
   597→   597→      }
   598→   598→      const cache = this.innerCache;
   599→   599→      const cacheCtx = this.innerCacheCtx!;
   600→   600→      const backing = Math.round(geom.total * dpr);
   601→   601→      if (cache.width !== backing) {
   602→   602→        cache.width = backing;
   603→   603→        cache.height = backing;
   604→   604→      }
   605→   605→      cacheCtx.setTransform(dpr, 0, 0, dpr, 0, 0);
   606→   606→      cacheCtx.clearRect(0, 0, geom.total, geom.total);
   607→   607→
   608→   608→      if (shape === "triangle") this.drawTriangleTo(cacheCtx, geom, rotation);
   609→   609→      else if (shape === "square") this.drawSquareTo(cacheCtx, geom);
   610→   610→      else if (shape === "disk") this.drawDiskTo(cacheCtx, geom);
   611→   611→
   612→   612→      this.innerCacheKey = key;
   613→   613→    }
   614→   614→
   615→   615→    this.ctx.drawImage(this.innerCache, 0, 0, geom.total, geom.total);
   616→   616→  }
   617→   617→
   618→   618→  /**
   619→   619→   * Returns the rotation angle (radians, 0 = top, clockwise) applied to the
   620→   620→   * inner shape. When `rotateShape` is enabled, the triangle's pure-hue
   621→   621→   * vertex tracks the current hue. For other shapes, always 0.
   622→   622→   */
   623→   623→  private getShapeRotation(): number {
   624→   624→    if (!this.options.rotateShape) return 0;
   625→   625→    if (this.options.shape !== "triangle") return 0;
   626→   626→    // The triangle's pure-hue vertex (B) sits at 120° clockwise from top in
   627→   627→    // its default orientation. Rotate so that vertex aligns with the hue.
   628→   628→    const hueRad = (this.hsv.h * Math.PI) / 180;
   629→   629→    const bAngleRad = (120 * Math.PI) / 180;
   630→   630→    return hueRad - bAngleRad;
   631→   631→  }
   632→   632→
   633→   633→  /** None mode: solid filled inner circle in the current color. */
   634→   634→  private renderNone(geom: Geometry): void {
   635→   635→    const { cx, cy, innerR } = geom;
   636→   636→    const { ctx, hsv } = this;
   637→   637→    const rgb = hsvToRgb(hsv.h, hsv.s, hsv.v);
   638→   638→    ctx.beginPath();
   639→   639→    ctx.arc(cx, cy, Math.max(0.5, innerR), 0, Math.PI * 2);
   640→   640→    ctx.fillStyle = `rgb(${rgb.r}, ${rgb.g}, ${rgb.b})`;
   641→   641→    ctx.fill();
   642→   642→  }
   643→   643→
   644→   644→  /**
   645→   645→   * Square inner shape: inscribed square (side = innerR · √2) with three
   646→   646→   * stacked layers — pure hue → white-to-transparent (S) → transparent-to-
   647→   647→   * black (V).
   648→   648→   */
   649→   649→  private drawSquareTo(ctx: CanvasRenderingContext2D, geom: Geometry): void {
   650→   650→    const { cx, cy, innerR } = geom;
   651→   651→    const { hsv } = this;
   652→   652→    const side = innerR * Math.SQRT2;
   653→   653→    const half = side / 2;
   654→   654→    const left = cx - half;
   655→   655→    const top = cy - half;
   656→   656→
   657→   657→    ctx.save();
   658→   658→    // Clip to the square so corners don't bleed into the wheel gap.
   659→   659→    ctx.beginPath();
   660→   660→    ctx.rect(left, top, side, side);
   661→   661→    ctx.clip();
   662→   662→
   663→   663→    // 1) Pure hue base.
   664→   664→    ctx.fillStyle = `hsl(${hsv.h}, 100%, 50%)`;
   665→   665→    ctx.fillRect(left, top, side, side);
   666→   666→
   667→   667→    // 2) White → transparent (saturation, left → right).
   668→   668→    const gSat = ctx.createLinearGradient(left, 0, left + side, 0);
   669→   669→    gSat.addColorStop(0, "rgba(255, 255, 255, 1)");
   670→   670→    gSat.addColorStop(1, "rgba(255, 255, 255, 0)");
   671→   671→    ctx.fillStyle = gSat;
   672→   672→    ctx.fillRect(left, top, side, side);
   673→   673→
   674→   674→    // 3) Transparent → black (value, top → bottom).
   675→   675→    const gVal = ctx.createLinearGradient(0, top, 0, top + side);
   676→   676→    gVal.addColorStop(0, "rgba(0, 0, 0, 0)");
   677→   677→    gVal.addColorStop(1, "rgba(0, 0, 0, 1)");
   678→   678→    ctx.fillStyle = gVal;
   679→   679→    ctx.fillRect(left, top, side, side);
   680→   680→
   681→   681→    ctx.restore();
   682→   682→  }
   683→   683→
   684→   684→  /**
   685→   685→   * Triangle inner shape: equilateral triangle inscribed in the inner circle.
   686→   686→   *   A = top             (S = 0, V = 1)  — white
   687→   687→   *   B = bottom-right    (S = 1, V = 1)  — pure hue
   688→   688→   *   C = bottom-left     (S = *, V = 0)  — black
   689→   689→   *
   690→   690→   * When `rotation` is non-zero, all three vertices are rotated around the
   691→   691→   * center by that angle (radians, 0 = top, clockwise) — used by the
   692→   692→   * `rotateShape` option so the pure-hue vertex tracks the wheel hue.
   693→   693→   *
   694→   694→   * Rendered per-pixel via barycentric coordinates with 2×2 supersampling
   695→   695→   * for smooth anti-aliased edges. A temp offscreen canvas is used because
   696→   696→   * `putImageData` ignores the context transform (dpr scaling).
   697→   697→   */
   698→   698→  private drawTriangleTo(
   699→   699→    ctx: CanvasRenderingContext2D,
   700→   700→    geom: Geometry,
   701→   701→    rotation: number,
   702→   702→  ): void {
   703→   703→    const { cx, cy, innerR } = geom;
   704→   704→    const r = Math.max(1, innerR);
   705→   705→
   706→   706→    // Base vertices (rotation = 0): A at top, B at bottom-right, C at bottom-left.
   707→   707→    const baseA = { x: cx, y: cy - r };
   708→   708→    const baseB = { x: cx + (r * Math.sqrt(3)) / 2, y: cy + r / 2 };
   709→   709→    const baseC = { x: cx - (r * Math.sqrt(3)) / 2, y: cy + r / 2 };
   710→   710→
   711→   711→    const verts = [baseA, baseB, baseC].map((p) =>
   712→   712→      rotatePoint(p, cx, cy, rotation),
   713→   713→    );
   714→   714→
   715→   715→    this.drawBarycentric(ctx, verts);
   716→   716→  }
   717→   717→
   718→   718→  /**
   719→   719→   * Per-pixel triangle renderer using barycentric coordinates with 2×2
   720→   720→   * supersampling. Renders at device resolution into a temp canvas, then
   721→   721→   * composites via `drawImage` (which respects the dpr transform) —
   722→   722→   * `putImageData` would ignore the transform and misplace the image on
   723→   723→   * high-DPI displays.
   724→   724→   *
   725→   725→   * For a point P with barycentric (α, β, γ) w.r.t. (A, B, C):
   726→   726→   *   P = α·A + β·B + γ·C,   α + β + γ = 1
   727→   727→   *   V = α + β       (white + pure-hue weights)
   728→   728→   *   S = β / V       (pure-hue share of the non-black weights)
   729→   729→   */
   730→   730→  private drawBarycentric(
   731→   731→    ctx: CanvasRenderingContext2D,
   732→   732→    verts: { x: number; y: number }[],
   733→   733→  ): void {
   734→   734→    const { hsv } = this;
   735→   735→    const [A, B, C] = verts;
   736→   736→
   737→   737→    // CSS-pixel bounding box.
   738→   738→    const minX = Math.floor(Math.min(A.x, B.x, C.x));
   739→   739→    const maxX = Math.ceil(Math.max(A.x, B.x, C.x));
   740→   740→    const minY = Math.floor(Math.min(A.y, B.y, C.y));
   741→   741→    const maxY = Math.ceil(Math.max(A.y, B.y, C.y));
   742→   742→    const cssW = maxX - minX;
   743→   743→    const cssH = maxY - minY;
   744→   744→    if (cssW <= 0 || cssH <= 0) return;
   745→   745→
   746→   746→    // Device-pixel dimensions for crisp rendering on high-DPI displays.
   747→   747→    const dpr =
   748→   748→      (typeof window !== "undefined" && window.devicePixelRatio) || 1;
   749→   749→    const dw = Math.max(1, Math.round(cssW * dpr));
   750→   750→    const dh = Math.max(1, Math.round(cssH * dpr));
   751→   751→
   752→   752→    // Barycentric determinant and inverse (computed in CSS-pixel space).
   753→   753→    const det = (B.y - C.y) * (A.x - C.x) + (C.x - B.x) * (A.y - C.y);
   754→   754→    if (Math.abs(det) < 1e-9) return;
   755→   755→    const invDet = 1 / det;
   756→   756→
   757→   757→    const hue = hsv.h;
   758→   758→
   759→   759→    // Build the ImageData at device resolution.
   760→   760→    const img = ctx.createImageData(dw, dh);
   761→   761→    const data = img.data;
   762→   762→
   763→   763→    for (let py = 0; py < dh; py++) {
   764→   764→      for (let px = 0; px < dw; px++) {
   765→   765→        // Convert device pixel to CSS-pixel sample point (pixel center).
   766→   766→        const cssPx = minX + (px + 0.5) / dpr;
   767→   767→        const cssPy = minY + (py + 0.5) / dpr;
   768→   768→
   769→   769→        // 2×2 supersampling: sample 4 sub-pixels for smooth edges.
   770→   770→        let rSum = 0,
   771→   771→          gSum = 0,
   772→   772→          bSum = 0,
   773→   773→          count = 0;
   774→   774→        for (let sy = 0; sy < 2; sy++) {
   775→   775→          for (let sx = 0; sx < 2; sx++) {
   776→   776→            const subPx = cssPx + ((sx - 0.5) * 0.5) / dpr;
   777→   777→            const subPy = cssPy + ((sy - 0.5) * 0.5) / dpr;
   778→   778→            const a =
   779→   779→              ((B.y - C.y) * (subPx - C.x) + (C.x - B.x) * (subPy - C.y)) *
   780→   780→              invDet;
   781→   781→            const b =
   782→   782→              ((C.y - A.y) * (subPx - C.x) + (A.x - C.x) * (subPy - C.y)) *
   783→   783→              invDet;
   784→   784→            const g = 1 - a - b;
   785→   785→            if (a >= 0 && b >= 0 && g >= 0) {
   786→   786→              const V = a + b;
   787→   787→              const S = V > 0 ? b / V : 0;
   788→   788→              const rgb = hsvToRgb(hue, S * 100, V * 100);
   789→   789→              rSum += rgb.r;
   790→   790→              gSum += rgb.g;
   791→   791→              bSum += rgb.b;
   792→   792→              count++;
   793→   793→            }
   794→   794→          }
   795→   795→        }
   796→   796→        if (count === 0) continue;
   797→   797→        const aa = count / 4;
   798→   798→        const i = (py * dw + px) * 4;
   799→   799→        data[i] = Math.round(rSum / count);
   800→   800→        data[i + 1] = Math.round(gSum / count);
   801→   801→        data[i + 2] = Math.round(bSum / count);
   802→   802→        data[i + 3] = Math.round(aa * 255);
   803→   803→      }
   804→   804→    }
   805→   805→
   806→   806→    // Composite via a temp canvas + drawImage so the dpr transform is
   807→   807→    // respected. putImageData would ignore it and misplace the image.
   808→   808→    const temp = document.createElement("canvas");
   809→   809→    temp.width = dw;
   810→   810→    temp.height = dh;
   811→   811→    const tctx = temp.getContext("2d");
   812→   812→    if (!tctx) return;
   813→   813→    tctx.putImageData(img, 0, 0);
   814→   814→    // drawImage respects the ctx transform — (minX, minY) are CSS coords.
   815→   815→    ctx.drawImage(temp, minX, minY);
   816→   816→  }
   817→   817→
   818→   818→  /**
   819→   819→   * Disk inner shape: polar mapping
   820→   820→   *   r → saturation (0 at center, 1 at edge)
   821→   821→   *   θ → value via V = (1 + cos θ) / 2
   822→   822→   *
   823→   823→   * Smooth and seam-free. Center maps to S = 0; angle at center is undefined
   824→   824→   * so V defaults to 1 (bright white center). Renders at device resolution
   825→   825→   * into a temp canvas, then composites via `drawImage` so the dpr transform
   826→   826→   * is respected (`putImageData` would ignore it).
   827→   827→   */
   828→   828→  private drawDiskTo(ctx: CanvasRenderingContext2D, geom: Geometry): void {
   829→   829→    const { cx, cy, innerR } = geom;
   830→   830→    const { hsv } = this;
   831→   831→    const r = Math.max(1, innerR);
   832→   832→
   833→   833→    const minX = Math.floor(cx - r);
   834→   834→    const minY = Math.floor(cy - r);
   835→   835→    const cssW = Math.ceil(2 * r);
   836→   836→    const cssH = cssW;
   837→   837→    if (cssW <= 0 || cssH <= 0) return;
   838→   838→
   839→   839→    const dpr =
   840→   840→      (typeof window !== "undefined" && window.devicePixelRatio) || 1;
   841→   841→    const dw = Math.max(1, Math.round(cssW * dpr));
   842→   842→    const dh = Math.max(1, Math.round(cssH * dpr));
   843→   843→
   844→   844→    const img = ctx.createImageData(dw, dh);
   845→   845→    const data = img.data;
   846→   846→
   847→   847→    const hue = hsv.h;
   848→   848→
   849→   849→    for (let py = 0; py < dh; py++) {
   850→   850→      for (let px = 0; px < dw; px++) {
   851→   851→        // Device pixel → CSS-pixel sample point.
   852→   852→        const cssPx = minX + (px + 0.5) / dpr;
   853→   853→        const cssPy = minY + (py + 0.5) / dpr;
   854→   854→        const dx = cssPx - cx;
   855→   855→        const dy = cssPy - cy;
   856→   856→        const dist = Math.sqrt(dx * dx + dy * dy);
   857→   857→        if (dist > r + 0.5) continue;
   858→   858→        // Anti-aliased edge.
   859→   859→        const aa = Math.max(0, Math.min(1, r - dist + 0.5));
   860→   860→        if (aa === 0) continue;
   861→   861→
   862→   862→        const S = Math.min(1, dist / r);
   863→   863→        let V: number;
   864→   864→        if (dist < 0.5) {
   865→   865→          V = 1; // center: angle is undefined
   866→   866→        } else {
   867→   867→          // atan2(dx, -dy) = angle from +up axis, increasing clockwise.
   868→   868→          const theta = Math.atan2(dx, -dy);
   869→   869→          V = (1 + Math.cos(theta)) / 2;
   870→   870→        }
   871→   871→        const rgb = hsvToRgb(hue, S * 100, V * 100);
   872→   872→
   873→   873→        const i = (py * dw + px) * 4;
   874→   874→        data[i] = rgb.r;
   875→   875→        data[i + 1] = rgb.g;
   876→   876→        data[i + 2] = rgb.b;
   877→   877→        data[i + 3] = Math.round(aa * 255);
   878→   878→      }
   879→   879→    }
   880→   880→
   881→   881→    // Composite via temp canvas + drawImage (respects dpr transform).
   882→   882→    const temp = document.createElement("canvas");
   883→   883→    temp.width = dw;
   884→   884→    temp.height = dh;
   885→   885→    const tctx = temp.getContext("2d");
   886→   886→    if (!tctx) return;
   887→   887→    tctx.putImageData(img, 0, 0);
   888→   888→    ctx.drawImage(temp, minX, minY);
   889→   889→  }

   891→   891→  // ---- Markers ------------------------------------------------------------
   892→   892→
   893→   893→  private renderMarkers(geom: Geometry): void {
   894→   894→    this.renderHueMarker(geom);
   895→   895→    if (this.options.shape !== "none") {
   896→   896→      this.renderInnerMarker(geom);
   897→   897→    }
   898→   898→  }
   899→   899→
   900→   900→  /**
   901→   901→   * Hue marker — a small disc sitting on the wheel ring at the angular
   902→   902→   * position of the current hue. White outer halo + dark inner ring + a
   903→   903→   * center dot in the pure-hue color guarantees visibility on any hue.
   904→   904→   */
   905→   905→  private renderHueMarker(geom: Geometry): void {
   906→   906→    const { cx, cy, wheelInnerR, wheelOuterR } = geom;
   907→   907→    const { ctx, hsv } = this;
   908→   908→    const midR = (wheelInnerR + wheelOuterR) / 2;
   909→   909→    const angle = -Math.PI / 2 + (hsv.h * Math.PI) / 180;
   910→   910→    const x = cx + midR * Math.cos(angle);
   911→   911→    const y = cy + midR * Math.sin(angle);
   912→   912→
   913→   913→    const thickness = wheelOuterR - wheelInnerR;
   914→   914→    const markerR = clamp(thickness / 2 - 1, 4, 9);
   915→   915→    const ringR = markerR + 1.5;
   916→   916→
   917→   917→    // White halo.
   918→   918→    ctx.beginPath();
   919→   919→    ctx.arc(x, y, ringR, 0, Math.PI * 2);
   920→   920→    ctx.lineWidth = 2.5;
   921→   921→    ctx.strokeStyle = "rgba(255, 255, 255, 0.95)";
   922→   922→    ctx.stroke();
   923→   923→
   924→   924→    // Dark inner ring.
   925→   925→    ctx.beginPath();
   926→   926→    ctx.arc(x, y, ringR, 0, Math.PI * 2);
   927→   927→    ctx.lineWidth = 1;
   928→   928→    ctx.strokeStyle = "rgba(0, 0, 0, 0.7)";
   929→   929→    ctx.stroke();
   930→   930→
   931→   931→    // Pure-hue center dot.
   932→   932→    const rgb = hsvToRgb(hsv.h, 100, 100);
   933→   933→    ctx.beginPath();
   934→   934→    ctx.arc(x, y, Math.max(1, markerR - 1.5), 0, Math.PI * 2);
   935→   935→    ctx.fillStyle = `rgb(${rgb.r}, ${rgb.g}, ${rgb.b})`;
   936→   936→    ctx.fill();
   937→   937→  }
   938→   938→
   939→   939→  /**
   940→   940→   * Inner marker — a small ring at the selected (S, V) position inside the
   941→   941→   * current shape. White halo + dark inner ring + center dot in the actual
   942→   942→   * selected color so it remains perceivable against any background.
   943→   943→   */
   944→   944→  private renderInnerMarker(geom: Geometry): void {
   945→   945→    const pos = this.getInnerMarkerPosition(geom);
   946→   946→    if (!pos) return;
   947→   947→    const { ctx, hsv } = this;
   948→   948→    const rgb = hsvToRgb(hsv.h, hsv.s, hsv.v);
   949→   949→
   950→   950→    const markerR = 7;
   951→   951→    // White halo.
   952→   952→    ctx.beginPath();
   953→   953→    ctx.arc(pos.x, pos.y, markerR, 0, Math.PI * 2);
   954→   954→    ctx.lineWidth = 3;
   955→   955→    ctx.strokeStyle = "rgba(255, 255, 255, 0.95)";
   956→   956→    ctx.stroke();
   957→   957→    // Dark inner ring.
   958→   958→    ctx.beginPath();
   959→   959→    ctx.arc(pos.x, pos.y, markerR, 0, Math.PI * 2);
   960→   960→    ctx.lineWidth = 1.5;
   961→   961→    ctx.strokeStyle = "rgba(0, 0, 0, 0.7)";
   962→   962→    ctx.stroke();
   963→   963→    // Selected color dot.
   964→   964→    ctx.beginPath();
   965→   965→    ctx.arc(pos.x, pos.y, Math.max(1, markerR - 3.5), 0, Math.PI * 2);
   966→   966→    ctx.fillStyle = `rgb(${rgb.r}, ${rgb.g}, ${rgb.b})`;
   967→   967→    ctx.fill();
   968→   968→  }
   969→   969→
   970→   970→  /** Compute the inner marker position from the current HSV state. */
   971→   971→  private getInnerMarkerPosition(
   972→   972→    geom: Geometry,
   973→   973→  ): { x: number; y: number } | null {
   974→   974→    const { shape } = this.options;
   975→   975→    const { cx, cy, innerR } = geom;
   976→   976→    const s = this.hsv.s / 100;
   977→   977→    const v = this.hsv.v / 100;
   978→   978→
   979→   979→    if (shape === "square") {
   980→   980→      const side = innerR * Math.SQRT2;
   981→   981→      const half = side / 2;
   982→   982→      const left = cx - half;
   983→   983→      const top = cy - half;
   984→   984→      return { x: left + s * side, y: top + (1 - v) * side };
   985→   985→    }
   986→   986→
   987→   987→    if (shape === "disk") {
   988→   988→      const r = s * innerR;
   989→   989→      // Use the tracked disk angle so the marker stays where the user put it.
   990→   990→      const theta = this.diskTheta;
   991→   991→      return {
   992→   992→        x: cx + r * Math.sin(theta),
   993→   993→        y: cy - r * Math.cos(theta),
   994→   994→      };
   995→   995→    }
   996→   996→
   997→   997→    if (shape === "triangle") {
   998→   998→      const r = innerR;
   999→   999→      const rotation = this.getShapeRotation();
  1000→  1000→      // Base (unrotated) vertices, then apply rotation.
  1001→  1001→      const A = rotatePoint({ x: cx, y: cy - r }, cx, cy, rotation);
  1002→  1002→      const B = rotatePoint(
  1003→  1003→        { x: cx + (r * Math.sqrt(3)) / 2, y: cy + r / 2 },
  1004→  1004→        cx,
  1005→  1005→        cy,
  1006→  1006→        rotation,
  1007→  1007→      );
  1008→  1008→      const C = rotatePoint(
  1009→  1009→        { x: cx - (r * Math.sqrt(3)) / 2, y: cy + r / 2 },
  1010→  1010→        cx,
  1011→  1011→        cy,
  1012→  1012→        rotation,
  1013→  1013→      );
  1014→  1014→      const alpha = v * (1 - s);
  1015→  1015→      const beta = s * v;
  1016→  1016→      const gamma = 1 - v;
  1017→  1017→      return {
  1018→  1018→        x: alpha * A.x + beta * B.x + gamma * C.x,
  1019→  1019→        y: alpha * A.y + beta * B.y + gamma * C.y,
  1020→  1020→      };
  1021→  1021→    }
  1022→  1022→
  1023→  1023→    return null;
  1024→  1024→  }

  1030→  1030→  private getCanvasPoint(e: PointerEvent): { x: number; y: number } {
  1031→  1031→    const rect = this.canvas.getBoundingClientRect();
  1032→  1032→    // Use rect.width to scale in case the canvas has been resized via CSS.
  1033→  1033→    const geom = this.getGeometry();
  1034→  1034→    const scaleX = geom.total / (rect.width || geom.total);
  1035→  1035→    const scaleY = geom.total / (rect.height || geom.total);
  1036→  1036→    return {
  1037→  1037→      x: (e.clientX - rect.left) * scaleX,
  1038→  1038→      y: (e.clientY - rect.top) * scaleY,
  1039→  1039→    };
  1040→  1040→  }
  1041→  1041→
  1042→  1042→  private onPointerDown(e: PointerEvent): void {
  1043→  1043→    if (this._destroyed) return;
  1044→  1044→    const { x, y } = this.getCanvasPoint(e);
  1045→  1045→    const geom = this.getGeometry();
  1046→  1046→    const dx = x - geom.cx;
  1047→  1047→    const dy = y - geom.cy;
  1048→  1048→    const dist = Math.sqrt(dx * dx + dy * dy);
  1049→  1049→
  1050→  1050→    const onWheel =
  1051→  1051→      dist >= geom.wheelInnerR - 1 && dist <= geom.wheelOuterR + 1;
  1052→  1052→    const inInner = dist <= geom.innerR;
  1053→  1053→
  1054→  1054→    if (onWheel) {
  1055→  1055→      this.pointerState = { type: "wheel", pointerId: e.pointerId };
  1056→  1056→      try {
  1057→  1057→        this.canvas.setPointerCapture(e.pointerId);
  1058→  1058→      } catch {
  1059→  1059→        /* ignore */
  1060→  1060→      }
  1061→  1061→      this.updateHueFromPoint(x, y, geom);
  1062→  1062→      e.preventDefault();
  1063→  1063→    } else if (inInner && this.options.shape !== "none") {
  1064→  1064→      this.pointerState = { type: "inner", pointerId: e.pointerId };
  1065→  1065→      try {
  1066→  1066→        this.canvas.setPointerCapture(e.pointerId);
  1067→  1067→      } catch {
  1068→  1068→        /* ignore */
  1069→  1069→      }
  1070→  1070→      this.updateInnerFromPoint(x, y, geom);
  1071→  1071→      e.preventDefault();
  1072→  1072→    }
  1073→  1073→  }
  1074→  1074→
  1075→  1075→  private onPointerMove(e: PointerEvent): void {
  1076→  1076→    if (!this.pointerState) return;
  1077→  1077→    const { x, y } = this.getCanvasPoint(e);
  1078→  1078→    const geom = this.getGeometry();
  1079→  1079→    if (this.pointerState.type === "wheel") {
  1080→  1080→      this.updateHueFromPoint(x, y, geom);
  1081→  1081→    } else {
  1082→  1082→      this.updateInnerFromPoint(x, y, geom);
  1083→  1083→    }
  1084→  1084→    e.preventDefault();
  1085→  1085→  }
  1086→  1086→
  1087→  1087→  private onPointerUp(e: PointerEvent): void {
  1088→  1088→    if (!this.pointerState) return;
  1089→  1089→    const id = this.pointerState.pointerId;
  1090→  1090→    try {
  1091→  1091→      this.canvas.releasePointerCapture(id);
  1092→  1092→    } catch {
  1093→  1093→      /* ignore */
  1094→  1094→    }
  1095→  1095→    this.pointerState = null;
  1096→  1096→    this.emit("change", this.getColor());
  1097→  1097→  }
  1098→  1098→
  1099→  1099→  private updateHueFromPoint(
  1100→  1100→    x: number,
  1101→  1101→    y: number,
  1102→  1102→    geom: Geometry,
  1103→  1103→  ): void {
  1104→  1104→    const dx = x - geom.cx;
  1105→  1105→    const dy = y - geom.cy;
  1106→  1106→    // Angle from "up" vector, increasing clockwise (matches wheel layout).
  1107→  1107→    let theta = (Math.atan2(dx, -dy) * 180) / Math.PI;
  1108→  1108→    if (theta < 0) theta += 360;
  1109→  1109→    this.hsv = { ...this.hsv, h: clampHue(theta) };
  1110→  1110→    this.options.hue = this.hsv.h;
  1111→  1111→    this.invalidateInnerCache();
  1112→  1112→    this.updateAria();
  1113→  1113→    this.scheduleRender();
  1114→  1114→    this.emit("input", this.getColor());
  1115→  1115→  }
  1116→  1116→
  1117→  1117→  private updateInnerFromPoint(
  1118→  1118→    x: number,
  1119→  1119→    y: number,
  1120→  1120→    geom: Geometry,
  1121→  1121→  ): void {
  1122→  1122→    const { shape } = this.options;
  1123→  1123→    const { cx, cy, innerR } = geom;
  1124→  1124→    let newHsv: HSV | null = null;
  1125→  1125→
  1126→  1126→    if (shape === "square") {
  1127→  1127→      const side = innerR * Math.SQRT2;
  1128→  1128→      const half = side / 2;
  1129→  1129→      const left = cx - half;
  1130→  1130→      const top = cy - half;
  1131→  1131→      const s = clamp((x - left) / side, 0, 1);
  1132→  1132→      const v = clamp(1 - (y - top) / side, 0, 1);
  1133→  1133→      newHsv = { ...this.hsv, s: s * 100, v: v * 100 };
  1134→  1134→    } else if (shape === "disk") {
  1135→  1135→      const dx = x - cx;
  1136→  1136→      const dy = y - cy;
  1137→  1137→      const dist = Math.sqrt(dx * dx + dy * dy);
  1138→  1138→      const s = clamp(dist / innerR, 0, 1);
  1139→  1139→      let v: number;
  1140→  1140→      if (dist < 0.5) {
  1141→  1141→        v = this.hsv.v / 100; // keep current V at the very center
  1142→  1142→      } else {
  1143→  1143→        const theta = Math.atan2(dx, -dy);
  1144→  1144→        v = (1 + Math.cos(theta)) / 2;
  1145→  1145→        this.diskTheta = theta;
  1146→  1146→      }
  1147→  1147→      newHsv = { ...this.hsv, s: s * 100, v: v * 100 };
  1148→  1148→    } else if (shape === "triangle") {
  1149→  1149→      const r = innerR;
  1150→  1150→      // When rotateShape is enabled, the triangle is drawn rotated. Un-rotate
  1151→  1151→      // the pointer point into the triangle's local coordinate space before
  1152→  1152→      // computing barycentric coordinates.
  1153→  1153→      const rotation = this.getShapeRotation();
  1154→  1154→      const local = rotatePoint({ x, y }, cx, cy, -rotation);
  1155→  1155→      const lx = local.x;
  1156→  1156→      const ly = local.y;
  1157→  1157→
  1158→  1158→      const Ax = cx;
  1159→  1159→      const Ay = cy - r;
  1160→  1160→      const Bx = cx + (r * Math.sqrt(3)) / 2;
  1161→  1161→      const By = cy + r / 2;
  1162→  1162→      const Cx = cx - (r * Math.sqrt(3)) / 2;
  1163→  1163→      const Cy = cy + r / 2;
  1164→  1164→
  1165→  1165→      const det = (By - Cy) * (Ax - Cx) + (Cx - Bx) * (Ay - Cy);
  1166→  1166→      if (Math.abs(det) < 1e-9) return;
  1167→  1167→      const invDet = 1 / det;
  1168→  1168→      const alpha = ((By - Cy) * (lx - Cx) + (Cx - Bx) * (ly - Cy)) * invDet;
  1169→  1169→      const beta = ((Cy - Ay) * (lx - Cx) + (Ax - Cx) * (ly - Cy)) * invDet;
  1170→  1170→      const gamma = 1 - alpha - beta;
  1171→  1171→
  1172→  1172→      // Clamp to the triangle (project to nearest edge if outside).
  1173→  1173→      let a = alpha < 0 ? 0 : alpha;
  1174→  1174→      let b = beta < 0 ? 0 : beta;
  1175→  1175→      let g = gamma < 0 ? 0 : gamma;
  1176→  1176→      const sum = a + b + g;
  1177→  1177→      if (sum <= 0) return;
  1178→  1178→      a /= sum;
  1179→  1179→      b /= sum;
  1180→  1180→      g /= sum;
  1181→  1181→
  1182→  1182→      const V = a + b;
  1183→  1183→      const S = V > 0 ? b / V : 0;
  1184→  1184→      newHsv = { ...this.hsv, s: S * 100, v: V * 100 };
  1185→  1185→    }
  1186→  1186→
  1187→  1187→    if (!newHsv) return;
  1188→  1188→    this.hsv = newHsv;
  1189→  1189→    this.scheduleRender();
  1190→  1190→    this.emit("input", this.getColor());
  1191→  1191→  }

  1197→  1197→  private onKeyDown(e: KeyboardEvent): void {
  1198→  1198→    if (this._destroyed) return;
  1199→  1199→    const step = e.shiftKey ? 10 : 1;
  1200→  1200→    const hasInner = this.options.shape !== "none";
  1201→  1201→    let handled = false;
  1202→  1202→
  1203→  1203→    switch (e.key) {
  1204→  1204→      case "ArrowLeft":
  1205→  1205→        if ((e.ctrlKey || e.metaKey) && hasInner) {
  1206→  1206→          this.adjustSaturation(-step);
  1207→  1207→        } else {
  1208→  1208→          this.adjustHue(-step);
  1209→  1209→        }
  1210→  1210→        handled = true;
  1211→  1211→        break;
  1212→  1212→      case "ArrowRight":
  1213→  1213→        if ((e.ctrlKey || e.metaKey) && hasInner) {
  1214→  1214→          this.adjustSaturation(step);
  1215→  1215→        } else {
  1216→  1216→          this.adjustHue(step);
  1217→  1217→        }
  1218→  1218→        handled = true;
  1219→  1219→        break;
  1220→  1220→      case "ArrowUp":
  1221→  1221→        if (hasInner) {
  1222→  1222→          this.adjustValue(step);
  1223→  1223→        } else {
  1224→  1224→          this.adjustHue(step);
  1225→  1225→        }
  1226→  1226→        handled = true;
  1227→  1227→        break;
  1228→  1228→      case "ArrowDown":
  1229→  1229→        if (hasInner) {
  1230→  1230→          this.adjustValue(-step);
  1231→  1231→        } else {
  1232→  1232→          this.adjustHue(-step);
  1233→  1233→        }
  1234→  1234→        handled = true;
  1235→  1235→        break;
  1236→  1236→      case "Home":
  1237→  1237→        this.adjustHue(-360);
  1238→  1238→        handled = true;
  1239→  1239→        break;
  1240→  1240→      case "End":
  1241→  1241→        this.adjustHue(360);
  1242→  1242→        handled = true;
  1243→  1243→        break;
  1244→  1244→      default:
  1245→  1245→        break;
  1246→  1246→    }
  1247→  1247→
  1248→  1248→    if (handled) {
  1249→  1249→      e.preventDefault();
  1250→  1250→      e.stopPropagation();
  1251→  1251→    }
  1252→  1252→  }
  1253→  1253→
  1254→  1254→  private adjustHue(delta: number): void {
  1255→  1255→    this.hsv = { ...this.hsv, h: clampHue(this.hsv.h + delta) };
  1256→  1256→    this.options.hue = this.hsv.h;
  1257→  1257→    this.invalidateInnerCache();
  1258→  1258→    this.updateAria();
  1259→  1259→    this.scheduleRender();
  1260→  1260→    const out = this.getColor();
  1261→  1261→    this.emit("input", out);
  1262→  1262→    this.emit("change", out);
  1263→  1263→  }
  1264→  1264→
  1265→  1265→  private adjustSaturation(delta: number): void {
  1266→  1266→    this.hsv = { ...this.hsv, s: clamp(this.hsv.s + delta, 0, 100) };
  1267→  1267→    this.scheduleRender();
  1268→  1268→    const out = this.getColor();
  1269→  1269→    this.emit("input", out);
  1270→  1270→    this.emit("change", out);
  1271→  1271→  }
  1272→  1272→
  1273→  1273→  private adjustValue(delta: number): void {
  1274→  1274→    this.hsv = { ...this.hsv, v: clamp(this.hsv.v + delta, 0, 100) };
  1275→  1275→    this.scheduleRender();
  1276→  1276→    const out = this.getColor();
  1277→  1277→    this.emit("input", out);
  1278→  1278→    this.emit("change", out);
  1279→  1279→  }

  1297→  1297→  private emit(event: string, color: ColorOutput): void {
  1298→  1298→    const set = this.listeners.get(event);
  1299→  1299→    if (!set) return;
  1300→  1300→    // Clone to a plain array so a handler unsubscribing during iteration
  1301→  1301→    // doesn't skip subsequent handlers.

  1303→  1303→    for (const h of handlers) {
  1304→  1304→      try {
  1305→  1305→        h(color);
  1306→  1306→      } catch (err) {
  1307→  1307→        // Never let a listener error take down the picker.
  1308→  1308→        console.error("ColorWheelPicker listener error:", err);
  1309→  1309→      }
  1310→  1310→    }
