     1→/**
     2→ * TableEditor — a spreadsheet-style table component (vanilla TypeScript).
     3→ *
     4→ * Features:
     5→ *   - Cell selection (single + range via drag)
     6→ *   - Inline cell editing (double-click or Enter)
     7→ *   - Column resize (drag header right edge)
     8→ *   - Column reorder (drag header)
     9→ *   - Sort (click header: asc → desc → none)
    10→ *   - Frozen header row + first column
    11→ *   - Cell types: text, number, checkbox, select
    12→ *   - Keyboard navigation (Arrows, Tab, Enter, Escape, Delete)
    13→ *   - Row numbers + column headers
    14→ *   - Copy/paste (Ctrl+C / Ctrl+V)
    15→ *   - Add/remove rows and columns
    16→ *   - Events for all changes
    17→ */
    18→
    19→// ---- Public types ----
    20→
    21→export type CellType = "text" | "number" | "checkbox" | "select";
    22→
    23→export interface TableColumn {
    24→  id: string;
    25→  label: string;
    26→  type?: CellType;
    27→  width?: number;
    28→  align?: "left" | "center" | "right";
    29→  sortable?: boolean;
    30→  editable?: boolean;
    31→  options?: { label: string; value: unknown }[];
    32→}
    33→
    34→export interface TableRow {
    35→  id: string;
    36→  [cellId: string]: unknown;
    37→}
    38→
    39→export interface TableEditorOptions {
    40→  columns?: TableColumn[];
    41→  rows?: TableRow[];
    42→  frozenHeader?: boolean;
    43→  frozenFirstCol?: boolean;
    44→  rowNumbers?: boolean;
    45→  showToolbar?: boolean;
    46→  selectable?: boolean;
    47→  editable?: boolean;
    48→  resizable?: boolean;
    49→  reorderable?: boolean;
    50→  sortable?: boolean;
    51→  striped?: boolean;
    52→  minColWidth?: number;
    53→}
    54→
    55→export interface TableEditorEventPayload {
    56→  type?: string;
    57→  rowId?: string;
    58→  colId?: string;
    59→  cellId?: string;
    60→  value?: unknown;
    61→  oldValue?: unknown;
    62→  selectedCells?: { row: number; col: number }[];
    63→  colId2?: string;
    64→  fromIndex?: number;
    65→  toIndex?: number;
    66→  col?: string;
    67→  direction?: "asc" | "desc" | null;
    68→  reason?: string;
    69→  source?: "pointer" | "keyboard" | "programmatic";
    70→}

   327→// ---- Component ----
   328→
   329→export class TableEditor {
   330→  private container: HTMLElement;
   331→  private root!: HTMLDivElement;
   332→  private scrollEl!: HTMLDivElement;
   333→  private gridEl!: HTMLDivElement;
   334→  private opts: Required<TableEditorOptions>;
   335→  private columns: TableColumn[];
   336→  private rows: TableRow[];
   337→  private selectedStart: { row: number; col: number } | null = null;
   338→  private selectedEnd: { row: number; col: number } | null = null;
   339→  private editingCell: { row: number; col: number } | null = null;
   340→  private sortState: { col: string; direction: "asc" | "desc" } | null = null;
   341→  private colWidths: Map<string, number> = new Map();
   342→  private listeners = new Map<string, Set<TableEditorHandler>>();
   343→  private _destroyed = false;
   344→  private cellEls: HTMLElement[][] = [];
   345→  private rowHeaderEls: HTMLElement[] = [];
   346→  private colHeaderEls: HTMLElement[] = [];
   347→
   348→  constructor(container: HTMLElement, options: TableEditorOptions = {}) {
   349→    this.container = container;
   350→    injectStyles();
   351→    this.opts = {
   352→      columns: options.columns ?? [],
   353→      rows: options.rows ?? [],
   354→      frozenHeader: options.frozenHeader ?? true,
   355→      frozenFirstCol: options.frozenFirstCol ?? false,
   356→      rowNumbers: options.rowNumbers ?? true,
   357→      showToolbar: options.showToolbar ?? true,
   358→      selectable: options.selectable ?? true,
   359→      editable: options.editable ?? true,
   360→      resizable: options.resizable ?? true,
   361→      reorderable: options.reorderable ?? true,
   362→      sortable: options.sortable ?? true,
   363→      striped: options.striped ?? true,
   364→      minColWidth: options.minColWidth ?? 60,
   365→    };
   366→    this.columns = (this.opts.columns as TableColumn[]).map(c => ({ ...c }));
   367→    this.rows = (this.opts.rows as TableRow[]).map(r => ({ ...r }));
   368→    // Initialize column widths
   369→    for (const col of this.columns) {
   370→      this.colWidths.set(col.id, col.width ?? DEFAULT_COL_WIDTH);
   371→    }
   372→    this.render();
   373→  }
   374→
   375→  // ---- Rendering ----
   376→
   377→  private render(): void {
   378→    if (this._destroyed) return;
   379→    const root = document.createElement("div");
   380→    root.className = "te-root";
   381→
   382→    if (this.opts.showToolbar) {
   383→      root.appendChild(this.renderToolbar());
   384→    }
   385→
   386→    const scroll = document.createElement("div");
   387→    scroll.className = "te-scroll";
   388→    scroll.tabIndex = 0;
   389→
   390→    const grid = document.createElement("div");
   391→    grid.className = "te-grid";
   392→
   393→    // Build column template string for the grid
   394→    const templateCols = (this.opts.rowNumbers ? ["40px"] : []).concat(
   395→      this.columns.map(c => `${this.colWidths.get(c.id) ?? DEFAULT_COL_WIDTH}px`)
   396→    );
   397→    grid.style.gridTemplateColumns = templateCols.join(" ");
   398→
   399→    this.cellEls = [];
   400→    this.rowHeaderEls = [];
   401→    this.colHeaderEls = [];
   402→
   403→    // Header row
   404→    const headerCells: HTMLElement[] = [];
   405→    if (this.opts.rowNumbers) {
   406→      const corner = document.createElement("div");
   407→      corner.className = "te-corner";
   408→      corner.textContent = "#";
   409→      headerCells.push(corner);
   410→    }
   411→    for (let c = 0; c < this.columns.length; c++) {
   412→      const col = this.columns[c];
   413→      const header = this.renderColHeader(col, c);
   414→      headerCells.push(header);
   415→      this.colHeaderEls.push(header);
   416→    }
   417→    headerCells.forEach(el => grid.appendChild(el));
   418→
   419→    // Body rows
   420→    for (let r = 0; r < this.rows.length; r++) {
   421→      const rowCells: HTMLElement[] = [];
   422→      if (this.opts.rowNumbers) {
   423→        const rh = document.createElement("div");
   424→        rh.className = "te-row-header";
   425→        if (this.opts.frozenFirstCol) rh.classList.add("te-frozen");
   426→        rh.textContent = String(r + 1);
   427→        rh.setAttribute("data-row", String(r));
   428→        rh.addEventListener("click", () => this.selectRow(r, "pointer"));
   429→        rowCells.push(rh);
   430→        this.rowHeaderEls.push(rh);
   431→      }
   432→      const cells: HTMLElement[] = [];
   433→      for (let c = 0; c < this.columns.length; c++) {
   434→        const col = this.columns[c];
   435→        const cell = this.renderCell(r, c, col);
   436→        rowCells.push(cell);
   437→        cells.push(cell);
   438→      }
   439→      rowCells.forEach(el => grid.appendChild(el));
   440→      this.cellEls.push(cells);
   441→    }
   442→
   443→    if (this.rows.length === 0) {
   444→      const empty = document.createElement("div");
   445→      empty.className = "te-empty";
   446→      empty.style.gridColumn = `1 / -1`;
   447→      empty.textContent = "No rows — use the toolbar to add one.";
   448→      grid.appendChild(empty);
   449→    }
   450→
   451→    scroll.appendChild(grid);
   452→    root.appendChild(scroll);
   453→    this.container.innerHTML = "";
   454→    this.container.appendChild(root);
   455→    this.root = root;
   456→    this.scrollEl = scroll;
   457→    this.gridEl = grid;
   458→
   459→    // Apply sort indicator
   460→    this.updateSortIndicators();
   461→    // Apply selection highlight
   462→    this.updateSelectionDom();
   463→
   464→    // Keyboard
   465→    scroll.addEventListener("keydown", (e) => this.onKeyDown(e));
   466→  }

   506→
   507→  private renderColHeader(col: TableColumn, idx: number): HTMLElement {
   508→    const el = document.createElement("div");
   509→    el.className = "te-col-header";
   510→    if (this.opts.frozenHeader) el.style.position = "sticky";
   511→    if (this.opts.frozenFirstCol && idx === 0) el.classList.add("te-frozen");
   512→    el.setAttribute("data-col", col.id);
   513→    el.setAttribute("data-col-index", String(idx));
   514→    el.style.width = `${this.colWidths.get(col.id) ?? DEFAULT_COL_WIDTH}px`;
   515→    el.style.position = "sticky";
   516→    el.style.top = "0";
   517→
   518→    // Sort indicator
   519→    if (this.sortState?.col === col.id) {
   520→      el.classList.add(this.sortState.direction === "asc" ? "te-sort-asc" : "te-sort-desc");
   521→      const sortIcon = document.createElement("span");
   522→      sortIcon.className = "te-sort-icon";
   523→      sortIcon.innerHTML = this.sortState.direction === "asc" ? ICON_SORT_ASC : ICON_SORT_DESC;
   524→      el.appendChild(sortIcon);
   525→    }
   526→
   527→    // Reorder grip
   528→    if (this.opts.reorderable) {
   529→      const grip = document.createElement("span");
   530→      grip.className = "te-reorder-grip";
   531→      grip.innerHTML = ICON_GRIP;
   532→      grip.setAttribute("aria-hidden", "true");
   533→      grip.addEventListener("pointerdown", (e) => this.onReorderStart(e, col.id));
   534→      el.appendChild(grip);
   535→    }
   536→
   537→    const label = document.createElement("span");
   538→    label.className = "te-col-label";
   539→    label.textContent = col.label;
   540→    el.appendChild(label);
   541→
   542→    // Sort on click (not on grip or resize)
   543→    if (this.opts.sortable) {
   544→      el.addEventListener("click", (e) => {
   545→        if ((e.target as HTMLElement).closest(".te-resize-handle") || (e.target as HTMLElement).closest(".te-reorder-grip")) return;
   546→        this.toggleSort(col.id);
   547→      });
   548→    }
   549→
   550→    // Resize handle
   551→    if (this.opts.resizable) {
   552→      const handle = document.createElement("div");
   553→      handle.className = "te-resize-handle";
   554→      handle.addEventListener("pointerdown", (e) => this.onResizeStart(e, col.id));
   555→      el.appendChild(handle);
   556→    }
   557→
   558→    return el;
   559→  }

   561→  private renderCell(rowIdx: number, colIdx: number, col: TableColumn): HTMLElement {
   562→    const el = document.createElement("div");
   563→    el.className = "te-cell";
   564→    if (this.opts.striped && rowIdx % 2 === 1) el.classList.add("te-striped");
   565→    if (this.opts.frozenFirstCol && colIdx === 0) el.classList.add("te-frozen");
   566→    if (col.align === "center") el.classList.add("te-align-center");
   567→    if (col.align === "right") el.classList.add("te-align-right");
   568→    el.setAttribute("data-row", String(rowIdx));
   569→    el.setAttribute("data-col", String(colIdx));
   570→    el.setAttribute("data-col-id", col.id);
   571→
   572→    const row = this.rows[rowIdx];
   573→    const value = row?.[col.id];
   574→    const editable = col.editable !== false && this.opts.editable;
   575→
   576→    if (col.type === "checkbox") {
   577→      const cb = document.createElement("input");
   578→      cb.type = "checkbox";
   579→      cb.checked = !!value;
   580→      cb.className = "te-cell-checkbox";
   581→      if (editable) {
   582→        cb.addEventListener("change", () => {
   583→          this.setCellValue(rowIdx, colIdx, cb.checked, "pointer");
   584→        });
   585→      } else {
   586→        cb.disabled = true;
   587→      }
   588→      el.appendChild(cb);
   589→    } else {
   590→      // Render text representation
   591→      if (value !== undefined && value !== null) {
   592→        el.textContent = String(value);
   593→      }
   594→    }
   595→
   596→    // Cell interactions
   597→    if (this.opts.selectable) {
   598→      el.addEventListener("pointerdown", (e) => this.onCellPointerDown(e, rowIdx, colIdx));
   599→      el.addEventListener("dblclick", () => {
   600→        if (editable && col.type !== "checkbox") this.startEditing(rowIdx, colIdx);
   601→      });
   602→    }
   603→
   604→    return el;
   605→  }

   675→  getSelectedCells(): { row: number; col: number }[] {
   676→    if (!this.selectedStart || !this.selectedEnd) return [];
   677→    const r1 = Math.min(this.selectedStart.row, this.selectedEnd.row);
   678→    const r2 = Math.max(this.selectedStart.row, this.selectedEnd.row);
   679→    const c1 = Math.min(this.selectedStart.col, this.selectedEnd.col);
   680→    const c2 = Math.max(this.selectedStart.col, this.selectedEnd.col);
   681→    const cells: { row: number; col: number }[] = [];
   682→    for (let r = r1; r <= r2; r++) {
   683→      for (let c = c1; c <= c2; c++) {
   684→        cells.push({ row: r, col: c });
   685→      }
   686→    }
   687→    return cells;
   688→  }

   713→  startEditing(row: number, col: number): void {
   714→    if (this.editingCell) this.cancelEditing();
   715→    if (row < 0 || row >= this.rows.length || col < 0 || col >= this.columns.length) return;
   716→    const colDef = this.columns[col];
   717→    if (colDef.editable === false) return;
   718→    this.editingCell = { row, col };
   719→    const cellEl = this.cellEls[row]?.[col];
   720→    if (!cellEl) return;
   721→    cellEl.classList.add("te-editing");
   722→    cellEl.innerHTML = "";
   723→    const oldValue = this.rows[row]?.[colDef.id];
   724→
   725→    if (colDef.type === "select") {
   726→      const sel = document.createElement("select");
   727→      sel.className = "te-cell-select";
   728→      for (const opt of colDef.options ?? []) {
   729→        const o = document.createElement("option");
   730→        o.value = String(opt.value);
   731→        o.textContent = opt.label;
   732→        if (opt.value === oldValue) o.selected = true;
   733→        sel.appendChild(o);
   734→      }
   735→      sel.addEventListener("change", () => {
   736→        this.setCellValue(row, col, sel.value, "pointer");
   737→        this.cancelEditing();
   738→      });
   739→      sel.addEventListener("blur", () => this.cancelEditing());
   740→      sel.addEventListener("keydown", (e) => {
   741→        if (e.key === "Escape") { e.preventDefault(); this.cancelEditing(); }
   742→        if (e.key === "Enter") { e.preventDefault(); this.cancelEditing(); }
   743→      });
   744→      cellEl.appendChild(sel);
   745→      sel.focus();
   746→    } else {
   747→      const input = document.createElement("input");
   748→      input.type = colDef.type === "number" ? "number" : "text";
   749→      input.className = "te-cell-input";
   750→      input.value = oldValue !== undefined && oldValue !== null ? String(oldValue) : "";
   751→      input.addEventListener("blur", () => this.cancelEditing());
   752→      input.addEventListener("keydown", (e) => {
   753→        if (e.key === "Enter") {
   754→          e.preventDefault();
   755→          const newVal = colDef.type === "number" ? Number(input.value) : input.value;
   756→          this.setCellValue(row, col, newVal, "keyboard");
   757→          this.cancelEditing();
   758→          // Move down
   759→          if (row + 1 < this.rows.length) this.selectCell(row + 1, col);
   760→        }
   761→        if (e.key === "Escape") { e.preventDefault(); this.cancelEditing(); }
   762→        if (e.key === "Tab") {
   763→          e.preventDefault();
   764→          const newVal = colDef.type === "number" ? Number(input.value) : input.value;
   765→          this.setCellValue(row, col, newVal, "keyboard");
   766→          this.cancelEditing();
   767→          if (col + 1 < this.columns.length) this.selectCell(row, col + 1);
   768→        }
   769→      });
   770→      cellEl.appendChild(input);
   771→      input.focus();
   772→      input.select();
   773→    }
   774→  }
   775→
   776→  cancelEditing(): void {
   777→    if (!this.editingCell) return;
   778→    const { row, col } = this.editingCell;
   779→    this.editingCell = null;
   780→    const colDef = this.columns[col];
   781→    const cellEl = this.cellEls[row]?.[col];
   782→    if (cellEl && colDef) {
   783→      cellEl.classList.remove("te-editing");
   784→      // Re-render cell content
   785→      const value = this.rows[row]?.[colDef.id];
   786→      cellEl.innerHTML = "";
   787→      if (colDef.type === "checkbox") {
   788→        const cb = document.createElement("input");
   789→        cb.type = "checkbox";
   790→        cb.checked = !!value;
   791→        cb.className = "te-cell-checkbox";
   792→        if (colDef.editable !== false && this.opts.editable) {
   793→          cb.addEventListener("change", () => this.setCellValue(row, col, cb.checked, "pointer"));
   794→        } else { cb.disabled = true; }
   795→        cellEl.appendChild(cb);
   796→      } else {
   797→        if (value !== undefined && value !== null) cellEl.textContent = String(value);
   798→      }
   799→    }
   800→  }

   802→  setCellValue(row: number, col: number, value: unknown, source: "pointer" | "keyboard" | "programmatic" = "programmatic"): void {
   803→    if (row < 0 || row >= this.rows.length || col < 0 || col >= this.columns.length) return;
   804→    const colDef = this.columns[col];
   805→    const oldValue = this.rows[row][colDef.id];
   806→    this.rows[row][colDef.id] = value;
   807→    // Update DOM if not editing
   808→    if (!this.editingCell || this.editingCell.row !== row || this.editingCell.col !== col) {
   809→      const cellEl = this.cellEls[row]?.[col];
   810→      if (cellEl) {
   811→        if (colDef.type === "checkbox") {
   812→          const cb = cellEl.querySelector("input") as HTMLInputElement;
   813→          if (cb) cb.checked = !!value;
   814→        } else {
   815→          cellEl.textContent = value !== undefined && value !== null ? String(value) : "";
   816→        }
   817→      }
   818→    }
   819→    this.emit("cell_change", { rowId: this.rows[row].id, colId: colDef.id, value, oldValue, source });
   820→  }

   822→  // ---- Keyboard ----
   823→
   824→  private onKeyDown(e: KeyboardEvent): void {
   825→    if (this.editingCell) return; // Let the input handle keys
   826→    if (!this.selectedStart) return;
   827→    let { row, col } = this.selectedStart;
   828→    let handled = true;
   829→
   830→    switch (e.key) {
   831→      case "ArrowUp": row = Math.max(0, row - 1); break;
   832→      case "ArrowDown": row = Math.min(this.rows.length - 1, row + 1); break;
   833→      case "ArrowLeft": col = Math.max(0, col - 1); break;
   834→      case "ArrowRight": col = Math.min(this.columns.length - 1, col + 1); break;
   835→      case "Tab":
   836→        col = e.shiftKey ? Math.max(0, col - 1) : Math.min(this.columns.length - 1, col + 1);
   837→        break;
   838→      case "Enter":
   839→        if (this.opts.editable) this.startEditing(row, col);
   840→        break;
   841→      case "F2":
   842→        if (this.opts.editable) this.startEditing(row, col);
   843→        break;
   844→      case "Delete":
   845→      case "Backspace":
   846→        // Clear selected cells
   847→        const cells = this.getSelectedCells();
   848→        for (const c of cells) {
   849→          this.setCellValue(c.row, c.col, "", "keyboard");
   850→        }
   851→        break;
   852→      case "Escape":
   853→        this.clearSelection();
   854→        break;
   855→      default:
   856→        // If it's a printable character, start editing
   857→        if (e.key.length === 1 && !e.ctrlKey && !e.metaKey && this.opts.editable) {
   858→          this.startEditing(row, col);
   859→          handled = false; // Let the input receive the key
   860→        } else {
   861→          handled = false;
   862→        }
   863→    }
   864→    if (handled) {
   865→      e.preventDefault();
   866→      if (e.key !== "Enter" && e.key !== "Delete" && e.key !== "Backspace" && e.key !== "Escape" && e.key !== "F2") {
   867→        this.selectCell(row, col);
   868→      }
   869→    }
   870→  }

   950→  toggleSort(colId: string): void {
   951→    if (this.sortState?.col === colId) {
   952→      if (this.sortState.direction === "asc") {
   953→        this.sortState = { col: colId, direction: "desc" };
   954→      } else {
   955→        this.sortState = null; // Cycle: asc → desc → none
   956→      }
   957→    } else {
   958→      this.sortState = { col: colId, direction: "asc" };
   959→    }
   960→    this.applySort();
   961→    this.render();
   962→    this.emit("sort_change", { col: colId, direction: this.sortState?.direction ?? null, source: "pointer" });
   963→  }
   964→
   965→  private applySort(): void {
   966→    if (!this.sortState) return;
   967→    const col = this.columns.find(c => c.id === this.sortState!.col);
   968→    if (!col) return;
   969→    const dir = this.sortState.direction === "asc" ? 1 : -1;
   970→    this.rows.sort((a, b) => {
   971→      const av = a[col.id];
   972→      const bv = b[col.id];
   973→      if (av === bv) return 0;
   974→      if (av === undefined || av === null) return 1;
   975→      if (bv === undefined || bv === null) return -1;
   976→      if (typeof av === "number" && typeof bv === "number") return (av - bv) * dir;
   977→      return String(av).localeCompare(String(bv)) * dir;
   978→    });
   979→  }

  1038→    this.columns.splice(idx, 1);
