     1→/**
     2→ * Calendar — a full-featured date picker calendar (vanilla TypeScript).
     3→ *
     4→ * DOM-based (no canvas, no external date library). Renders a month grid with
     5→ * weekday headers, optional week numbers, prev/next navigation, optional
     6→ * "Today" button, and three selection modes: single, range, multi.
     7→ *
     8→ * Public API:
     9→ *   new Calendar(container, options?)
    10→ *   getSelectedDate(): Date | null
    11→ *   getSelectedRange(): { start: Date | null; end: Date | null }
    12→ *   getSelectedDates(): Date[]
    13→ *   setSelectedDate(date | null): void
    14→ *   setSelectedRange(start, end): void
    15→ *   setSelectedDates(dates): void
    16→ *   setMode(mode): void
    17→ *   setMinDate(date): void
    18→ *   setMaxDate(date): void
    19→ *   setFirstDayOfWeek(day): void
    20→ *   setShowWeekNumbers(on): void
    21→ *   setCellSize(n): void
    22→ *   today(): void
    23→ *   on(event: "select" | "change" | "navigate", handler): void
    24→ *   off(event, handler): void
    25→ *   destroy(): void
    26→ *
    27→ * Framework-independent. DOM-based. No React. No external dependencies.
    28→ * Only accent color used is the brand lemon (#D9FF66) — no indigo/blue.
    29→ */

   324→// ---- Date helpers (pure, no side effects) ----
   325→
   326→/** Strip time components — returns a new Date at local midnight. */
   327→function atMidnight(d: Date): Date {
   328→  return new Date(d.getFullYear(), d.getMonth(), d.getDate());
   329→}
   330→
   331→/** Returns 0 if same day, -1 if a < b, 1 if a > b (date-only comparison). */
   332→function compareDate(a: Date, b: Date): number {
   333→  const aT = atMidnight(a).getTime();
   334→  const bT = atMidnight(b).getTime();
   335→  if (aT === bT) return 0;
   336→  return aT < bT ? -1 : 1;
   337→}
   338→
   339→function isSameDay(a: Date | null, b: Date | null): boolean {
   340→  if (!a || !b) return false;
   341→  return compareDate(a, b) === 0;
   342→}
   343→
   344→/** Clone a Date (or null). */
   345→function cloneDate(d: Date | null): Date | null {
   346→  return d ? new Date(d.getTime()) : null;
   347→}
   348→
   349→/**
   350→ * ISO week number (Mon = first day, week 1 contains the year's first Thursday).
   351→ * Matches the week-numbering used by most desktop calendar apps.
   352→ */
   353→function isoWeek(date: Date): number {
   354→  const d = new Date(Date.UTC(date.getFullYear(), date.getMonth(), date.getDate()));
   355→  const dayNum = d.getUTCDay() || 7;
   356→  d.setUTCDate(d.getUTCDate() + 4 - dayNum);
   357→  const yearStart = new Date(Date.UTC(d.getUTCFullYear(), 0, 1));
   358→  return Math.ceil(((d.getTime() - yearStart.getTime()) / 86400000 + 1) / 7);
   359→}
   360→
   361→/** Returns a stable key like "2024-06-12" for a date — used for set lookups. */
   362→function dateKey(d: Date): string {
   363→  const y = d.getFullYear();
   364→  const m = String(d.getMonth() + 1).padStart(2, "0");
   365→  const day = String(d.getDate()).padStart(2, "0");
   366→  return `${y}-${m}-${day}`;
   367→}
   368→
   369→/** Format a Date as YYYY-MM-DD for the host's output. */
   370→export function formatDateISO(d: Date | null): string {
   371→  if (!d) return "";
   372→  const y = d.getFullYear();
   373→  const m = String(d.getMonth() + 1).padStart(2, "0");
   374→  const day = String(d.getDate()).padStart(2, "0");
   375→  return `${y}-${m}-${day}`;
   376→}

   556→
   557→  setSelectedRange(start: Date | null, end: Date | null): void {
   558→    let s = start ? atMidnight(start) : null;
   559→    let e = end ? atMidnight(end) : null;
   560→    if (s && e && compareDate(s, e) > 0) {
   561→      const tmp = s; s = e; e = tmp;
   562→    }

   733→  private emit(event: "select" | "change" | "navigate", opts: { source: CalendarEventPayload["source"] }): void {
   734→    if (this.destroyed) return;
   735→    const set = this.listeners.get(event);
   736→    if (!set || set.size === 0) return;
   737→    const payload: CalendarEventPayload = {
   738→      date: null,
   739→      selectedDate: this.getSelectedDate(),
   740→      selectedRange: this.getSelectedRange(),
   741→      selectedDates: this.getSelectedDates(),
   742→      viewDate: new Date(this.viewDate.getTime()),
   743→      mode: this.mode,
   744→      source: opts.source,
   745→    };
   746→    // Clone the set so a handler unsubscribing during dispatch doesn't skip siblings.
   747→    for (const h of Array.from(set)) {
   748→      try {
   749→        h(payload);
   750→      } catch (err) {
   751→        // Log but don't crash the dispatch.
   752→        console.error("Calendar handler error:", err);
   753→      }
   754→    }
   755→  }

   757→  private render(): void {
   758→    // Title
   759→    const monthName = MONTH_NAMES[this.viewDate.getMonth()];
   760→    this.titleEl.textContent = `${monthName} ${this.viewDate.getFullYear()}`;
   761→    this.titleEl.setAttribute("aria-label", `${monthName} ${this.viewDate.getFullYear()}`);
   762→
   763→    // Clear grid
   764→    this.gridEl.innerHTML = "";
   765→    this.gridEl.classList.toggle("calendar__grid--with-weeks", this.showWeekNumbers);
   766→    this.cells = [];
   767→
   768→    // Build weekday header row
   769→    const labels = WEEKDAY_LABELS[this.weekdayFormat];
   770→    for (let i = 0; i < 7; i++) {
   771→      const idx = (this.firstDayOfWeek + i) % 7;
   772→      const cell = document.createElement("div");
   773→      cell.className = "calendar__weekday";
   774→      if (idx === 0 || idx === 6) cell.classList.add("calendar__weekday--weekend");
   775→      cell.textContent = labels[idx];
   776→      cell.setAttribute("aria-hidden", "true");
   777→      this.gridEl.appendChild(cell);
   778→    }
   779→
   780→    // Compute the first visible day
   781→    const year = this.viewDate.getFullYear();
   782→    const month = this.viewDate.getMonth();
   783→    const firstOfMonth = new Date(year, month, 1);
   784→    const offset = (firstOfMonth.getDay() - this.firstDayOfWeek + 7) % 7;
   785→    const startDate = new Date(year, month, 1 - offset);
   786→
   787→    // 6 rows × 7 days is enough for every month.
   788→    const today = atMidnight(new Date());
   789→
   790→    // Track week numbers per row (only when enabled).
   791→    for (let row = 0; row < 6; row++) {
   792→      if (this.showWeekNumbers) {
   793→        const weekCell = document.createElement("div");
   794→        weekCell.className = "calendar__weeknum";
   795→        const weekStartDate = new Date(startDate);
   796→        weekStartDate.setDate(startDate.getDate() + row * 7);
   797→        weekCell.textContent = String(isoWeek(weekStartDate));
   798→        weekCell.setAttribute("aria-hidden", "true");
   799→        this.gridEl.appendChild(weekCell);
   800→      }
   801→      for (let col = 0; col < 7; col++) {
   802→        const idx = row * 7 + col;
   803→        const date = new Date(startDate);
   804→        date.setDate(startDate.getDate() + idx);
   805→        this.appendDayCell(date, month, today);
   806→      }
   807→    }
   808→  }

   810→  private appendDayCell(date: Date, visibleMonth: number, today: Date): void {
   811→    const dayOfWeek = date.getDay();
   812→    const isAdjacent = date.getMonth() !== visibleMonth;
   813→    const isWeekend = dayOfWeek === 0 || dayOfWeek === 6;
   814→    const isDisabled = this.isDateDisabled(date);
   815→    const isToday = compareDate(date, today) === 0;
   816→
   817→    const btn = document.createElement("button");
   818→    btn.type = "button";
   819→    btn.className = "calendar__day";
   820→    btn.textContent = String(date.getDate());
   821→    btn.setAttribute("data-date", dateKey(date));
   822→
   823→    if (isAdjacent) {
   824→      btn.classList.add("calendar__day--adjacent");
   825→      if (!this.showAdjacentMonths) {
   826→        btn.classList.add("calendar__day--disabled");
   827→        btn.setAttribute("aria-hidden", "true");
   828→        btn.tabIndex = -1;
   829→        btn.disabled = true;
   830→        btn.style.visibility = "hidden";
   831→      }
   832→    }
   833→    if (isWeekend && this.weekendHighlight) {
   834→      btn.classList.add("calendar__day--weekend");
   835→    } else if (isWeekend && !isAdjacent) {
   836→      // Default: subtle weekend treatment even without weekendHighlight, so the
   837→      // weekend column reads as a unit. (Overridden by selected/in-range.)
   838→      btn.classList.add("calendar__day--weekend");
   839→    }
   840→    if (isDisabled) {
   841→      btn.classList.add("calendar__day--disabled");
   842→      btn.setAttribute("aria-disabled", "true");
   843→      btn.disabled = true;
   844→    }
   845→    if (isToday && this.todayHighlight) {
   846→      btn.classList.add("calendar__day--today");
   847→    }
   848→    // Don't bother applying selection state to hidden adjacent days.
   849→    if (!(isAdjacent && !this.showAdjacentMonths)) {
   850→      this.applySelectionState(btn, date, isDisabled);
   851→    }
   852→
   853→    // Label for screen readers
   854→    const labelStr = `${MONTH_NAMES[date.getMonth()]} ${date.getDate()}, ${date.getFullYear()}`;
   855→    btn.setAttribute("aria-label", labelStr);
   856→    if (this.isCellSelected(date)) {
   857→      btn.setAttribute("aria-selected", "true");
   858→    } else {
   859→      btn.removeAttribute("aria-selected");
   860→    }
   861→
   862→    const cell: DayCell = { el: btn, date, isAdjacent, isDisabled, dayOfWeek };
   863→
   864→    const onClick = (e: MouseEvent): void => {
   865→      e.preventDefault();
   866→      this.handleDayClick(cell);
   867→    };
   868→    const onEnter = (): void => {
   869→      if (this.mode === "range" && this.isRangeAwaitingEnd && !isDisabled) {
   870→        this.setHoverPreview(date);
   871→      }
   872→    };
   873→    // Stash handlers on the element so destroy() can remove them.
   874→    (btn as HTMLButtonElement & { _calClick?: EventListener })._calClick = onClick as EventListener;
   875→    (btn as HTMLButtonElement & { _calEnter?: EventListener })._calEnter = onEnter as EventListener;
   876→    btn.addEventListener("click", onClick);
   877→    btn.addEventListener("pointerenter", onEnter);
   878→
   879→    this.gridEl.appendChild(btn);
   880→    this.cells.push(cell);
   881→  }

   883→  private applySelectionState(el: HTMLButtonElement, date: Date, isDisabled: boolean): void {
   884→    if (this.mode === "single") {
   885→      if (isSameDay(this.selectedDate, date)) {
   886→        el.classList.add("calendar__day--selected");
   887→      }
   888→    } else if (this.mode === "range") {
   889→      const start = this.range.start;
   890→      const end = this.range.end;
   891→      if (isSameDay(start, date)) {
   892→        el.classList.add("calendar__day--range-start");
   893→      }
   894→      if (isSameDay(end, date)) {
   895→        el.classList.add("calendar__day--range-end");
   896→      }
   897→      if (start && end && compareDate(date, start) > 0 && compareDate(date, end) < 0) {
   898→        el.classList.add("calendar__day--in-range");
   899→      }
   900→      // Hover preview (only when start is set but end isn't).
   901→      if (
   902→        this.isRangeAwaitingEnd &&
   903→        this.hoverPreviewDate &&
   904→        start &&
   905→        !end &&
   906→        !isSameDay(start, date) &&
   907→        !isDisabled
   908→      ) {
   909→        const lo = compareDate(start, this.hoverPreviewDate) < 0 ? start : this.hoverPreviewDate;
   910→        const hi = compareDate(start, this.hoverPreviewDate) < 0 ? this.hoverPreviewDate : start;
   911→        if (compareDate(date, lo) > 0 && compareDate(date, hi) < 0) {
   912→          el.classList.add("calendar__day--preview");
   913→        }
   914→      }
   915→    } else if (this.mode === "multi") {
   916→      for (const d of this.multi) {
   917→        if (isSameDay(d, date)) {
   918→          el.classList.add("calendar__day--selected");
   919→          break;
   920→        }
   921→      }
   922→    }
   923→  }

   925→  private isCellSelected(date: Date): boolean {
   926→    if (this.mode === "single") return isSameDay(this.selectedDate, date);
   927→    if (this.mode === "range") {
   928→      return isSameDay(this.range.start, date) || isSameDay(this.range.end, date);
   929→    }
   930→    if (this.mode === "multi") return this.multi.some((d) => isSameDay(d, date));
   931→    return false;
   932→  }
   933→
   934→  private isDateDisabled(date: Date): boolean {
   935→    if (this.disabledDates.has(dateKey(date))) return true;
   936→    if (this.disabledDaysOfWeek.has(date.getDay())) return true;
   937→    if (this.minDate && compareDate(date, this.minDate) < 0) return true;
   938→    if (this.maxDate && compareDate(date, this.maxDate) > 0) return true;
   939→    return false;
   940→  }

   942→  private setHoverPreview(date: Date | null): void {
   943→    if (this.mode !== "range" || !this.isRangeAwaitingEnd) return;
   944→    this.hoverPreviewDate = date ? atMidnight(date) : null;
   945→    this.refreshSelectionClasses();

   948→  /**
   949→   * Re-apply selection/preview classes without rebuilding the grid.
   950→   * Used on hover so the cursor doesn't lose its target cell.
   951→   */
   952→  private refreshSelectionClasses(): void {
   953→    const today = atMidnight(new Date());
   954→    for (const cell of this.cells) {
   955→      const el = cell.el;
   956→      // Clear only the selection-related classes; keep state classes (adjacent,
   957→      // weekend, today, disabled) which depend on the date, not the selection.
   958→      el.classList.remove(
   959→        "calendar__day--selected",
   960→        "calendar__day--range-start",
   961→        "calendar__day--range-end",
   962→        "calendar__day--in-range",
   963→        "calendar__day--preview",
   964→      );
   965→      if (cell.isAdjacent && !this.showAdjacentMonths) continue;
   966→      this.applySelectionState(el, cell.date, cell.isDisabled);
   967→      // Re-sync aria-selected + today-dot visibility.
   968→      if (this.isCellSelected(cell.date)) {
   969→        el.setAttribute("aria-selected", "true");
   970→      } else {
   971→        el.removeAttribute("aria-selected");
   972→      }
   973→      // Suppress unused warning for `today` when no refresh path uses it.
   974→      void today;
   975→    }
   976→  }

   980→  private handleDayClick(cell: DayCell): void {
   981→    if (cell.isDisabled) return;
   982→    if (cell.isAdjacent && !this.showAdjacentMonths) return;
   983→    const date = cell.date;
   984→
   985→    if (this.mode === "single") {
   986→      this.selectedDate = date;
   987→      // Snap the view to the clicked month in case it was an adjacent-day click.
   988→      this.snapViewToMonth(date);
   989→      this.render();
   990→      this.emit("select", { source: "select" });
   991→      this.emit("change", { source: "select" });
   992→      return;
   993→    }
   994→
   995→    if (this.mode === "range") {
   996→      if (!this.range.start || (this.range.start && this.range.end)) {
   997→        // Start a fresh range.
   998→        this.range = { start: date, end: null };
   999→        this.isRangeAwaitingEnd = true;
  1000→        this.hoverPreviewDate = null;
  1001→        this.snapViewToMonth(date);
  1002→        this.render();
  1003→        this.emit("select", { source: "select" });
  1004→        this.emit("change", { source: "select" });
  1005→        return;
  1006→      }
  1007→      // We have a start, set the end. Swap if end < start.
  1008→      let start = this.range.start;
  1009→      let end = date;
  1010→      if (compareDate(end, start) < 0) {
  1011→        const tmp = start; start = end; end = tmp;
  1012→      }
  1013→      this.range = { start, end };
  1014→      this.isRangeAwaitingEnd = false;
  1015→      this.hoverPreviewDate = null;
  1016→      this.snapViewToMonth(date);
  1017→      this.render();
  1018→      this.emit("select", { source: "select" });
  1019→      this.emit("change", { source: "select" });
  1020→      return;
  1021→    }
  1022→
  1023→    if (this.mode === "multi") {
  1024→      const idx = this.multi.findIndex((d) => isSameDay(d, date));
  1025→      if (idx >= 0) {
  1026→        this.multi.splice(idx, 1);
  1027→      } else {
  1028→        this.multi.push(new Date(date.getTime()));
  1029→      }
  1030→      this.snapViewToMonth(date);
  1031→      this.render();
  1032→      this.emit("select", { source: "select" });
  1033→      this.emit("change", { source: "select" });
  1034→      return;
  1035→    }
  1036→  }

  1038→  /** If the clicked date is in an adjacent month, snap the view to it. */
  1039→  private snapViewToMonth(date: Date): void {
  1040→    if (date.getMonth() !== this.viewDate.getMonth() || date.getFullYear() !== this.viewDate.getFullYear()) {
  1041→      this.viewDate = new Date(date.getFullYear(), date.getMonth(), 1);
  1042→      this.emit("navigate", { source: "select" });
  1043→    }
  1044→  }

  1046→  private handleKeyDown(e: KeyboardEvent): void {
  1047→    // Find the focused cell, if any.
  1048→    const target = e.target as HTMLElement;
  1049→    const cell = this.cells.find((c) => c.el === target);
  1050→    if (!cell) return;
  1051→    if (cell.isDisabled) return;
  1052→
  1053→    let deltaDays = 0;
  1054→    let deltaMonths = 0;
  1055→    switch (e.key) {
  1056→      case "ArrowRight": deltaDays = 1; break;
  1057→      case "ArrowLeft": deltaDays = -1; break;
  1058→      case "ArrowDown": deltaDays = 7; break;
  1059→      case "ArrowUp": deltaDays = -7; break;
  1060→      case "PageDown":
  1061→        deltaMonths = e.shiftKey ? 12 : 1;
  1062→        break;
  1063→      case "PageUp":
  1064→        deltaMonths = e.shiftKey ? -12 : -1;
  1065→        break;
  1066→      case "Home": {
  1067→        // First day of the visible week
  1068→        e.preventDefault();
  1069→        const firstOfWeek = this.firstOfDayInWeek(cell.date);
  1070→        this.focusDate(firstOfWeek);
  1071→        return;
  1072→      }
  1073→      case "End": {
  1074→        e.preventDefault();
  1075→        const lastOfWeek = this.lastOfDayInWeek(cell.date);
  1076→        this.focusDate(lastOfWeek);
  1077→        return;
  1078→      }
  1079→      case "Enter":
  1080→      case " ": {
  1081→        e.preventDefault();
  1082→        this.handleDayClick(cell);
  1083→        return;
  1084→      }
  1085→      default:
  1086→        return;
  1087→    }
  1088→
  1089→    e.preventDefault();
  1090→    let next: Date;
  1091→    if (deltaDays !== 0) {
  1092→      next = new Date(cell.date.getTime());
  1093→      next.setDate(next.getDate() + deltaDays);
  1094→    } else {
  1095→      next = new Date(cell.date.getFullYear(), cell.date.getMonth() + deltaMonths, 1);
  1096→    }
  1097→    this.focusDate(next);
  1098→  }
  1099→
  1100→  private firstOfDayInWeek(date: Date): Date {
  1101→    const offset = (date.getDay() - this.firstDayOfWeek + 7) % 7;
  1102→    const d = new Date(date.getTime());
  1103→    d.setDate(d.getDate() - offset);
  1104→    return d;
  1105→  }
  1106→
  1107→  private lastOfDayInWeek(date: Date): Date {
  1108→    const first = this.firstOfDayInWeek(date);
  1109→    const d = new Date(first.getTime());
  1110→    d.setDate(d.getDate() + 6);
  1111→    return d;
  1112→  }

  1114→  /**
  1115→   * Focus a specific date. If it's outside the visible month, navigate to it.
  1116→   * Disabled cells receive focus but are not clickable.
  1117→   */
  1118→  private focusDate(date: Date): void {
  1119→    if (date.getMonth() !== this.viewDate.getMonth() || date.getFullYear() !== this.viewDate.getFullYear()) {
  1120→      this.viewDate = new Date(date.getFullYear(), date.getMonth(), 1);
  1121→      this.render();
  1122→      this.emit("navigate", { source: "select" });
  1123→    }
  1124→    const target = this.cells.find((c) => isSameDay(c.date, date));
  1125→    if (target) {
  1126→      target.el.focus();
  1127→    }
  1128→  }
