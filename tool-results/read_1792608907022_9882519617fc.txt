    98→// ---- Playground ---------------------------------------------------------
    99→
   100→export function BoardPlayground() {
   101→  const meta = getComponent("board")!;
   102→  const [state, setState] = React.useState<PlaygroundState>(DEFAULTS);
   103→  const [items, setItems] = React.useState<BoardItem[]>([]);
   104→  const [selectedIds, setSelectedIds] = React.useState<string[]>([]);
   105→  const [view, setView] = React.useState<BoardView>({ panX: 0, panY: 0, zoom: 1 });
   106→  const [activeTab, setActiveTab] = React.useState<"javascript" | "html" | "css" | "api">("javascript");
   107→  const [registryVersion, setRegistryVersion] = React.useState(0);
   108→
   109→  const boardRef = React.useRef<import("@/lib/components/board").Board | null>(null);
   110→  const liveAreaRef = React.useRef<HTMLDivElement>(null);
   111→  const handlerRef = React.useRef<ShortcutHandler | null>(null);
   112→
   113→  // Register the built-in commands on mount.
   114→  React.useEffect(() => {
   115→    registerBuiltinCommands();
   116→    setRegistryVersion((v) => v + 1);
   117→  }, []);
   118→
   119→  // Seed a few starter items once the board is created so the canvas isn't empty.
   120→  React.useEffect(() => {
   121→    const board = boardRef.current;
   122→    if (!board) return;
   123→    // Only seed if there are no items yet.
   124→    if (board.getItems().length > 0) return;
   125→    board.addItem("rect", -120, -40, 120, 90, 0, "#D9FF66");
   126→    board.addItem("circle", 60, -60, 100, 100, 0, "#B9A0FF");
   127→    board.addItem("triangle", -40, 80, 110, 100, 0, "#FF858B");
   128→    setItems(board.getItems());
   129→    // Intentionally empty deps — runs once after the board mounts.
   130→  }, []);
   131→
   132→  // ---- Action handlers ----
   133→  //
   134→  // All actions read from refs (boardRef, liveAreaRef) and call stable
   135→  // setState setters, so their behavior is independent of render scope.
   136→  // We store the latest versions in an actionsRef (updated via a render-
   137→  // scoped effect to satisfy the React Compiler's "no ref writes during
   138→  // render" rule) so the ShortcutHandler — created once on mount — always
   139→  // dispatches to the freshest implementation.
   140→
   141→  const actionsRef = React.useRef<{
   142→    addShape: (type: ShapeType) => void;
   143→    deleteSelected: () => void;
   144→    fitAll: () => void;
   145→    fitSelection: () => void;
   146→    align: (type: AlignType) => void;
   147→    distribute: (type: DistributeType) => void;
   148→  }>({
   149→    addShape: () => {},
   150→    deleteSelected: () => {},
   151→    fitAll: () => {},
   152→    fitSelection: () => {},
   153→    align: () => {},
   154→    distribute: () => {},
   155→  });
   156→
   157→  const addShape = (type: ShapeType) => actionsRef.current.addShape(type);
   158→  const deleteSelected = () => actionsRef.current.deleteSelected();
   159→  const fitAll = () => actionsRef.current.fitAll();
   160→  const fitSelection = () => actionsRef.current.fitSelection();
   161→  const align = (type: AlignType) => actionsRef.current.align(type);
   162→  const distribute = (type: DistributeType) => actionsRef.current.distribute(type);
   163→
   164→  // Keep the actions ref fresh after every render.
   165→  React.useEffect(() => {
   166→    actionsRef.current.addShape = (type: ShapeType) => {
   167→      const board = boardRef.current;
   168→      const liveArea = liveAreaRef.current;
   169→      if (!board) return;
   170→      let cx = 0;
   171→      let cy = 0;
   172→      if (liveArea) {
   173→        const v = board.getView();
   174→        const w = liveArea.clientWidth;
   175→        const h = liveArea.clientHeight;
   176→        cx = (w / 2 - v.panX) / v.zoom;
   177→        cy = (h / 2 - v.panY) / v.zoom;
   178→      }
   179→      // Random offset near the center.
   180→      const offX = (Math.random() - 0.5) * 140;
   181→      const offY = (Math.random() - 0.5) * 140;
   182→      const color = SHAPE_COLORS[Math.floor(Math.random() * SHAPE_COLORS.length)];
   183→      const id = board.addItem(type, cx + offX, cy + offY, DEFAULT_SHAPE_SIZE, DEFAULT_SHAPE_SIZE, 0, color);
   184→      board.selectItem(id, false);
   185→      setItems(board.getItems());
   186→      setSelectedIds(board.getSelectedIds());
   187→    };
   188→    actionsRef.current.deleteSelected = () => {
   189→      const board = boardRef.current;
   190→      if (!board) return;
   191→      const ids = board.getSelectedIds();
   192→      if (ids.length === 0) return;
   193→      for (const id of ids) board.removeItem(id);
   194→      setItems(board.getItems());
   195→      setSelectedIds(board.getSelectedIds());
   196→    };
   197→    actionsRef.current.fitAll = () => {
   198→      boardRef.current?.zoomToFit();
   199→    };
   200→    actionsRef.current.fitSelection = () => {
   201→      boardRef.current?.zoomToFitSelection();
   202→    };
   203→    actionsRef.current.align = (type: AlignType) => {
   204→      const board = boardRef.current;
   205→      if (!board) return;
   206→      if (board.getSelectedIds().length < 2) {
   207→        toast.message("Select 2+ items to align");
   208→        return;
   209→      }
   210→      board.align(type);
   211→      setItems(board.getItems());
   212→    };
   213→    actionsRef.current.distribute = (type: DistributeType) => {
   214→      const board = boardRef.current;
   215→      if (!board) return;
   216→      if (board.getSelectedIds().length < 3) {
   217→        toast.message("Select 3+ items to distribute");
   218→        return;
   219→      }
   220→      board.distribute(type);
   221→      setItems(board.getItems());
   222→    };
   223→  });
   224→
   225→  // ---- Shortcut handler (scoped to board command IDs) ----
   226→
   227→  React.useEffect(() => {
   228→    const handler = new ShortcutHandler([...BOARD_COMMAND_IDS]);
   229→    handler.on("board.add-circle", () => actionsRef.current.addShape("circle"));
   230→    handler.on("board.add-rect", () => actionsRef.current.addShape("rect"));
   231→    handler.on("board.add-triangle", () => actionsRef.current.addShape("triangle"));
   232→    handler.on("board.delete-selected", () => actionsRef.current.deleteSelected());
   233→    handler.on("board.fit-all", () => actionsRef.current.fitAll());
   234→    handler.on("board.fit-selection", () => actionsRef.current.fitSelection());
   235→    handler.on("board.align-left", () => actionsRef.current.align("left"));
   236→    handler.on("board.align-center-h", () => actionsRef.current.align("center-h"));
   237→    handler.on("board.align-right", () => actionsRef.current.align("right"));
   238→    handler.on("board.align-top", () => actionsRef.current.align("top"));
   239→    handler.on("board.align-center-v", () => actionsRef.current.align("center-v"));
   240→    handler.on("board.align-bottom", () => actionsRef.current.align("bottom"));
   241→    handler.on("board.distribute-h", () => actionsRef.current.distribute("horizontal"));
   242→    handler.on("board.distribute-v", () => actionsRef.current.distribute("vertical"));
   243→    handler.attach();
   244→    handlerRef.current = handler;
   245→    return () => {
   246→      handler.destroy();
   247→      handlerRef.current = null;
   248→    };
   249→    // Intentionally empty deps — the handler reads through actionsRef so
   250→    // it always dispatches to the latest implementation without re-attaching.
   251→  }, []);
   252→
   253→  // ---- State updaters ----
   254→
   255→  const update = React.useCallback(
   256→    <K extends keyof PlaygroundState>(key: K, value: PlaygroundState[K]) => {
   257→      setState((s) => ({ ...s, [key]: value }));
   258→    },
   259→    [],
   260→  );
   261→
   262→  const reset = () => {
   263→    setState(DEFAULTS);
   264→    const board = boardRef.current;
   265→    if (board) {
   266→      // Remove all items, then re-seed.
   267→      for (const it of board.getItems()) board.removeItem(it.id);
   268→      board.addItem("rect", -120, -40, 120, 90, 0, "#D9FF66");
   269→      board.addItem("circle", 60, -60, 100, 100, 0, "#B9A0FF");
   270→      board.addItem("triangle", -40, 80, 110, 100, 0, "#FF858B");
   271→      board.clearSelection();
   272→      board.setView(0, 0, 1);
   273→      setItems(board.getItems());
   274→      setSelectedIds([]);
   275→      setView(board.getView());
   276→    }
   277→    toast.success("Reset to defaults");
   278→  };
   279→
   280→  // ---- Code generation ----
   281→
   282→  const jsCode = React.useMemo(() => generateJsCode(state), [state]);
   283→  const htmlCode = React.useMemo(() => generateHtmlCode(), []);
   284→  const cssCode = React.useMemo(() => generateCssCode(), []);
   285→  const apiCode = React.useMemo(() => generateApiCode(), []);
   286→
   287→  const boardCommands = React.useMemo(
   288→    () => commandRegistry.byIds([...BOARD_COMMAND_IDS]),
   289→    [registryVersion],
   290→  );
   291→
   292→  // ---- Render ----
   293→
   294→  return (
   295→    <div>
   296→      {/* Page header */}
   297→      <div className="border-b border-border bg-background">
   298→        <div className="mx-auto w-full max-w-[1400px] px-6 pt-6 pb-8 lg:px-10">
   299→          <Breadcrumbs />
   300→          <div className="mt-6 flex flex-col gap-4 lg:flex-row lg:items-end lg:justify-between">
   301→            <div>
   302→              <PageOverline>Components / Layout</PageOverline>
   303→              <h1 className="text-3xl font-semibold tracking-tight text-foreground sm:text-4xl">
   304→                {meta.name}
   305→              </h1>
   306→              <p className="mt-2 max-w-2xl text-[15px] text-muted-foreground">
   307→                {meta.longDescription}
   308→              </p>
   309→            </div>
   310→            <div className="flex flex-wrap items-center gap-2">
   311→              {meta.tags.map((tag) => (
   312→                <span
   313→                  key={tag}
   314→                  className="inline-flex items-center gap-1.5 rounded-md border border-border bg-muted/30 px-2.5 py-1 text-[11px] font-mono text-muted-foreground"
   315→                >
   316→                  {tag}
   317→                </span>
   318→              ))}
   319→              <span className="inline-flex items-center gap-1.5 rounded-md border border-brand-lemon/30 bg-brand-lemon/10 px-2.5 py-1 text-[11px] font-mono text-brand-lemon">
   320→                <span className="h-1.5 w-1.5 rounded-full bg-brand-lemon" />
   321→                {meta.status}
   322→              </span>
   323→            </div>
   324→          </div>
   325→        </div>
   326→      </div>
   327→
   328→      {/* Playground workspace */}
   329→      <div className="mx-auto w-full max-w-[1400px] px-6 py-8 lg:px-10">
   330→        <div className="grid grid-cols-1 gap-6 lg:grid-cols-[1fr_380px]">
   331→          {/* Live canvas panel */}
   332→          <section aria-label="Board live canvas" className="panel relative flex min-h-[640px] flex-col">
   333→            {/* Toolbar */}
   334→            <div className="flex flex-col gap-2 border-b border-border px-3 py-2.5">
   335→              <div className="flex flex-wrap items-center gap-1.5">
   336→                <ToolbarButton onClick={() => addShape("circle")} title="Add Circle (1)" variant="primary">
   337→                  <Circle className="h-3.5 w-3.5" />
   338→                  <span>Circle</span>
   339→                </ToolbarButton>
   340→                <ToolbarButton onClick={() => addShape("rect")} title="Add Rectangle (2)" variant="primary">
   341→                  <Square className="h-3.5 w-3.5" />
   342→                  <span>Rect</span>
   343→                </ToolbarButton>
   344→                <ToolbarButton onClick={() => addShape("triangle")} title="Add Triangle (3)" variant="primary">
   345→                  <Triangle className="h-3.5 w-3.5" />
   346→                  <span>Triangle</span>
   347→                </ToolbarButton>
   348→                <ToolbarDivider />
   349→                <ToolbarButton onClick={deleteSelected} title="Delete Selected (Delete)" variant="danger">
   350→                  <Trash2 className="h-3.5 w-3.5" />
   351→                  <span>Delete</span>
   352→                </ToolbarButton>
   353→                <ToolbarDivider />
   354→                <ToolbarButton onClick={fitAll} title="Zoom to Fit All (Cmd+0)">
   355→                  <Maximize2 className="h-3.5 w-3.5" />
   356→                  <span>Fit All</span>
   357→                </ToolbarButton>
   358→                <ToolbarButton onClick={fitSelection} title="Zoom to Fit Selection (Shift+Cmd+0)">
   359→                  <Scan className="h-3.5 w-3.5" />
   360→                  <span>Fit Selection</span>
   361→                </ToolbarButton>
   362→                <div className="ml-auto flex items-center gap-1">
   363→                  <ToolbarButton onClick={reset} title="Reset to defaults">
   364→                    <RotateCcw className="h-3.5 w-3.5" />
   365→                    <span>Reset</span>
   366→                  </ToolbarButton>
   367→                </div>
   368→              </div>
   369→              {/* Align + Distribute row */}
   370→              <div className="flex flex-wrap items-center gap-1.5">
   371→                <span className="eyebrow mr-1">Align</span>
   372→                {ALIGN_BUTTONS.map((b) => {
   373→                  const Icon = b.icon;
   374→                  return (
   375→                    <IconButton
   376→                      key={b.type}
   377→                      onClick={() => align(b.type)}
   378→                      title={`${b.label} (${formatShortcut(commandRegistry.get(b.cmdId)?.shortcut ?? { keys: "" })})`}
   379→                      aria-label={b.label}
   380→                    >
   381→                      <Icon className="h-3.5 w-3.5" />
   382→                    </IconButton>
   383→                  );
   384→                })}
   385→                <ToolbarDivider />
   386→                <span className="eyebrow mr-1">Distribute</span>
   387→                {DISTRIBUTE_BUTTONS.map((b) => {
   388→                  const Icon = b.icon;
   389→                  return (
   390→                    <IconButton
   391→                      key={b.type}
   392→                      onClick={() => distribute(b.type)}
   393→                      title={`${b.label} (${formatShortcut(commandRegistry.get(b.cmdId)?.shortcut ?? { keys: "" })})`}
   394→                      aria-label={b.label}
   395→                    >
   396→                      <Icon className="h-3.5 w-3.5" />
   397→                    </IconButton>
   398→                  );
   399→                })}
   400→              </div>
   401→            </div>
   402→
   403→            {/* Live Board canvas — fixed height is required */}
   404→            <div ref={liveAreaRef} className="relative h-[520px] w-full">
   405→              <BoardReact
   406→                className="absolute inset-0"
   407→                gridMode={state.gridMode}
   408→                gridSize={state.gridSize}
   409→                subGridDivisions={state.subGridDivisions}
   410→                snapToGrid={state.snapToGrid}
   411→                snapToItems={state.snapToItems}
   412→                onSelection={(e) => setSelectedIds(e.ids)}
   413→                onChange={() => setItems(boardRef.current?.getItems() ?? [])}
   414→                onView={(v) => setView(v)}
   415→                instanceRef={boardRef}
   416→              />
   417→              <div className="pointer-events-none absolute bottom-2 left-1/2 -translate-x-1/2 rounded-full bg-background/70 px-3 py-1 text-[10px] text-muted-foreground backdrop-blur">
   418→                Click to select. Shift-click for multi. Drag empty space for rubber-band. Space-drag to pan. Cmd+scroll to zoom.
   419→              </div>
   420→            </div>
   421→
   422→            {/* Output — item list */}
   423→            <div className="border-t border-border p-4">
   424→              <div className="mb-3 flex items-center justify-between">
   425→                <span className="eyebrow">Output — items</span>
   426→                <span className="font-mono text-[11px] text-muted-foreground">
   427→                  {items.length} item{items.length === 1 ? "" : "s"} · {selectedIds.length} selected
   428→                </span>
   429→              </div>
   430→              {items.length === 0 ? (
   431→                <p className="text-[12px] text-muted-foreground/70">
   432→                  No items on the board. Click <span className="text-brand-lemon">Circle</span>,{" "}
   433→                  <span className="text-brand-lemon">Rect</span>, or{" "}
   434→                  <span className="text-brand-lemon">Triangle</span> above, or press 1 / 2 / 3.
   435→                </p>
   436→              ) : (
   437→                <div className="max-h-56 overflow-y-auto rounded-md border border-border">
   438→                  <table className="w-full text-left text-[12px]">
   439→                    <thead className="sticky top-0 bg-muted/40 backdrop-blur">
   440→                      <tr>
   441→                        <th className="px-2.5 py-1.5 font-mono text-[10px] uppercase text-muted-foreground">id</th>
   442→                        <th className="px-2.5 py-1.5 font-mono text-[10px] uppercase text-muted-foreground">type</th>
   443→                        <th className="px-2.5 py-1.5 font-mono text-[10px] uppercase text-muted-foreground">x, y</th>
   444→                        <th className="px-2.5 py-1.5 font-mono text-[10px] uppercase text-muted-foreground">w × h</th>
   445→                        <th className="px-2.5 py-1.5 font-mono text-[10px] uppercase text-muted-foreground">rot</th>
   446→                        <th className="px-2.5 py-1.5 font-mono text-[10px] uppercase text-muted-foreground">color</th>
   447→                      </tr>
   448→                    </thead>
   449→                    <tbody className="divide-y divide-border">
   450→                      {items.map((item) => {
   451→                        const isSel = selectedIds.includes(item.id);
   452→                        return (
   453→                          <tr
   454→                            key={item.id}
   455→                            className={cn(
   456→                              "transition-colors",
   457→                              isSel ? "bg-brand-lemon/10" : "hover:bg-muted/20",
   458→                            )}
   459→                          >
   460→                            <td className="px-2.5 py-1.5 font-mono text-brand-lemon">{item.id}</td>
   461→                            <td className="px-2.5 py-1.5 text-foreground">{item.type}</td>
   462→                            <td className="px-2.5 py-1.5 font-mono text-muted-foreground">
   463→                              {Math.round(item.x)}, {Math.round(item.y)}
   464→                            </td>
   465→                            <td className="px-2.5 py-1.5 font-mono text-muted-foreground">
   466→                              {Math.round(item.width)} × {Math.round(item.height)}
   467→                            </td>
   468→                            <td className="px-2.5 py-1.5 font-mono text-muted-foreground">
   469→                              {Math.round(item.rotation)}°
   470→                            </td>
   471→                            <td className="px-2.5 py-1.5">
   472→                              <span className="inline-flex items-center gap-1.5">
   473→                                <span
   474→                                  className="inline-block h-3 w-3 rounded-sm border border-border"
   475→                                  style={{ background: item.color }}
   476→                                  aria-hidden="true"
   477→                                />
   478→                                <span className="font-mono text-[10px] text-muted-foreground">{item.color}</span>
   479→                              </span>
   480→                            </td>
   481→                          </tr>
   482→                        );
   483→                      })}
