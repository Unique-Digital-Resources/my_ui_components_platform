    70→    setHexInput(DEFAULTS.color);

    85→      update("color", value.startsWith("#") ? value : `#${value}`);

    87→      setHexError("Invalid color value");
    88→    }
    89→  }, [update]);

    91→  const rgbStr = color ? `rgb(${color.rgb.r}, ${color.rgb.g}, ${color.rgb.b})` : "—";

    93→    ? `${Math.round(color.hsv.h)}°, ${Math.round(color.hsv.s)}%, ${Math.round(color.hsv.v)}%`

    96→    ? `${Math.round(color.hsl.h)}°, ${Math.round(color.hsl.s)}%, ${Math.round(color.hsl.l)}%`

    99→  // Code generation — reflects current configuration
   100→  const jsCode = React.useMemo(() => generateJsCode(state), [state]);
   101→  const htmlCode = React.useMemo(() => generateHtmlCode(state), [state]);
   102→  const cssCode = React.useMemo(() => generateCssCode(), []);
   103→  const apiCode = React.useMemo(() => generateApiCode(), []);

   114→              <h1 className="text-3xl font-semibold tracking-tight text-foreground sm:text-4xl">
   115→                {meta.name}
   116→              </h1>
   117→              <p className="mt-2 max-w-2xl text-[15px] text-muted-foreground">
   118→                {meta.longDescription}
   119→              </p>
   120→            </div>
   121→            <div className="flex flex-wrap items-center gap-2">
   122→              {meta.tags.map((tag) => (
   123→                <span
   124→                  key={tag}
   125→                  className="inline-flex items-center gap-1.5 rounded-md border border-border bg-muted/30 px-2.5 py-1 text-[11px] font-mono text-muted-foreground"
   126→                >
   127→                  {tag}
   128→                </span>
   129→              ))}
   130→              <span className="inline-flex items-center gap-1.5 rounded-md border border-brand-lemon/30 bg-brand-lemon/10 px-2.5 py-1 text-[11px] font-mono text-brand-lemon">
   131→                <span className="h-1.5 w-1.5 rounded-full bg-brand-lemon" />
   132→                {meta.status}
   133→              </span>
   134→            </div>
   135→          </div>
   136→        </div>
   137→      </div>
   138→
   139→      {/* Playground workspace */}
   140→      <div className="mx-auto w-full max-w-[1400px] px-6 py-8 lg:px-10">
   141→        <div className="grid grid-cols-1 gap-6 lg:grid-cols-[1fr_360px]">
   142→          {/* Live preview */}

   144→            aria-label="Live preview"
   145→            className="panel relative flex min-h-[480px] flex-col"
   146→          >
   147→            <div className="flex items-center justify-between border-b border-border px-4 py-2.5">
   148→              <div className="flex items-center gap-2">
   149→                <span className="h-2 w-2 rounded-full bg-brand-lemon animate-pulse" />
   150→                <span className="eyebrow">Live canvas</span>
   151→              </div>
   152→              <div className="flex items-center gap-1">
   153→                <button
   154→                  type="button"
   155→                  onClick={reset}
   156→                  className="inline-flex h-7 items-center gap-1.5 rounded-md border border-border px-2 text-[12px] text-muted-foreground transition-colors hover:bg-accent hover:text-foreground"
   157→                >
   158→                  <RotateCcw className="h-3 w-3" />

   160→                </button>

   163→            <div className="flex flex-1 items-center justify-center bg-grid p-8">

   166→                shape={state.shape}
   167→                size={state.size}

   282→                    spellCheck={false}
   283→                    className={cn(
   284→                      "h-9 flex-1 rounded-md border bg-background px-2.5 font-mono text-[12px] text-foreground focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring",

   288→                  />
   289→                </div>
   290→              </PropertyField>
   291→
   292→              {/* Rotate shape with hue */}

   295→                  <div className="text-[12px] font-medium text-foreground">

   299→                    Triangle tracks the wheel&apos;s hue position

   301→                </div>
   302→                <button
   303→                  type="button"
   304→                  role="switch"

   308→                  className={cn(
   309→                    "relative inline-flex h-5 w-9 shrink-0 items-center rounded-full transition-colors focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring",
   310→                    state.rotateShape
   311→                      ? "bg-brand-lemon"
   312→                      : "bg-muted-foreground/30"
   313→                  )}
   314→                >
   315→                  <span
   316→                    className={cn(
   317→                      "inline-block h-3.5 w-3.5 rounded-full bg-background shadow-sm transition-transform",
   318→                      state.rotateShape ? "translate-x-4" : "translate-x-0.5"
   319→                    )}
   320→                  />
   321→                </button>
   322→              </div>
   323→            </div>
   324→
   325→            {/* Color output values */}
   326→            <div className="mt-auto border-t border-border p-4">
   327→              <span className="eyebrow mb-2 block">Output</span>
   328→              <div className="space-y-1.5">
   329→                <CopyableValue label="HEX" value={color?.hex ?? "#4287f5"} />
   330→                <CopyableValue label="RGB" value={rgbStr} />
   331→                <CopyableValue label="HSV" value={hsvStr} />
   332→                <CopyableValue label="HSL" value={hslStr} />
   333→              </div>
   334→            </div>
   335→          </aside>
   336→        </div>
   337→
   338→        {/* Code panel */}
   339→        <section className="panel mt-6 overflow-hidden">
   340→          <div className="flex items-center justify-between border-b border-border px-2">
   341→            <div className="flex">
   342→              {(
   343→                [
   344→                  ["javascript", "JavaScript"],
   345→                  ["html", "HTML"],
   346→                  ["css", "CSS"],
   347→                  ["api", "API"],
   348→                ] as const
   349→              ).map(([key, label]) => (
   350→                <button
   351→                  key={key}
   352→                  type="button"
   353→                  onClick={() => setActiveTab(key)}
   354→                  className={cn(
   355→                    "relative h-9 px-3 text-[12px] font-medium transition-colors focus-visible:outline-none",
   356→                    activeTab === key
   357→                      ? "text-foreground"
   358→                      : "text-muted-foreground hover:text-foreground"
   359→                  )}
   360→                >
   361→                  {label}
   362→                  {activeTab === key && (
   363→                    <span className="absolute inset-x-2 -bottom-px h-px bg-brand-lemon" />
   364→                  )}
   365→                </button>
   366→              ))}
   367→            </div>
   368→            <span className="pr-2 text-[11px] font-mono text-muted-foreground/70">
   369→              Synchronized with live configuration
   370→            </span>
   371→          </div>
   372→          <div className="p-4">
   373→            {activeTab === "javascript" && (
   374→              <CodeBlock code={jsCode} language="javascript" filename="init.js" />
   375→            )}
   376→            {activeTab === "html" && (
   377→              <CodeBlock code={htmlCode} language="html" filename="index.html" />
   378→            )}
   379→            {activeTab === "css" && (
   380→              <CodeBlock code={cssCode} language="css" filename="styles.css" />
   381→            )}
   382→            {activeTab === "api" && (
   383→              <CodeBlock code={apiCode} language="typescript" filename="api.ts" />
   384→            )}
   385→          </div>
   386→        </section>
   387→
   388→        {/* Documentation below the playground */}
   389→        <ComponentDocumentation />
   390→      </div>
   391→    </div>
   392→  );
   393→}

   395→function PropertyField({
   396→  label,
   397→  hint,
   398→  value,
   399→  error,
   400→  children,
   401→}: {
   402→  label: string;
   403→  hint?: string;
   404→  value?: string;
   405→  error?: string | null;
   406→  children: React.ReactNode;
   407→}) {
   408→  return (
   409→    <div>
   410→      <div className="mb-1.5 flex items-center justify-between">
   411→        <label className="text-[12px] font-medium text-foreground">{label}</label>
   412→        {value && (
   413→          <span className="font-mono text-[11px] text-brand-lemon">{value}</span>
   414→        )}
   415→      </div>
   416→      {children}
   417→      {error ? (
   418→        <p className="mt-1 text-[11px] text-destructive">{error}</p>
   419→      ) : hint ? (
   420→        <p className="mt-1 text-[11px] text-muted-foreground/70">{hint}</p>
   421→      ) : null}
   422→    </div>
   423→  );
   424→}

   426→function RangeInput({
   427→  min,
   428→  max,
   429→  step,
   430→  value,
   431→  onChange,
   432→}: {
   433→  min: number;
   434→  max: number;
   435→  step: number;
   436→  value: number;
   437→  onChange: (v: number) => void;
   438→}) {
   439→  return (
   440→    <div className="flex items-center gap-2">
   441→      <input
   442→        type="range"
   443→        min={min}
   444→        max={max}
   445→        step={step}
   446→        value={value}
   447→        onChange={(e) => onChange(Number(e.target.value))}
   448→        className="flex-1 h-1.5 cursor-pointer appearance-none rounded-full bg-muted accent-brand-lemon"
   449→        style={{
   450→          background: `linear-gradient(to right, var(--brand-lemon) 0%, var(--brand-lemon) ${
   451→            ((value - min) / (max - min)) * 100
   452→          }%, var(--muted) ${((value - min) / (max - min)) * 100}%, var(--muted) 100%)`,
   453→        }}
   454→      />
   455→      <input
   456→        type="number"
   457→        min={min}
   458→        max={max}
   459→        value={value}
   460→        onChange={(e) => onChange(Number(e.target.value))}
   461→        className="h-7 w-14 rounded-md border border-border bg-background px-1.5 text-center font-mono text-[11px] text-foreground focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring"
   462→      />
   463→    </div>
   464→  );
   465→}

   564→function generateApiCode(): string {
   565→  return `// Public API
   566→
   567→type Shape = "triangle" | "square" | "disk" | "none";
   568→
   569→interface ColorOutput {
   570→  hex: string;                              // #RRGGBB
   571→  rgb: { r: number; g: number; b: number }; // 0-255
   572→  hsv: { h: number; s: number; v: number }; // h:0-360, s/v:0-100
   573→  hsl: { h: number; s: number; l: number }; // h:0-360, s/l:0-100
   574→}
