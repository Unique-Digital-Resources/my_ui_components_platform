     1→/**
     2→ * GanttTimeline — a Gantt Chart Timeline Bars component (vanilla TypeScript).
     3→ *
     4→ * Spec reference: §15 Minimum viable specification.
     5→ *
     6→ *   - Display generic items as horizontal bars.
     7→ *   - Configure start and end fields independently of item content.
     8→ *   - Map time intervals to a shared horizontal coordinate system.
     9→ *   - Support configurable visible range, scale, and scrolling.
    10→ *   - Support optional row assignment and row headers.
    11→ *   - Provide a minimal default bar appearance.
    12→ *   - Allow custom templates without requiring them.
    13→ *   - Keep bar geometry independent of rendered content.
    14→ *   - Support selection and optional move/resize interactions.
    15→ *   - Define interval validation, snapping, and time-zone behavior.
    16→ *   - Expose view-range, scale, selection, move, and resize events.
    17→ *   - Support responsive layout and usable targets for short bars.
    18→ *   - Provide keyboard alternatives and accessible item descriptions.
    19→ *   - Separate visualization behavior from scheduling logic and persistence.
    20→ *
    21→ * Architecture: single-file DOM-based renderer with scoped CSS (`.gantt-`
    22→ * prefix). Time is epoch milliseconds (number) throughout — the host converts
    23→ * to/from timezones. The time-to-pixel mapping is linear: x(t) = W * (t - T0)
    24→ * / (T1 - T0). Bar content is laid out by the content-template system
    25→ * (src/lib/templates/); pass "none" to use native rendering (plain bar).
    26→ */

   541→// ---- Utilities ----
   542→
   543→function uid(prefix: string): string {
   544→  return `${prefix}-${Math.random().toString(36).slice(2, 9)}-${Date.now().toString(36).slice(-4)}`;
   545→}
   546→
   547→function clamp(n: number, min: number, max: number): number {
   548→  return Math.max(min, Math.min(max, n));
   549→}
   550→
   551→/** Snap a time (epoch ms) to the nearest multiple of snapInterval. */
   552→function snapTime(t: number, snap: number): number {
   553→  if (!snap || snap <= 0) return t;
   554→  return Math.round(t / snap) * snap;
   555→}

   610→      resizable: options.resizable ?? true,
   611→      rowMovable: options.rowMovable ?? false,
   612→      snapInterval: options.snapInterval ?? SCALE_SNAP[options.scale ?? "day"],
   613→      showGrid: options.showGrid ?? true,
   614→      showToday: options.showToday ?? true,
   615→      showAxis: options.showAxis ?? true,
   616→      axisPosition: options.axisPosition ?? "top",
   617→      readOnly: options.readOnly ?? false,
   618→      disabled: options.disabled ?? false,
   619→      progressEnabled: options.progressEnabled ?? true,
   620→      minDuration: options.minDuration ?? 0,
   621→      templateId: options.templateId ?? "gantt.standard-bar",
   622→    };
   623→
   624→    // Defensive clone
   625→    this.items = (this.opts.items as GanttItem[]).map((it) => ({ ...it }));
   626→    this.rows = (this.opts.rows as GanttRow[]).map((r) => ({ ...r }));
   627→
   628→    // If no rows provided, create a single default row
   629→    if (this.rows.length === 0) {
   630→      this.rows = [{ id: "default", title: "Timeline" }];
   631→    }
   632→
   633→    this.render();
   634→  }
   635→
   636→  // ---- Time-to-pixel mapping ----
   637→
   638→  /** Convert a time (epoch ms) to a pixel x within the grid. */
   639→  private timeToX(t: number): number {
   640→    const range = this.opts.viewEnd - this.opts.viewStart;
   641→    if (range <= 0) return 0;
   642→    return ((t - this.opts.viewStart) / range) * this.getGridWidth();
   643→  }
   644→
   645→  /** Convert a pixel x to a time (epoch ms). */
   646→  private xToTime(x: number): number {
   647→    const w = this.getGridWidth();
   648→    if (w <= 0) return this.opts.viewStart;
   649→    return this.opts.viewStart + (x / w) * (this.opts.viewEnd - this.opts.viewStart);
   650→  }
   651→
   652→  /** The grid width in pixels. At least the scroll container's clientWidth. */
   653→  private getGridWidth(): number {
   654→    if (!this.scrollEl) return 0;
   655→    return Math.max(this.scrollEl.clientWidth, this.opts.rowHeaderWidth ? 0 : 0) || this.scrollEl.clientWidth;
   656→  }

   658→  // ---- Rendering ----
   659→
   660→  private render(): void {
   661→    if (this._destroyed) return;
   662→    this.barEls.clear();
   663→    this.rowEls.clear();
   664→
   665→    const root = document.createElement("div");
   666→    root.className = "gantt-timeline";
   667→    if (this.opts.axisPosition === "bottom") root.classList.add("gantt-axis-bottom");
   668→    if (this.opts.readOnly) root.classList.add("gantt-readonly");
   669→    if (this.opts.disabled) root.classList.add("gantt-disabled");
   670→    root.style.setProperty("--gt-bar-h", `${this.opts.barHeight}px`);
   671→
   672→    // Axis
   673→    if (this.opts.showAxis) {
   674→      const axis = document.createElement("div");
   675→      axis.className = "gantt-axis";
   676→      if (this.opts.showRowHeaders) {
   677→        const spacer = document.createElement("div");
   678→        spacer.className = "gantt-axis-spacer";
   679→        spacer.style.width = `${this.opts.rowHeaderWidth}px`;
   680→        axis.appendChild(spacer);
   681→      }
   682→      const labels = document.createElement("div");
   683→      labels.className = "gantt-axis-labels";
   684→      this.renderAxisLabels(labels);
   685→      axis.appendChild(labels);
   686→      root.appendChild(axis);
   687→    }
   688→
   689→    // Body
   690→    const body = document.createElement("div");
   691→    body.className = "gantt-body";
   692→
   693→    // Row headers
   694→    if (this.opts.showRowHeaders) {
   695→      const headers = document.createElement("div");
   696→      headers.className = "gantt-row-headers";
   697→      headers.style.width = `${this.opts.rowHeaderWidth}px`;
   698→      for (const row of this.rows) {
   699→        const h = document.createElement("div");
   700→        h.className = "gantt-row-header";
   701→        h.style.height = `${this.opts.rowHeight}px`;
   702→        if (row.color) {
   703→          const dot = document.createElement("span");
   704→          dot.className = "gantt-row-header-color";
   705→          dot.style.background = row.color;
   706→          h.appendChild(dot);
   707→        }
   708→        const title = document.createElement("span");
   709→        title.className = "gantt-row-header-title";
   710→        title.textContent = row.title ?? row.id;
   711→        h.appendChild(title);
   712→        const count = this.items.filter((it) => (it.rowId ?? "default") === row.id).length;
   713→        const countEl = document.createElement("span");
   714→        countEl.className = "gantt-row-header-count";
   715→        countEl.textContent = String(count);
   716→        h.appendChild(countEl);
   717→        headers.appendChild(h);
   718→      }
   719→      body.appendChild(headers);
   720→    }
   721→
   722→    // Scrollable grid
   723→    const scroll = document.createElement("div");
   724→    scroll.className = "gantt-scroll";
   725→    scroll.addEventListener("scroll", () => {
   726→      if (this.opts.showAxis) {
   727→        this.axisLabelsEl.style.transform = `translateX(${-scroll.scrollLeft}px)`;
   728→      }
   729→    });
   730→    this.scrollEl = scroll;
   731→
   732→    const grid = document.createElement("div");
   733→    grid.className = "gantt-grid";
   734→    grid.style.minWidth = "100%";
   735→    grid.style.width = `${this.getGridWidth()}px`;
   736→    this.gridEl = grid;
   737→
   738→    // Grid lines
   739→    if (this.opts.showGrid) {
   740→      this.renderGridLines(grid);
   741→    }
   742→
   743→    // Rows + bars
   744→    for (const row of this.rows) {
   745→      const rowEl = this.renderRow(row);
   746→      grid.appendChild(rowEl);
   747→      this.rowEls.set(row.id, rowEl);
   748→    }
   749→
   750→    // Today marker
   751→    if (this.opts.showToday) {
   752→      this.renderTodayMarker(grid);
   753→    }
   754→
   755→    scroll.appendChild(grid);
   756→    body.appendChild(scroll);
   757→    root.appendChild(body);
   758→
   759→    this.container.innerHTML = "";
   760→    this.container.appendChild(root);
   761→    this.root = root;
   762→
   763→    // After layout, set the grid width to match or exceed the scroll width
   764→    requestAnimationFrame(() => {
   765→      if (!this._destroyed) this.updateGridWidth();
   766→    });
   767→  }
   768→
   769→  private updateGridWidth(): void {
   770→    if (!this.gridEl || !this.scrollEl) return;
   771→    const w = this.scrollEl.clientWidth;
   772→    this.gridEl.style.width = `${w}px`;
   773→    // Re-position bars after width change
   774→    this.positionAllBars();
   775→    if (this.opts.showAxis) this.renderAxisLabels(this.axisLabelsEl);
   776→    if (this.opts.showGrid) this.renderGridLines(this.gridEl);
   777→    if (this.opts.showToday) this.renderTodayMarker(this.gridEl);
   778→  }
   779→
   780→  private renderAxisLabels(container: HTMLElement): void {
   781→    container.innerHTML = "";
   782→    this.axisLabelsEl = container as HTMLDivElement;
   783→    const w = container.clientWidth || this.getGridWidth();
   784→    const majorTick = SCALE_MAJOR_TICK[this.opts.scale];
   785→    const start = Math.ceil(this.opts.viewStart / majorTick) * majorTick;
   786→    for (let t = start; t <= this.opts.viewEnd; t += majorTick) {
   787→      const x = ((t - this.opts.viewStart) / (this.opts.viewEnd - this.opts.viewStart)) * w;
   788→      const label = document.createElement("div");
   789→      label.className = "gantt-axis-major";
   790→      label.textContent = SCALE_FORMAT[this.opts.scale](t, "en-US");
   791→      label.style.left = `${x}px`;
   792→      container.appendChild(label);
   793→    }
   794→  }
   795→
   796→  private renderGridLines(grid: HTMLElement): void {
   797→    // Remove old grid lines
   798→    grid.querySelectorAll(".gantt-grid-line").forEach((el) => el.remove());
   799→    const w = this.gridEl.clientWidth || this.getGridWidth();
   800→    const minorTick = SCALE_TICK[this.opts.scale];
   801→    const majorTick = SCALE_MAJOR_TICK[this.opts.scale];
   802→    const start = Math.ceil(this.opts.viewStart / minorTick) * minorTick;
   803→    for (let t = start; t <= this.opts.viewEnd; t += minorTick) {
   804→      const x = ((t - this.opts.viewStart) / (this.opts.viewEnd - this.opts.viewStart)) * w;
   805→      const line = document.createElement("div");
   806→      line.className = "gantt-grid-line";
   807→      if (t % majorTick === 0) line.classList.add("gantt-grid-line-major");
   808→      line.style.left = `${x}px`;
   809→      grid.appendChild(line);
   810→    }
   811→  }

   813→  private renderTodayMarker(grid: HTMLElement): void {
   814→    if (this.todayEl) this.todayEl.remove();
   815→    const now = Date.now();
   816→    if (now < this.opts.viewStart || now > this.opts.viewEnd) return;
   817→    const x = this.timeToX(now);
   818→    const marker = document.createElement("div");
   819→    marker.className = "gantt-today";
   820→    marker.style.left = `${x}px`;
   821→    const label = document.createElement("div");
   822→    label.className = "gantt-today-label";
   823→    label.textContent = "Today";
   824→    marker.appendChild(label);
   825→    grid.appendChild(marker);
   826→    this.todayEl = marker;
   827→  }

   829→  private renderRow(row: GanttRow): HTMLElement {
   830→    const el = document.createElement("div");
   831→    el.className = "gantt-row";
   832→    el.setAttribute("data-row-id", row.id);
   833→    el.style.height = `${this.opts.rowHeight}px`;
   834→
   835→    const rowItems = this.items.filter((it) => (it.rowId ?? "default") === row.id);
   836→    if (rowItems.length === 0) {
   837→      // Keep the row but don't render an empty-state inside the grid (would
   838→      // be clipped). The row header shows the count.
   839→    }
   840→    for (const item of rowItems) {
   841→      const barEl = this.renderBar(item, row);
   842→      el.appendChild(barEl);
   843→      this.barEls.set(item.id, barEl);
   844→    }
   845→    return el;
   846→  }

   848→  private renderBar(item: GanttItem, _row: GanttRow): HTMLElement {
   849→    const el = document.createElement("div");
   850→    el.className = "gantt-bar";
   851→    el.setAttribute("data-item-id", item.id);
   852→    el.setAttribute("role", "button");
   853→    el.setAttribute("tabindex", this.opts.readOnly ? "-1" : "0");
   854→    el.setAttribute(
   855→      "aria-label",
   856→      this.buildBarAriaLabel(item),
   857→    );
   858→    if (this.selectedItemId === item.id) el.classList.add("gantt-selected");
   859→    if (this.opts.readOnly) el.classList.add("gantt-readonly-bar");
   860→    if (item.disabled) el.classList.add("gantt-disabled-bar");
   861→    if (item.color) el.style.background = item.color;
   862→
   863→    // Progress overlay
   864→    if (this.opts.progressEnabled && typeof item.progress === "number" && item.progress > 0) {
   865→      const prog = document.createElement("div");
   866→      prog.className = "gantt-bar-progress";
   867→      prog.style.width = `${clamp(item.progress, 0, 100)}%`;
   868→      el.appendChild(prog);
   869→    }
   870→
   871→    // Templated content
   872→    const tpl = this.getTemplate();
   873→    const density: TemplateDensity = "standard";
   874→    const hostState = {
   875→      readOnly: this.opts.readOnly,
   876→      movable: this.opts.movable,
   877→      resizable: this.opts.resizable,
   878→      scale: this.opts.scale,
   879→    };
   880→    if (tpl) {
   881→      for (const slot of tpl.slots) {
   882→        if (!slotVisible(slot, density)) continue;
   883→        const renderer = rendererRegistry.get(slot.renderer);
   884→        if (!renderer) continue;
   885→        const slotEl = renderer({ item, slot, density, host: hostState });
   886→        if (slotEl) {
   887→          slotEl.style.pointerEvents = "none";
   888→          el.appendChild(slotEl);
   889→        }
   890→      }
   891→    } else {
   892→      // Native rendering: plain bar with optional title
   893→      if (item.title) {
   894→        const title = document.createElement("div");
   895→        title.className = "gantt-bar-title";
   896→        title.textContent = item.title;
   897→        el.appendChild(title);
   898→      }
   899→    }
   900→
   901→    // Resize handles
   902→    if (this.opts.resizable && !this.opts.readOnly && !item.disabled) {
   903→      const left = document.createElement("div");
   904→      left.className = "gantt-resize-handle gantt-resize-handle-left";
   905→      left.setAttribute("role", "slider");
   906→      left.setAttribute("aria-label", `Resize start of ${item.title ?? item.id}`);
   907→      left.addEventListener("pointerdown", (e) => this.onResizeStart(e, item, "start"));
   908→      el.appendChild(left);
   909→
   910→      const right = document.createElement("div");
   911→      right.className = "gantt-resize-handle gantt-resize-handle-right";
   912→      right.setAttribute("role", "slider");
   913→      right.setAttribute("aria-label", `Resize end of ${item.title ?? item.id}`);
   914→      right.addEventListener("pointerdown", (e) => this.onResizeStart(e, item, "end"));
   915→      el.appendChild(right);
   916→    }
   917→
   918→    // Pointer interactions
   919→    if (!this.opts.readOnly && !item.disabled) {
   920→      el.addEventListener("pointerdown", (e) => this.onBarPointerDown(e, item));
   921→      el.addEventListener("click", (e) => {
   922→        const target = e.target as HTMLElement;
   923→        if (target.closest(".gantt-resize-handle")) return;
   924→        if (this.opts.selectable) this.selectBar(item.id, "pointer");
   925→      });
   926→      el.addEventListener("dblclick", () => {
   927→        this.emit("bar_activated", { itemId: item.id, source: "pointer" });
   928→      });
   929→      el.addEventListener("keydown", (e) => this.onBarKeyDown(e, item));
   930→    } else if (this.opts.selectable) {
   931→      el.addEventListener("click", () => this.selectBar(item.id, "pointer"));
   932→    }
   933→
   934→    // Position the bar
   935→    this.positionBar(el, item);
   936→    return el;
   937→  }
   938→
   939→  private positionBar(el: HTMLElement, item: GanttItem): void {
   940→    const x = this.timeToX(item.start);
   941→    const w = Math.max(this.timeToX(item.end) - x, this.opts.barMinWidth);
   942→    el.style.left = `${x}px`;
   943→    el.style.width = `${w}px`;
   944→    el.style.height = `${this.opts.barHeight}px`;
   945→  }
   946→
   947→  private positionAllBars(): void {
   948→    for (const [id, el] of this.barEls) {
   949→      const item = this.findItem(id);
   950→      if (item) this.positionBar(el, item);
   951→    }
   952→  }
   953→
   954→  private buildBarAriaLabel(item: GanttItem): string {
   955→    const parts: string[] = [];
   956→    if (item.title) parts.push(item.title);
   957→    const fmt = (t: number) => new Date(t).toLocaleDateString("en-US", {
   958→      month: "short", day: "numeric", year: "numeric",
   959→    });
   960→    parts.push(`${fmt(item.start)} to ${fmt(item.end)}`);
   961→    if (typeof item.progress === "number") parts.push(`${Math.round(item.progress)}% complete`);
   962→    return parts.join(", ");
   963→  }

   985→  selectBar(itemId: string | null, source: "pointer" | "keyboard" | "programmatic" = "programmatic"): void {
   986→    if (this._destroyed) return;
   987→    if (!this.opts.selectable) return;

   995→      this.emit("bar_selected", { itemId, previousItemId: prev, source });
   996→      this.emit("selection_changed", {
   997→        itemIds: itemId ? [itemId] : [],
   998→        previousItemIds: prev ? [prev] : [],
   999→        source,
  1000→      });
  1001→    }
  1002→  }

  1014→  moveBar(itemId: string, newStart: number, newEnd: number, source: "pointer" | "keyboard" | "programmatic" = "programmatic"): void {
  1015→    if (this._destroyed) return;
  1016→    if (this.opts.readOnly) {
  1017→      this.emit("bar_move_rejected", { itemId, start: newStart, end: newEnd, reason: "read_only", source });
  1018→      return;
  1019→    }
  1020→    const item = this.findItem(itemId);
  1021→    if (!item) {
  1022→      this.emit("bar_move_rejected", { itemId, start: newStart, end: newEnd, reason: "invalid_item", source });
  1023→      return;
  1024→    }
  1025→    if (newEnd < newStart) {
  1026→      this.emit("bar_move_rejected", { itemId, start: newStart, end: newEnd, reason: "invalid_interval", source });
  1027→      return;
  1028→    }
  1029→    if (this.opts.minDuration > 0 && newEnd - newStart < this.opts.minDuration) {
  1030→      this.emit("bar_move_rejected", { itemId, start: newStart, end: newEnd, reason: "min_duration", source });
  1031→      return;
  1032→    }
  1033→    const oldStart = item.start;
  1034→    const oldEnd = item.end;
  1035→    item.start = newStart;
  1036→    item.end = newEnd;
  1037→    const el = this.barEls.get(itemId);
  1038→    if (el) this.positionBar(el, item);
  1039→    this.emit("bar_moved", { itemId, oldStart, oldEnd, newStart, newEnd, source });
  1040→  }
  1041→
  1042→  resizeBar(itemId: string, endpoint: GanttResizeEndpoint, time: number, source: "pointer" | "keyboard" | "programmatic" = "programmatic"): void {
  1043→    if (this._destroyed) return;
  1044→    if (this.opts.readOnly || !this.opts.resizable) {
  1045→      this.emit("bar_move_rejected", { itemId, endpoint, reason: "read_only", source });
  1046→      return;
  1047→    }
  1048→    const item = this.findItem(itemId);
  1049→    if (!item) {
  1050→      this.emit("bar_move_rejected", { itemId, endpoint, reason: "invalid_item", source });
  1051→      return;
  1052→    }
  1053→    const oldStart = item.start;
  1054→    const oldEnd = item.end;
  1055→    if (endpoint === "start") {
  1056→      const newStart = Math.min(time, item.end - (this.opts.minDuration || 1));
  1057→      if (newStart >= item.end) {
  1058→        this.emit("bar_move_rejected", { itemId, endpoint, reason: "invalid_interval", source });
  1059→        return;
  1060→      }
  1061→      item.start = newStart;
  1062→    } else {
  1063→      const newEnd = Math.max(time, item.start + (this.opts.minDuration || 1));
  1064→      if (newEnd <= item.start) {
  1065→        this.emit("bar_move_rejected", { itemId, endpoint, reason: "invalid_interval", source });
  1066→        return;
  1067→      }
  1068→      item.end = newEnd;
  1069→    }
  1070→    const el = this.barEls.get(itemId);
  1071→    if (el) this.positionBar(el, item);
  1072→    this.emit("bar_resized", { itemId, oldStart, oldEnd, newStart: item.start, newEnd: item.end, endpoint, source });
  1073→  }

  1075→  moveBarToRow(itemId: string, rowId: string, source: "pointer" | "keyboard" | "programmatic" = "programmatic"): void {
  1076→    if (this._destroyed) return;
  1077→    if (this.opts.readOnly || !this.opts.rowMovable) {
  1078→      this.emit("bar_move_rejected", { itemId, targetRowId: rowId, reason: "row_movable_disabled", source });
  1079→      return;
  1080→    }
  1081→    const item = this.findItem(itemId);
  1082→    if (!item) return;
  1083→    const row = this.findRow(rowId);
  1084→    if (!row) {
  1085→      this.emit("bar_move_rejected", { itemId, targetRowId: rowId, reason: "invalid_row", source });
  1086→      return;
  1087→    }
  1088→    const prevRowId = item.rowId ?? "default";
  1089→    if (prevRowId === rowId) return;
  1090→    item.rowId = rowId;
  1091→    this.render();
  1092→    this.emit("bar_row_changed", { itemId, previousRowId: prevRowId, rowId, source });
  1093→  }

  1114→  zoomIn(): void {

  1180→  private onBarPointerDown(e: PointerEvent, item: GanttItem): void {
  1181→    if (this.opts.readOnly || item.disabled) return;
  1182→    const target = e.target as HTMLElement;
  1183→    if (target.closest(".gantt-resize-handle")) return;
  1184→    if (!this.opts.movable) return;
  1185→    e.preventDefault();
  1186→    this.dragItemId = item.id;
  1187→    this.dragMode = "move";
  1188→    this.dragStartX = e.clientX;
  1189→    this.dragOriginalStart = item.start;
  1190→    this.dragOriginalEnd = item.end;
  1191→    this.dragOriginalRowId = item.rowId ?? "default";
  1192→    this.dragHappened = false;
  1193→    (e.target as HTMLElement).setPointerCapture(e.pointerId);
  1194→    const barEl = this.barEls.get(item.id);
  1195→    if (barEl) barEl.classList.add("gantt-dragging");
  1196→    this.emit("drag_started", { itemId: item.id, operation: "move", source: "pointer" });
  1197→
  1198→    const onMove = (ev: PointerEvent) => this.onDragMove(ev);
  1199→    const onUp = (ev: PointerEvent) => {
  1200→      this.onDragEnd(ev, onMove, onUp);
  1201→    };
  1202→    window.addEventListener("pointermove", onMove);
  1203→    window.addEventListener("pointerup", onUp);
  1204→  }

  1206→  private onResizeStart(e: PointerEvent, item: GanttItem, endpoint: GanttResizeEndpoint): void {
  1207→    if (this.opts.readOnly || item.disabled || !this.opts.resizable) return;
  1208→    e.preventDefault();
  1209→    e.stopPropagation();
  1210→    this.dragItemId = item.id;
  1211→    this.dragMode = endpoint === "start" ? "resize-start" : "resize-end";
  1212→    this.dragStartX = e.clientX;
  1213→    this.dragOriginalStart = item.start;
  1214→    this.dragOriginalEnd = item.end;
  1215→    this.dragHappened = false;
  1216→    (e.target as HTMLElement).setPointerCapture(e.pointerId);
  1217→    const handle = e.target as HTMLElement;
  1218→    handle.classList.add("gantt-active");
  1219→    this.emit("drag_started", { itemId: item.id, operation: this.dragMode, source: "pointer" });
  1220→
  1221→    const onMove = (ev: PointerEvent) => this.onDragMove(ev);
  1222→    const onUp = (ev: PointerEvent) => {
  1223→      handle.classList.remove("gantt-active");
  1224→      this.onDragEnd(ev, onMove, onUp);
  1225→    };
  1226→    window.addEventListener("pointermove", onMove);
  1227→    window.addEventListener("pointerup", onUp);

  1230→  private onDragMove(e: PointerEvent): void {
  1231→    if (!this.dragItemId || !this.dragMode) return;
  1232→    const item = this.findItem(this.dragItemId);
  1233→    if (!item) return;
  1234→    const dx = e.clientX - this.dragStartX;
  1235→    if (Math.abs(dx) < 2 && !this.dragHappened) return;
  1236→    this.dragHappened = true;
  1237→
  1238→    const dt = this.xToTime(this.timeToX(this.dragOriginalStart) + dx) - this.dragOriginalStart;
  1239→    const snap = this.opts.snapInterval;
  1240→
  1241→    if (this.dragMode === "move") {
  1242→      const newStart = snapTime(this.dragOriginalStart + dt, snap);
  1243→      const newEnd = snapTime(this.dragOriginalEnd + dt, snap);
  1244→      // Update visual position without committing
  1245→      item.start = newStart;
  1246→      item.end = newEnd;
  1247→      const el = this.barEls.get(item.id);
  1248→      if (el) this.positionBar(el, item);
  1249→      this.emit("bar_move_requested", {
  1250→        itemId: item.id,
  1251→        oldStart: this.dragOriginalStart,
  1252→        oldEnd: this.dragOriginalEnd,
  1253→        start: newStart,
  1254→        end: newEnd,
  1255→        source: "pointer",
  1256→      });
  1257→    } else if (this.dragMode === "resize-start") {
  1258→      const newStart = snapTime(this.dragOriginalStart + dt, snap);
  1259→      if (newStart < this.dragOriginalEnd - (this.opts.minDuration || 1)) {
  1260→        item.start = newStart;
  1261→        const el = this.barEls.get(item.id);
  1262→        if (el) this.positionBar(el, item);
  1263→        this.emit("bar_resize_requested", {
  1264→          itemId: item.id,
  1265→          endpoint: "start",
  1266→          oldStart: this.dragOriginalStart,
  1267→          start: newStart,
  1268→          source: "pointer",
  1269→        });
  1270→      }
  1271→    } else if (this.dragMode === "resize-end") {
  1272→      const newEnd = snapTime(this.dragOriginalEnd + dt, snap);
  1273→      if (newEnd > this.dragOriginalStart + (this.opts.minDuration || 1)) {
  1274→        item.end = newEnd;
  1275→        const el = this.barEls.get(item.id);
  1276→        if (el) this.positionBar(el, item);
  1277→        this.emit("bar_resize_requested", {
  1278→          itemId: item.id,
  1279→          endpoint: "end",
  1280→          oldEnd: this.dragOriginalEnd,
  1281→          end: newEnd,
  1282→          source: "pointer",
  1283→        });
  1284→      }
  1285→    }
  1286→  }
  1287→
  1288→  private onDragEnd(_e: PointerEvent, onMove: (ev: PointerEvent) => void, onUp: (ev: PointerEvent) => void): void {
  1289→    window.removeEventListener("pointermove", onMove);
  1290→    window.removeEventListener("pointerup", onUp);
  1291→    if (!this.dragItemId || !this.dragMode) return;
  1292→    const item = this.findItem(this.dragItemId);
  1293→    const barEl = this.barEls.get(this.dragItemId);
  1294→    if (barEl) barEl.classList.remove("gantt-dragging");
  1295→
  1296→    if (this.dragHappened && item) {
  1297→      if (this.dragMode === "move") {
  1298→        this.emit("bar_moved", {
  1299→          itemId: item.id,
  1300→          oldStart: this.dragOriginalStart,
  1301→          oldEnd: this.dragOriginalEnd,
  1302→          newStart: item.start,
  1303→          newEnd: item.end,
  1304→          source: "pointer",
  1305→        });
  1306→      } else {
  1307→        this.emit("bar_resized", {
  1308→          itemId: item.id,
  1309→          oldStart: this.dragOriginalStart,
  1310→          oldEnd: this.dragOriginalEnd,
  1311→          newStart: item.start,
  1312→          newEnd: item.end,
  1313→          endpoint: this.dragMode === "resize-start" ? "start" : "end",
  1314→          source: "pointer",
  1315→        });
  1316→      }
  1317→    } else if (!this.dragHappened) {
  1318→      this.emit("drag_cancelled", { itemId: this.dragItemId, operation: this.dragMode, source: "pointer" });
  1319→    }
  1320→
  1321→    this.dragItemId = null;
  1322→    this.dragMode = null;
  1323→    this.dragHappened = false;
  1324→  }

  1326→  // ---- Keyboard ----

  1328→  private onBarKeyDown(e: KeyboardEvent, item: GanttItem): void {
  1329→    if (this.opts.readOnly || item.disabled) return;
  1330→    const snap = this.opts.snapInterval || SCALE_TICK[this.opts.scale];
  1331→    let handled = true;
  1332→
  1333→    switch (e.key) {
  1334→      case " ":
  1335→      case "Enter":
  1336→        if (this.opts.selectable) this.selectBar(item.id, "keyboard");
  1337→        else this.emit("bar_activated", { itemId: item.id, source: "keyboard" });
  1338→        break;
  1339→      case "ArrowLeft":
  1340→        if (this.opts.movable && !e.shiftKey) {
  1341→          this.moveBar(item.id, item.start - snap, item.end - snap, "keyboard");
  1342→        } else if (this.opts.resizable && e.shiftKey) {
  1343→          this.resizeBar(item.id, "start", item.start - snap, "keyboard");
  1344→        }
  1345→        break;
  1346→      case "ArrowRight":
  1347→        if (this.opts.movable && !e.shiftKey) {
  1348→          this.moveBar(item.id, item.start + snap, item.end + snap, "keyboard");
  1349→        } else if (this.opts.resizable && e.shiftKey) {
  1350→          this.resizeBar(item.id, "end", item.end + snap, "keyboard");
  1351→        }
  1352→        break;
  1353→      case "ArrowUp":
  1354→        if (this.opts.rowMovable) {
  1355→          const idx = this.rows.findIndex((r) => r.id === (item.rowId ?? "default"));
  1356→          if (idx > 0) this.moveBarToRow(item.id, this.rows[idx - 1].id, "keyboard");
  1357→        }
  1358→        break;
  1359→      case "ArrowDown":
  1360→        if (this.opts.rowMovable) {
  1361→          const idx = this.rows.findIndex((r) => r.id === (item.rowId ?? "default"));
  1362→          if (idx >= 0 && idx < this.rows.length - 1) this.moveBarToRow(item.id, this.rows[idx + 1].id, "keyboard");
  1363→        }
  1364→        break;
  1365→      case "Home":
  1366→        if (this.opts.resizable) this.resizeBar(item.id, "start", this.opts.viewStart, "keyboard");
  1367→        break;
  1368→      case "End":
  1369→        if (this.opts.resizable) this.resizeBar(item.id, "end", this.opts.viewEnd, "keyboard");
  1370→        break;
  1371→      case "Delete":
  1372→      case "Backspace":
  1373→        this.emit("bar_delete_requested", { itemId: item.id, source: "keyboard" });
  1374→        break;
  1375→      case "Escape":
  1376→        this.clearSelection();
  1377→        break;
  1378→      default:
  1379→        handled = false;
  1380→    }
  1381→    if (handled) e.preventDefault();
  1382→  }
