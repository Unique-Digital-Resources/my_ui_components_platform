   423→function formatLabel(mode: ColorRangeMode, value: number): string {
   424→  if (mode === "hue") return `${Math.round(value)}°`;
   425→  if (mode === "alpha") return value.toFixed(2);
   426→  return `${Math.round(value)}%`;
   427→}

   429→function PropertyField({
   430→  label,
   431→  hint,
   432→  value,
   433→  error,
   434→  children,
   435→}: {
   436→  label: string;
   437→  hint?: string;
   438→  value?: string;
   439→  error?: string | null;
   440→  children: React.ReactNode;
   441→}) {
   442→  return (
   443→    <div>
   444→      <div className="mb-1.5 flex items-center justify-between">
   445→        <label className="text-[12px] font-medium text-foreground">{label}</label>
   446→        {value && (
   447→          <span className="font-mono text-[11px] text-brand-lemon">{value}</span>
   448→        )}
   449→      </div>
   450→      {children}
   451→      {error ? (
   452→        <p className="mt-1 text-[11px] text-destructive">{error}</p>
   453→      ) : hint ? (
   454→        <p className="mt-1 text-[11px] text-muted-foreground/70">{hint}</p>
   455→      ) : null}
   456→    </div>
   457→  );
   458→}

   460→function RangeInput({
   461→  min,
   462→  max,
   463→  step,
   464→  value,
   465→  onChange,
   466→}: {
   467→  min: number;
   468→  max: number;
   469→  step: number;
   470→  value: number;
   471→  onChange: (v: number) => void;
   472→}) {
   473→  return (
   474→    <div className="flex items-center gap-2">
   475→      <input
   476→        type="range"
   477→        min={min}
   478→        max={max}
   479→        step={step}
   480→        value={value}
   481→        onChange={(e) => onChange(Number(e.target.value))}
   482→        className="h-1.5 flex-1 cursor-pointer appearance-none rounded-full bg-muted accent-brand-lemon"
   483→        style={{
   484→          background: `linear-gradient(to right, var(--brand-lemon) 0%, var(--brand-lemon) ${
   485→            ((value - min) / (max - min)) * 100

   487→        }}
   488→      />
   489→      <input
   490→        type="number"
   491→        min={min}
   492→        max={max}
   493→        value={value}
   494→        onChange={(e) => onChange(Number(e.target.value))}

   496→      />
   497→    </div>
   498→  );
   499→}

   585→function generateApiCode(): string {
   586→  return `// Public API
   587→
   588→type ColorRangeMode = "hue" | "lightness" | "contrast" | "alpha";
   589→type ColorRangeOrientation = "horizontal" | "vertical";
   590→
   591→interface ColorRangeValueOutput {
   592→  mode: ColorRangeMode;
   593→  value: number;      // 0..360 for hue, 0..100 for lightness/contrast, 0..1 for alpha
   594→  label: string;      // human-readable, e.g. "210°" / "45%" / "0.80"
   595→}
   596→
