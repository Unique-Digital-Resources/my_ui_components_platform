     1→/**
     2→ * KanbanBoard — a reusable Kanban board component (vanilla TypeScript).
     3→ *
     4→ * Spec reference: §14 Minimum viable specification.
     5→ *
     6→ *   - Render configurable columns and ordered cards.
     7→ *   - Accept generic item records with stable identifiers.
     8→ *   - Support horizontal scrolling and responsive column sizing.
     9→ *   - Display optional card content and metadata.
    10→ *   - Add, select, and move cards when enabled.
    11→ *   - Support drag-and-drop and a non-drag move alternative.
    12→ *   - Support reordering within and between columns.
    13→ *   - Handle empty columns and invalid drop targets.
    14→ *   - Define editable, read-only, and disabled behavior.
    15→ *   - Expose configurable card and column styles.
    16→ *   - Emit consistent selection, move, reorder, and rejection events.
    17→ *   - Support keyboard navigation and accessible controls.
    18→ *
    19→ * Architecture: single-file DOM-based renderer with scoped CSS (`.kanban-` prefix).
    20→ * Drag-and-drop uses the HTML5 Drag-and-Drop API. Item ordering is by array
    21→ * position within each column's items list.
    22→ *
    23→ * The board owns a local copy of items and columns; the host application is
    24→ * notified of changes through events and may persist the resulting data.
    25→ */
    26→

   518→// ---- Utilities ----
   519→
   520→function uid(prefix: string): string {
   521→  return `${prefix}-${Math.random().toString(36).slice(2, 9)}-${Date.now().toString(36).slice(-4)}`;
   522→}
   523→
   524→function initials(name: string): string {
   525→  if (!name) return "?";
   526→  const parts = name.trim().split(/\s+/);
   527→  if (parts.length === 1) return parts[0].slice(0, 2).toUpperCase();
   528→  return (parts[0][0] + parts[parts.length - 1][0]).toUpperCase();
   529→}
   530→
   531→function clamp(n: number, min: number, max: number): number {
   532→  return Math.max(min, Math.min(max, n));
   533→}

   867→  private canDropOnColumn(columnId: string): { ok: boolean; reason?: string } {
   868→    const col = this.findColumn(columnId);
   869→    if (!col) return { ok: false, reason: "invalid_target" };
   870→    if (col.collapsed) return { ok: false, reason: "invalid_target" };
   871→    if (col.itemLimit) {
   872→      const count = this.orderedColumnItems(columnId).length;
   873→      // If the dragged item is already in this column, it doesn't add to count
   874→      const isSameColumn = this.dragSourceColumnId === columnId;
   875→      const effectiveCount = isSameColumn ? count : count + 1;
   876→      if (effectiveCount > col.itemLimit) return { ok: false, reason: "column_limit" };
   877→    }
   878→    return { ok: true };
   879→  }

   911→  // ---- Move / reorder ----
   912→
   913→  /**
   914→   * Move an item to a (possibly different) column at a given position.
   915→   * position is the index within the target column's item list; out-of-range
   916→   * values are clamped to the end.
   917→   */
   918→  moveItem(
   919→    itemId: string,
   920→    targetColumnId: string,
   921→    position: number,
   922→    source: "pointer" | "keyboard" | "programmatic" = "programmatic",
   923→  ): void {
   924→    if (this._destroyed) return;
   925→    if (this.opts.mode === "read_only") {
   926→      this.emit("item_move_rejected", { itemId, targetColumnId, position, reason: "read_only", source });
   927→      return;
   928→    }
   929→    const item = this.findItem(itemId);
   930→    if (!item) {
   931→      this.emit("item_move_rejected", { itemId, targetColumnId, position, reason: "invalid_item", source });
   932→      return;
   933→    }
   934→    const targetCol = this.findColumn(targetColumnId);
   935→    if (!targetCol) {
   936→      this.emit("item_move_rejected", { itemId, targetColumnId, position, reason: "invalid_target", source });
   937→      return;
   938→    }
   939→    if (targetCol.collapsed) {
   940→      this.emit("item_move_rejected", { itemId, targetColumnId, position, reason: "invalid_target", source });
   941→      return;
   942→    }
   943→    // Validate cross-column policy
   944→    const sameColumn = item.columnId === targetColumnId;
   945→    if (!sameColumn && !this.opts.moveBetweenColumns) {
   946→      this.emit("item_move_rejected", { itemId, targetColumnId, position, reason: "cross_column_disabled", source });
   947→      return;
   948→    }
   949→    if (sameColumn && !this.opts.reorderWithinColumn) {
   950→      const currentPos = this.orderedColumnItems(targetColumnId).findIndex((it) => it.id === itemId);
   951→      if (position !== currentPos) {
   952→        this.emit("item_move_rejected", { itemId, targetColumnId, position, reason: "reorder_disabled", source });
   953→        return;
   954→      }
   955→    }
   956→    // Item limit check
   957→    if (targetCol.itemLimit) {
   958→      const count = this.orderedColumnItems(targetColumnId).length;
   959→      const effectiveCount = sameColumn ? count : count + 1;
   960→      if (effectiveCount > targetCol.itemLimit) {
   961→        this.emit("item_move_rejected", { itemId, targetColumnId, position, reason: "column_limit", source });
   962→        return;
   963→      }
   964→    }
   965→
   966→    const oldColumnId = item.columnId;
   967→    const oldPosition = this.orderedColumnItems(oldColumnId).findIndex((it) => it.id === itemId);
   968→
   969→    // Remove the item from the master list, then re-insert at the right spot.
   970→    const idx = this.indexOfItem(itemId);
   971→    if (idx < 0) return;
   972→    this.items.splice(idx, 1);
   973→
   974→    // Compute target position in the master list. We need to insert after
   975→    // `position` items in the target column's current list (after removal).
   976→    const targetList = this.orderedColumnItems(targetColumnId);
   977→    const clampedPos = clamp(position, 0, targetList.length);
   978→    let insertAt: number;
   979→    if (clampedPos === 0) {
   980→      // Insert before the first item in target column, or at start if empty
   981→      if (targetList.length === 0) {
   982→        insertAt = this.items.length; // append
   983→      } else {
   984→        const firstTarget = targetList[0];
   985→        insertAt = this.items.findIndex((it) => it.id === firstTarget.id);
   986→        if (insertAt < 0) insertAt = this.items.length;
   987→      }
   988→    } else {
   989→      const ref = targetList[clampedPos - 1];
   990→      const refIdx = this.items.findIndex((it) => it.id === ref.id);
   991→      insertAt = refIdx < 0 ? this.items.length : refIdx + 1;
   992→    }
   993→
   994→    const movedItem: KanbanItem = { ...item, columnId: targetColumnId };
   995→    this.items.splice(insertAt, 0, movedItem);
   996→
   997→    this.render();
   998→    this.emit("item_moved", {
   999→      itemId,
  1000→      sourceColumnId: oldColumnId,
  1001→      targetColumnId,
  1002→      oldPosition,
  1003→      newPosition: clampedPos,
  1004→      source,
  1005→    });
  1006→    if (oldColumnId !== targetColumnId) {
  1007→      this.emit("item_reordered", {
  1008→        itemId,
  1009→        columnId: targetColumnId,
  1010→        oldPosition,
  1011→        newPosition: clampedPos,
  1012→        source,
  1013→      });
  1014→    }
  1015→  }
  1016→
  1017→  reorderItem(itemId: string, position: number, source: "pointer" | "keyboard" | "programmatic" = "programmatic"): void {
  1018→    if (this._destroyed) return;
  1019→    const item = this.findItem(itemId);
  1020→    if (!item) return;
  1021→    const oldPosition = this.orderedColumnItems(item.columnId).findIndex((it) => it.id === itemId);
  1022→    this.moveItem(itemId, item.columnId, position, source);
  1023→    this.emit("item_reordered", {
  1024→      itemId,
  1025→      columnId: item.columnId,
  1026→      oldPosition,
  1027→      newPosition: position,
  1028→      source,
  1029→    });
  1030→  }

  1034→  createItem(columnId: string, item: Partial<KanbanItem>): string {
  1035→    if (this._destroyed) return "";
  1036→    const col = this.findColumn(columnId);
  1037→    if (!col) {
  1038→      this.emit("item_move_rejected", { itemId: null, targetColumnId: columnId, reason: "invalid_target", source: "programmatic" });
  1039→      return "";
  1040→    }
  1041→    const id = item.id ?? uid("item");
  1042→    const newItem: KanbanItem = {
  1043→      id,
  1044→      columnId,
  1045→      title: item.title ?? "Untitled",
  1046→      description: item.description,
  1047→      labels: item.labels ? [...item.labels] : undefined,
  1048→      priority: item.priority,
  1049→      assignee: item.assignee,
  1050→    };
  1051→    // Validate limit
  1052→    if (col.itemLimit) {
  1053→      const count = this.orderedColumnItems(columnId).length;
  1054→      if (count >= col.itemLimit) {
  1055→        this.emit("item_move_rejected", { itemId: id, targetColumnId: columnId, reason: "column_limit", source: "programmatic" });
  1056→        return id;
  1057→      }
  1058→    }
  1059→    this.items.push(newItem);
  1060→    this.render();
  1061→    return id;
  1062→  }

  1067→    if (idx < 0) return;
  1068→    const prev = this.items[idx];
  1069→    this.items[idx] = { ...prev, ...changes, id: prev.id };
  1070→    this.render();
  1071→  }

  1077→    this.items.splice(idx, 1);
  1078→    if (this.selectedItemId === itemId) this.selectedItemId = null;
  1079→    this.render();
  1080→  }

  1102→  setItems(items: KanbanItem[]): void {

  1215→    if (cardEl) cardEl.classList.remove("kanban-dragging");
  1216→    if (!this.dragHappened) {
  1217→      this.emit("drag_cancelled", { itemId: item.id, sourceColumnId: item.columnId, source: "pointer" });
  1218→    }

  1220→    this.dragItemId = null;
  1221→    this.dragSourceColumnId = null;
  1222→    this.dragTargetColumnId = null;
  1223→    this.dragTargetPosition = null;

  1227→  private attachColumnDropHandlers(body: HTMLElement, columnId: string): void {
  1228→    body.addEventListener("dragenter", (e) => {
  1229→      if (!this.dragItemId) return;
  1230→      e.preventDefault();
  1231→      this.updateDropTarget(columnId, e);
  1232→    });
  1233→    body.addEventListener("dragover", (e) => {
  1234→      if (!this.dragItemId) return;
  1235→      e.preventDefault();
  1236→      if (e.dataTransfer) e.dataTransfer.dropEffect = "move";
  1237→      this.updateDropTarget(columnId, e);
  1238→    });
  1239→    body.addEventListener("dragleave", (e) => {
  1240→      if (!this.dragItemId) return;
  1241→      // Only clear if we're leaving the column body entirely (not entering a child)
  1242→      const related = e.relatedTarget as Node | null;
  1243→      if (related && body.contains(related)) return;
  1244→      if (this.dragTargetColumnId === columnId) {
  1245→        this.clearDropPreview();
  1246→        this.dragTargetColumnId = null;
  1247→        this.dragTargetPosition = null;
  1248→      }
  1249→    });
  1250→    body.addEventListener("drop", (e) => {
  1251→      if (!this.dragItemId) return;
  1252→      e.preventDefault();
  1253→      e.stopPropagation();
  1254→      this.dragHappened = true;
  1255→      const targetCol = this.dragTargetColumnId ?? columnId;
  1256→      const targetPos = this.dragTargetPosition ?? this.orderedColumnItems(targetCol).length;
  1257→      const itemId = this.dragItemId;
  1258→      const sourceColumnId = this.dragSourceColumnId;
  1259→      this.clearDropPreview();
  1260→      // Validate
  1261→      const validation = this.canDropOnColumn(targetCol);
  1262→      if (!validation.ok) {
  1263→        this.emit("item_move_rejected", {
  1264→          itemId,
  1265→          sourceColumnId,
  1266→          targetColumnId: targetCol,
  1267→          position: targetPos,
  1268→          reason: validation.reason,
  1269→          source: "pointer",
  1270→        });
  1271→        this.dragItemId = null;
  1272→        this.dragSourceColumnId = null;
  1273→        this.dragTargetColumnId = null;
  1274→        this.dragTargetPosition = null;
  1275→        return;
  1276→      }
  1277→      this.moveItem(itemId, targetCol, targetPos, "pointer");
  1278→      this.dragItemId = null;
  1279→      this.dragSourceColumnId = null;
  1280→      this.dragTargetColumnId = null;
  1281→      this.dragTargetPosition = null;
  1282→    });
  1283→  }

  1285→  private updateDropTarget(columnId: string, e: DragEvent): void {
  1286→    const validation = this.canDropOnColumn(columnId);
  1287→    // Update column highlight
  1288→    this.columnEls.forEach((el, id) => {
  1289→      el.classList.remove("kanban-drop-target", "kanban-drop-invalid");
  1290→      if (id === columnId) {
  1291→        if (validation.ok) el.classList.add("kanban-drop-target");
  1292→        else el.classList.add("kanban-drop-invalid");
  1293→      }
  1294→    });
  1295→
  1296→    const prevTarget = this.dragTargetColumnId;
  1297→    const prevPos = this.dragTargetPosition;
  1298→    this.dragTargetColumnId = columnId;
  1299→
  1300→    // Compute position
  1301→    const pos = this.computeDropPosition(e, columnId);
  1302→    this.dragTargetPosition = pos;
  1303→
  1304→    if (prevTarget !== columnId || prevPos !== pos) {
  1305→      this.emit("drop_target_changed", {
  1306→        itemId: this.dragItemId,
  1307→        targetColumnId: columnId,
  1308→        targetPosition: pos,
  1309→        source: "pointer",
  1310→      });
  1311→    }
  1312→
  1313→    if (this.opts.showDropPreview && validation.ok) {
  1314→      this.renderDropPreview(columnId, pos);
  1315→    } else {
  1316→      this.clearDropPreview();
  1317→    }
  1318→  }

  1320→  private computeDropPosition(e: DragEvent, columnId: string): number {
  1321→    const body = this.columnEls.get(columnId)?.querySelector(".kanban-column-body") as HTMLElement | null;
  1322→    if (!body) return 0;
  1323→    const cards = Array.from(body.querySelectorAll(".kanban-card:not(.kanban-dragging)")) as HTMLElement[];
  1324→    if (cards.length === 0) return 0;
  1325→    const isVertical = this.opts.layout === "vertical";
  1326→    const refPos = isVertical ? e.clientX : e.clientY;
  1327→    for (let i = 0; i < cards.length; i++) {
  1328→      const rect = cards[i].getBoundingClientRect();
  1329→      const mid = isVertical ? rect.left + rect.width / 2 : rect.top + rect.height / 2;
  1330→      if (refPos < mid) return i;
  1331→    }
  1332→    return cards.length;
  1333→  }

  1378→  // ---- Keyboard ----

  1380→  private onCardKeyDown(e: KeyboardEvent, item: KanbanItem): void {
  1381→    // Don't intercept typing in inputs
  1382→    const target = e.target as HTMLElement;
  1383→    if (target.closest("input, textarea, select")) return;
  1384→
  1385→    switch (e.key) {
  1386→      case "Enter": {
  1387→        e.preventDefault();
  1388→        this.emit("item_activated", { itemId: item.id, source: "keyboard" });
  1389→        break;
  1390→      }
  1391→      case " ": {
  1392→        e.preventDefault();
  1393→        if (this.opts.selectable) {
  1394→          this.selectItem(item.id, "keyboard");
  1395→        }
  1396→        break;
  1397→      }
  1398→      case "m":
  1399→      case "M": {
  1400→        if (this.opts.mode !== "editable" || !this.opts.moveBetweenColumns) return;
  1401→        e.preventDefault();
  1402→        const cardEl = this.cardEls.get(item.id);
  1403→        if (cardEl) this.openMovePopover(item.id, cardEl);
  1404→        break;
  1405→      }
  1406→      case "ArrowUp": {
  1407→        if (this.opts.mode !== "editable" || !this.opts.reorderWithinColumn) return;
  1408→        e.preventDefault();
  1409→        const list = this.orderedColumnItems(item.columnId);
  1410→        const i = list.findIndex((it) => it.id === item.id);
  1411→        if (i > 0) {
  1412→          this.moveItem(item.id, item.columnId, i - 1, "keyboard");
  1413→          requestAnimationFrame(() => this.cardEls.get(item.id)?.focus());
  1414→        }
  1415→        break;
  1416→      }
  1417→      case "ArrowDown": {
  1418→        if (this.opts.mode !== "editable" || !this.opts.reorderWithinColumn) return;
  1419→        e.preventDefault();
  1420→        const list = this.orderedColumnItems(item.columnId);
  1421→        const i = list.findIndex((it) => it.id === item.id);
  1422→        if (i < list.length - 1) {
  1423→          this.moveItem(item.id, item.columnId, i + 1, "keyboard");
  1424→          requestAnimationFrame(() => this.cardEls.get(item.id)?.focus());
  1425→        }
  1426→        break;
  1427→      }
  1428→      case "ArrowRight": {
  1429→        if (this.opts.mode !== "editable" || !this.opts.moveBetweenColumns) return;
  1430→        e.preventDefault();
  1431→        const i = this.columns.findIndex((c) => c.id === item.columnId);
  1432→        for (let k = i + 1; k < this.columns.length; k++) {
  1433→          const nextCol = this.columns[k];
  1434→          if (!nextCol.collapsed) {
  1435→            this.moveItem(item.id, nextCol.id, this.orderedColumnItems(nextCol.id).length, "keyboard");
  1436→            requestAnimationFrame(() => this.cardEls.get(item.id)?.focus());
  1437→            break;
  1438→          }
  1439→        }
  1440→        break;
  1441→      }
  1442→      case "ArrowLeft": {
  1443→        if (this.opts.mode !== "editable" || !this.opts.moveBetweenColumns) return;
  1444→        e.preventDefault();
  1445→        const i = this.columns.findIndex((c) => c.id === item.columnId);
  1446→        for (let k = i - 1; k >= 0; k--) {
  1447→          const prevCol = this.columns[k];
  1448→          if (!prevCol.collapsed) {
  1449→            this.moveItem(item.id, prevCol.id, this.orderedColumnItems(prevCol.id).length, "keyboard");
  1450→            requestAnimationFrame(() => this.cardEls.get(item.id)?.focus());
  1451→            break;
  1452→          }
  1453→        }
  1454→        break;
  1455→      }
  1456→      case "Delete":
  1457→      case "Backspace": {
  1458→        if (this.opts.mode !== "editable") return;
  1459→        e.preventDefault();
  1460→        this.emit("item_delete_requested", { itemId: item.id, source: "keyboard" });
  1461→        break;
  1462→      }
  1463→      case "Escape": {
  1464→        e.preventDefault();
  1465→        this.closePopover();
  1466→        this.clearSelection();
  1467→        break;
  1468→      }
  1469→    }
  1470→  }

  1606→    set.forEach((h) => {
  1607→      try { h(fullPayload); } catch { /* swallow handler errors */ }
  1608→    });
  1609→  }
