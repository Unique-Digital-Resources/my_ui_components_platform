   128→// ---- Playground ---------------------------------------------------------
   129→
   130→export function BoardPlayground() {
   131→  const meta = getComponent("board")!;
   132→  const [state, setState] = React.useState<PlaygroundState>(DEFAULTS);
   133→  const [items, setItems] = React.useState<BoardItem[]>([]);
   134→  const [selectedIds, setSelectedIds] = React.useState<string[]>([]);
   135→  const [view, setView] = React.useState<BoardView>({ panX: 0, panY: 0, zoom: 1 });
   136→  const [activeTab, setActiveTab] = React.useState<"javascript" | "html" | "css" | "api">("javascript");
   137→  const [registryVersion, setRegistryVersion] = React.useState(0);
   138→  const [rotationValue, setRotationValue] = React.useState(0);
   139→
   140→  const boardRef = React.useRef<import("@/lib/components/board").Board | null>(null);
   141→  const liveAreaRef = React.useRef<HTMLDivElement>(null);
   142→  const handlerRef = React.useRef<ShortcutHandler | null>(null);
   143→
   144→  // Register the built-in commands on mount.
   145→  React.useEffect(() => {
   146→    registerBuiltinCommands();
   147→    setRegistryVersion((v) => v + 1);
   148→  }, []);
   149→
   150→  // Seed a few starter items once the board is created so the canvas isn't empty.
   151→  React.useEffect(() => {
   152→    const board = boardRef.current;
   153→    if (!board) return;
   154→    // Only seed if there are no items yet.
   155→    if (board.getItems().length > 0) return;
   156→    board.addItem("rect", -120, -40, 120, 90, 0, "#D9FF66");
   157→    board.addItem("circle", 60, -60, 100, 100, 0, "#B9A0FF");
   158→    board.addItem("triangle", -40, 80, 110, 100, 0, "#FF858B");
   159→    setItems(board.getItems());
   160→    // Intentionally empty deps — runs once after the board mounts.
   161→  }, []);
   162→
   163→  // ---- Action handlers ----
   164→  //
   165→  // All actions read from refs (boardRef, liveAreaRef) and call stable
   166→  // setState setters, so their behavior is independent of render scope.
   167→  // We store the latest versions in an actionsRef (updated via a render-
   168→  // scoped effect to satisfy the React Compiler's "no ref writes during
   169→  // render" rule) so the ShortcutHandler — created once on mount — always
   170→  // dispatches to the freshest implementation.
   171→
   172→  const actionsRef = React.useRef<{
   173→    addShape: (type: ShapeType) => void;
   174→    deleteSelected: () => void;
   175→    fitAll: () => void;
   176→    fitSelection: () => void;
   177→    align: (type: AlignType) => void;
   178→    distribute: (type: DistributeType) => void;
   179→  }>({
   180→    addShape: () => {},
   181→    deleteSelected: () => {},
   182→    fitAll: () => {},
   183→    fitSelection: () => {},
   184→    align: () => {},
   185→    distribute: () => {},
   186→  });
   187→
   188→  const addShape = (type: ShapeType) => actionsRef.current.addShape(type);
   189→  const deleteSelected = () => actionsRef.current.deleteSelected();
   190→  const fitAll = () => actionsRef.current.fitAll();
   191→  const fitSelection = () => actionsRef.current.fitSelection();
   192→  const align = (type: AlignType) => actionsRef.current.align(type);
   193→  const distribute = (type: DistributeType) => actionsRef.current.distribute(type);
   194→
   195→  // Keep the actions ref fresh after every render.
   196→  React.useEffect(() => {
   197→    actionsRef.current.addShape = (type: ShapeType) => {
   198→      const board = boardRef.current;
   199→      const liveArea = liveAreaRef.current;
   200→      if (!board) return;
   201→      let cx = 0;
   202→      let cy = 0;
   203→      if (liveArea) {
   204→        const v = board.getView();
   205→        const w = liveArea.clientWidth;
   206→        const h = liveArea.clientHeight;
   207→        cx = (w / 2 - v.panX) / v.zoom;
   208→        cy = (h / 2 - v.panY) / v.zoom;
   209→      }
   210→      // Random offset near the center.
   211→      const offX = (Math.random() - 0.5) * 140;
   212→      const offY = (Math.random() - 0.5) * 140;
   213→      const color = SHAPE_COLORS[Math.floor(Math.random() * SHAPE_COLORS.length)];
   214→      const id = board.addItem(type, cx + offX, cy + offY, DEFAULT_SHAPE_SIZE, DEFAULT_SHAPE_SIZE, 0, color);
   215→      board.selectItem(id, false);
   216→      setItems(board.getItems());
   217→      setSelectedIds(board.getSelectedIds());
   218→    };
   219→    actionsRef.current.deleteSelected = () => {
   220→      const board = boardRef.current;
   221→      if (!board) return;
   222→      const ids = board.getSelectedIds();
   223→      if (ids.length === 0) return;
   224→      for (const id of ids) board.removeItem(id);
   225→      setItems(board.getItems());
   226→      setSelectedIds(board.getSelectedIds());
   227→    };
   228→    actionsRef.current.fitAll = () => {
   229→      boardRef.current?.zoomToFit();
   230→    };
   231→    actionsRef.current.fitSelection = () => {
   232→      boardRef.current?.zoomToFitSelection();
   233→    };
   234→    actionsRef.current.align = (type: AlignType) => {
   235→      const board = boardRef.current;
   236→      if (!board) return;
   237→      if (board.getSelectedIds().length < 2) {
   238→        toast.message("Select 2+ items to align");
   239→        return;
   240→      }
   241→      board.align(type);
   242→      setItems(board.getItems());
   243→    };
   244→    actionsRef.current.distribute = (type: DistributeType) => {
   245→      const board = boardRef.current;
   246→      if (!board) return;
   247→      if (board.getSelectedIds().length < 3) {
   248→        toast.message("Select 3+ items to distribute");
   249→        return;
   250→      }
   251→      board.distribute(type);
   252→      setItems(board.getItems());
   253→    };
   254→  });
   255→
   256→  // ---- Shortcut handler (scoped to board command IDs) ----
   257→
   258→  React.useEffect(() => {
   259→    const handler = new ShortcutHandler([...BOARD_COMMAND_IDS]);
   260→    handler.on("board.add-circle", () => actionsRef.current.addShape("circle"));
   261→    handler.on("board.add-rect", () => actionsRef.current.addShape("rect"));
   262→    handler.on("board.add-triangle", () => actionsRef.current.addShape("triangle"));
   263→    handler.on("board.delete-selected", () => actionsRef.current.deleteSelected());
   264→    handler.on("board.fit-all", () => actionsRef.current.fitAll());
   265→    handler.on("board.fit-selection", () => actionsRef.current.fitSelection());
   266→    handler.on("board.align-left", () => actionsRef.current.align("left"));
   267→    handler.on("board.align-center-h", () => actionsRef.current.align("center-h"));
   268→    handler.on("board.align-right", () => actionsRef.current.align("right"));
   269→    handler.on("board.align-top", () => actionsRef.current.align("top"));
   270→    handler.on("board.align-center-v", () => actionsRef.current.align("center-v"));
   271→    handler.on("board.align-bottom", () => actionsRef.current.align("bottom"));
   272→    handler.on("board.distribute-h", () => actionsRef.current.distribute("horizontal"));
   273→    handler.on("board.distribute-v", () => actionsRef.current.distribute("vertical"));
   274→    handler.attach();
   275→    handlerRef.current = handler;
   276→    return () => {
   277→      handler.destroy();
   278→      handlerRef.current = null;
   279→    };
   280→    // Intentionally empty deps — the handler reads through actionsRef so
   281→    // it always dispatches to the latest implementation without re-attaching.
   282→  }, []);
   283→
   284→  // ---- State updaters ----
   285→
   286→  const update = React.useCallback(
   287→    <K extends keyof PlaygroundState>(key: K, value: PlaygroundState[K]) => {
   288→      setState((s) => ({ ...s, [key]: value }));
   289→    },
   290→    [],
   291→  );
   292→
   293→  // Sync the rotation input with the first selected item's rotation whenever the selection changes.
   294→  React.useEffect(() => {
   295→    if (selectedIds.length === 0) return;
   296→    const board = boardRef.current;
   297→    if (!board) return;
   298→    const first = board.getItems().find((i) => i.id === selectedIds[0]);
   299→    if (first) setRotationValue(first.rotation);
   300→  }, [selectedIds]);
   301→
   302→  // Apply a rotation delta to all selected items and update the rotation input.
   303→  const applyRotation = (newRotation: number) => {
   304→    const board = boardRef.current;
   305→    if (!board || selectedIds.length === 0) return;
   306→    const first = board.getItems().find((i) => i.id === selectedIds[0]);
   307→    if (!first) return;
   308→    const delta = newRotation - first.rotation;
   309→    if (Math.abs(delta) < 0.01) return;
   310→    board.rotateSelected(delta);
   311→    setRotationValue(newRotation);
   312→    setItems(board.getItems());
   313→  };
   314→
   315→  const reset = () => {
   316→    setState(DEFAULTS);
   317→    const board = boardRef.current;
   318→    if (board) {
   319→      // Remove all items, then re-seed.
   320→      for (const it of board.getItems()) board.removeItem(it.id);
   321→      board.addItem("rect", -120, -40, 120, 90, 0, "#D9FF66");
   322→      board.addItem("circle", 60, -60, 100, 100, 0, "#B9A0FF");
   323→      board.addItem("triangle", -40, 80, 110, 100, 0, "#FF858B");
   324→      board.clearSelection();
   325→      board.setView(0, 0, 1);
   326→      setItems(board.getItems());
   327→      setSelectedIds([]);
   328→      setView(board.getView());
   329→      setRotationValue(0);
   330→    }
   331→    toast.success("Reset to defaults");
   332→  };
   333→
   334→  // ---- Code generation ----
   335→
   336→  const jsCode = React.useMemo(() => generateJsCode(state), [state]);
   337→  const htmlCode = React.useMemo(() => generateHtmlCode(), []);
   338→  const cssCode = React.useMemo(() => generateCssCode(), []);
   339→  const apiCode = React.useMemo(() => generateApiCode(), []);
   340→
   341→  const boardCommands = React.useMemo(
   342→    () => commandRegistry.byIds([...BOARD_COMMAND_IDS]),
   343→    [registryVersion],
   344→  );
   345→
   346→  // ---- Render ----
   347→
   348→  return (
   349→    <div>
   350→      {/* Page header */}
   351→      <div className="border-b border-border bg-background">
   352→        <div className="mx-auto w-full max-w-[1400px] px-6 pt-6 pb-8 lg:px-10">
   353→          <Breadcrumbs />
   354→          <div className="mt-6 flex flex-col gap-4 lg:flex-row lg:items-end lg:justify-between">
   355→            <div>
   356→              <PageOverline>Components / Layout</PageOverline>
   357→              <h1 className="text-3xl font-semibold tracking-tight text-foreground sm:text-4xl">
   358→                {meta.name}
   359→              </h1>
   360→              <p className="mt-2 max-w-2xl text-[15px] text-muted-foreground">
   361→                {meta.longDescription}
   362→              </p>
   363→            </div>
   364→            <div className="flex flex-wrap items-center gap-2">
   365→              {meta.tags.map((tag) => (
   366→                <span
   367→                  key={tag}
   368→                  className="inline-flex items-center gap-1.5 rounded-md border border-border bg-muted/30 px-2.5 py-1 text-[11px] font-mono text-muted-foreground"
   369→                >
   370→                  {tag}
   371→                </span>
   372→              ))}
   373→              <span className="inline-flex items-center gap-1.5 rounded-md border border-brand-lemon/30 bg-brand-lemon/10 px-2.5 py-1 text-[11px] font-mono text-brand-lemon">
   374→                <span className="h-1.5 w-1.5 rounded-full bg-brand-lemon" />
   375→                {meta.status}
   376→              </span>
   377→            </div>
   378→          </div>
   379→        </div>
   380→      </div>
   381→
   382→      {/* Playground workspace */}
   383→      <div className="mx-auto w-full max-w-[1400px] px-6 py-8 lg:px-10">
   384→        <div className="grid grid-cols-1 gap-6 lg:grid-cols-[1fr_380px] lg:items-start">
   385→          {/* Live canvas panel */}
   386→          <section aria-label="Board live canvas" className="panel relative flex h-[600px] flex-col overflow-hidden">
   387→            {/* Toolbar */}
   388→            <div className="flex flex-col gap-2 border-b border-border px-3 py-2.5">
   389→              <div className="flex flex-wrap items-center gap-1.5">
   390→                <ToolbarButton onClick={() => addShape("circle")} title="Add Circle (1)" variant="primary">
   391→                  <Circle className="h-3.5 w-3.5" />
   392→                  <span>Circle</span>
   393→                </ToolbarButton>
   394→                <ToolbarButton onClick={() => addShape("rect")} title="Add Rectangle (2)" variant="primary">
   395→                  <Square className="h-3.5 w-3.5" />
   396→                  <span>Rect</span>
   397→                </ToolbarButton>
   398→                <ToolbarButton onClick={() => addShape("triangle")} title="Add Triangle (3)" variant="primary">
   399→                  <Triangle className="h-3.5 w-3.5" />
   400→                  <span>Triangle</span>
   401→                </ToolbarButton>
   402→                <ToolbarDivider />
   403→                <ToolbarButton onClick={deleteSelected} title="Delete Selected (Delete)" variant="danger">
   404→                  <Trash2 className="h-3.5 w-3.5" />
   405→                  <span>Delete</span>
   406→                </ToolbarButton>
   407→                <ToolbarDivider />
   408→                <ToolbarButton onClick={fitAll} title="Zoom to Fit All (Cmd+0)">
   409→                  <Maximize2 className="h-3.5 w-3.5" />
   410→                  <span>Fit All</span>
   411→                </ToolbarButton>
   412→                <ToolbarButton onClick={fitSelection} title="Zoom to Fit Selection (Shift+Cmd+0)">
   413→                  <Scan className="h-3.5 w-3.5" />
   414→                  <span>Fit Selection</span>
   415→                </ToolbarButton>
   416→                <div className="ml-auto flex items-center gap-1">
   417→                  <ToolbarButton onClick={reset} title="Reset to defaults">
   418→                    <RotateCcw className="h-3.5 w-3.5" />
   419→                    <span>Reset</span>
   420→                  </ToolbarButton>
   421→                </div>
   422→              </div>
   423→              {/* Align + Distribute row */}
   424→              <div className="flex flex-wrap items-center gap-1.5">
   425→                <span className="eyebrow mr-1">Align</span>
   426→                {ALIGN_BUTTONS.map((b) => {
   427→                  const Icon = b.icon;
   428→                  return (
   429→                    <IconButton
   430→                      key={b.type}
   431→                      onClick={() => align(b.type)}
   432→                      title={`${b.label} (${formatShortcut(commandRegistry.get(b.cmdId)?.shortcut ?? { keys: "" })})`}
   433→                      aria-label={b.label}
   434→                    >
   435→                      <Icon className="h-3.5 w-3.5" />
   436→                    </IconButton>
   437→                  );
   438→                })}
   439→                <ToolbarDivider />
   440→                <span className="eyebrow mr-1">Distribute</span>
   441→                {DISTRIBUTE_BUTTONS.map((b) => {
   442→                  const Icon = b.icon;
   443→                  return (
   444→                    <IconButton
   445→                      key={b.type}
   446→                      onClick={() => distribute(b.type)}
   447→                      title={`${b.label} (${formatShortcut(commandRegistry.get(b.cmdId)?.shortcut ?? { keys: "" })})`}
   448→                      aria-label={b.label}
   449→                    >
   450→                      <Icon className="h-3.5 w-3.5" />
   451→                    </IconButton>
   452→                  );
   453→                })}
   454→              </div>
   455→            </div>
   456→
   457→            {/* Live Board canvas — expands to fill the remaining panel space */}
   458→            <div ref={liveAreaRef} className="relative flex-1 w-full">
   459→              <BoardReact
   460→                className="absolute inset-0"
   461→                gridMode={state.gridMode}
   462→                gridSize={state.gridSize}
   463→                subGridDivisions={state.subGridDivisions}
   464→                snapToGrid={state.snapToGrid}
   465→                snapToItems={state.snapToItems}
   466→                transformMode={state.transformMode}
   467→                latticeDisplay={state.latticeDisplay}
   468→                selectionPivot={state.selectionPivot}
   469→                onSelection={(e) => setSelectedIds(e.ids)}
   470→                onChange={() => setItems(boardRef.current?.getItems() ?? [])}
   471→                onView={(v) => setView(v)}
   472→                instanceRef={boardRef}
   473→              />
   474→              <div className="pointer-events-none absolute bottom-2 left-1/2 -translate-x-1/2 rounded-full bg-background/70 px-3 py-1 text-[10px] text-muted-foreground backdrop-blur">
   475→                Click to select. Shift-click for multi. Drag empty space for rubber-band. Space-drag to pan. Cmd+scroll to zoom. Drag the lemon rotation handle above the selection to rotate; drag a white square handle to resize.
   476→              </div>
   477→            </div>
   478→
   479→            {/* Output — item list */}
   480→            <div className="border-t border-border p-4">
   481→              <div className="mb-3 flex items-center justify-between">
   482→                <span className="eyebrow">Output — items</span>
   483→                <span className="font-mono text-[11px] text-muted-foreground">
   484→                  {items.length} item{items.length === 1 ? "" : "s"} · {selectedIds.length} selected
   485→                </span>
   486→              </div>
   487→              {items.length === 0 ? (
